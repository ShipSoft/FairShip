31// Keep the nominal bin width and reserve one visible bin at each end.
32template <typename Histogram>
33Histogram* WithFlowBins(TDirectory* dir, const char* name, const char* title,
34 int bins, double low, double high) {
35 const double width = (high - low) / bins;
36 auto* histogram =
37 new Histogram(name, title, bins + 2, low - width, high + width);
38 histogram->SetDirectory(dir);
39 return histogram;
40}

42template <typename Histogram>
43Histogram* WithFlowBins(TDirectory* dir, const char* name, const char* title,
44 int nx, double xmin, double xmax, int ny, double ymin,
45 double ymax) {
46 const double dx = (xmax - xmin) / nx, dy = (ymax - ymin) / ny;
47 auto* histogram = new Histogram(name, title, nx + 2, xmin - dx, xmax + dx,
48 ny + 2, ymin - dy, ymax + dy);
49 histogram->SetDirectory(dir);
50 return histogram;
51}

53double VisibleFlowValue(const TAxis* axis, double value) {
54 if (value < axis->GetBinLowEdge(2)) return axis->GetBinCenter(1);
55 if (value >= axis->GetBinLowEdge(axis->GetNbins()))
56 return axis->GetBinCenter(axis->GetNbins());
57 return value;
58}

89int KinematicBin(double value, const double* edges, unsigned bins) {
90 if (!std::isfinite(value) || value < edges[0] || value > edges[bins])
91 return -1;
92 for (unsigned bin = 0; bin < bins; ++bin)
93 if (value < edges[bin + 1] || bin + 1 == bins) return bin;
94 return -1;
95}

109void MuDISFilter::SetFilterEfficiencyBinning(double xmin, double xmax,
110 unsigned xbins, double ymin,
111 double ymax, unsigned ybins,
112 double zmin, double zmax,
113 unsigned zbins) {
114 const auto symmetricAboutZero = [](double low, double high) {
115 return std::abs(low + high) <=
116 1.e-12 * std::max({1., std::abs(low), std::abs(high)});
117 };
118 if (!std::isfinite(xmin) || !std::isfinite(xmax) || !std::isfinite(ymin) ||
119 !std::isfinite(ymax) || !std::isfinite(zmin) || !std::isfinite(zmax) ||
120 xmin >= xmax || ymin >= ymax || zmin >= zmax || xbins == 0 ||
121 ybins == 0 || zbins == 0 || xbins % 2 == 0 || ybins % 2 == 0 ||
122 !symmetricAboutZero(xmin, xmax) || !symmetricAboutZero(ymin, ymax))
123 throw std::invalid_argument(
124 "Filter-efficiency x/y axes must be symmetric about zero with an odd "
125 "positive number of bins; all axis bounds must increase");
126 fEfficiencyXYBounds = {xmin, xmax, ymin, ymax};
127 fEfficiencyXBins = xbins;
128 fEfficiencyYBins = ybins;
129 fEfficiencyZBounds = {zmin, zmax};
130 fEfficiencyZBins = zbins;
131}

236void MuDISFilter::BookFilterEfficiencyHistograms(TDirectory* dir) {
237 dir->cd();
238 fFilterEfficiency = {};
239 const double xmin = fEfficiencyXYBounds[0], xmax = fEfficiencyXYBounds[1];
240 const double ymin = fEfficiencyXYBounds[2], ymax = fEfficiencyXYBounds[3];
241 for (unsigned ip = 0; ip < kMomentumBins; ++ip)
242 for (unsigned ipt = 0; ipt < kPtBins; ++ipt)
243 for (unsigned imat = 0; imat < kEfficiencyMaterials; ++imat) {
244 const TString tag =
245 Form("p%u_pt%u_%s", ip, ipt, MatTypeStr[imat].Data());
246 const TString title = Form(
247 "DIS filter efficiency, %.0f #leq p < %.0f GeV, "
248 "%.0f #leq p_{T} < %.0f GeV - %s",
249 kMomentumEdges[ip], kMomentumEdges[ip + 1], kPtEdges[ipt],
250 kPtEdges[ipt + 1], MatTypeStr[imat].Data());
251 const TString axTitles = ";x_{#mu,start} [cm];y_{#mu,start} [cm]";
252 auto*& all = fFilterEfficiency.allXY[ip][ipt][imat];
253 auto*& passed = fFilterEfficiency.passedXY[ip][ipt][imat];
254 all = new TH2D(Form("filter_efficiency_all_%s", tag.Data()),
255 title + axTitles, fEfficiencyXBins, xmin, xmax,
256 fEfficiencyYBins, ymin, ymax);
257 passed = new TH2D(Form("filter_efficiency_passed_%s", tag.Data()),
258 title + axTitles, fEfficiencyXBins, xmin, xmax,
259 fEfficiencyYBins, ymin, ymax);
260 all->SetDirectory(dir);
261 all->Sumw2();
262 passed->SetDirectory(dir);
263 passed->Sumw2();
264 if (!fDetailedHistograms) continue;
265 auto& allZByX = fFilterEfficiency.allZ[ip][ipt];
266 auto& passedZByX = fFilterEfficiency.passedZ[ip][ipt];
267 allZByX.resize(fEfficiencyXBins);
268 passedZByX.resize(fEfficiencyXBins);
269 for (unsigned ix = 0; ix < fEfficiencyXBins; ++ix) {
270 allZByX[ix].resize(fEfficiencyYBins);
271 passedZByX[ix].resize(fEfficiencyYBins);
272 for (unsigned iy = 0; iy < fEfficiencyYBins; ++iy) {
273 auto*& allZ = fFilterEfficiency.allZ[ip][ipt][ix][iy][imat];
274 auto*& passedZ = fFilterEfficiency.passedZ[ip][ipt][ix][iy][imat];
275 allZ = new TH1D(
276 Form("filter_efficiency_z_all_%s_x%u_y%u", tag.Data(), ix, iy),
277 title + Form(";z_{DIS} [cm];Weighted DIS events"),
278 fEfficiencyZBins, fEfficiencyZBounds[0], fEfficiencyZBounds[1]);
279 passedZ = new TH1D(Form("filter_efficiency_z_passed_%s_x%u_y%u",
280 tag.Data(), ix, iy),
281 allZ->GetTitle(), fEfficiencyZBins,
282 fEfficiencyZBounds[0], fEfficiencyZBounds[1]);
283 allZ->SetDirectory(dir);
284 allZ->Sumw2();
285 passedZ->SetDirectory(dir);
286 passedZ->Sumw2();
287 }
288 }
289 }

292void MuDISFilter::FillFilterEfficiencyHistograms(unsigned material,
293 double momentum, double pt,
294 double x, double y, double z,
295 double weight, bool passed) {
296 if (material >= kEfficiencyMaterials || !std::isfinite(weight)) return;
297 const int ip = KinematicBin(momentum, kMomentumEdges.data(), kMomentumBins);
298 const int ipt = KinematicBin(pt, kPtEdges.data(), kPtBins);
299 if (ip < 0 || ipt < 0) return;
300 auto* all = fFilterEfficiency.allXY[ip][ipt][material];
301 auto* accepted = fFilterEfficiency.passedXY[ip][ipt][material];
302 all->Fill(x, y, weight);
303 if (passed) accepted->Fill(x, y, weight);
304 if (!fDetailedHistograms) return;
305 const int ix = all->GetXaxis()->FindFixBin(x) - 1;
306 const int iy = all->GetYaxis()->FindFixBin(y) - 1;
307 if (ix < 0 || ix >= static_cast<int>(fEfficiencyXBins) || iy < 0 ||
308 iy >= static_cast<int>(fEfficiencyYBins))
309 return;
310 fFilterEfficiency.allZ[ip][ipt][ix][iy][material]->Fill(z, weight);
311 if (passed)
312 fFilterEfficiency.passedZ[ip][ipt][ix][iy][material]->Fill(z, weight);
313}

336Bool_t MuDISFilter::InitFiles(const std::vector<std::string>& fileNames,
337 const int startEvent) {
338 if (startEvent < 0) return kFALSE;
339 fstartEvt = startEvent;
340 if (fileNames.empty()) {
341 LOG(error) << "MuDISFilter: no input files provided. "
342 << "Check the -f/--inputFile argument or input file glob.";
343 return kFALSE;
344 }
345 for (const auto& fileName : fileNames) {
346 if (fileName.empty()) {
347 LOG(error) << "MuDISFilter: received an empty input file name. "
348 << "Check the -f/--inputFile argument.";
349 return kFALSE;
350 }
351 }
352
353 for (const auto& name : fileNames) {
354 std::unique_ptr<TFile> file(TFile::Open(name.c_str(), "READ"));
355 TTree* tree = nullptr;
356 if (file && !file->IsZombie()) file->GetObject("MuonDIS", tree);
357 if (!tree) {
358 LOG(error) << "MuDISFilter: missing MuonDIS tree in " << name;
359 return kFALSE;
360 }
361 MuonInBranches check;
362 if (!check.Setup(tree)) return kFALSE;
363 tree->ResetBranchAddresses();
364 }
365
366 {
367 delete ftree;
368 ftree = new TChain("MuonDIS");
369 for (const auto& name : fileNames) {
370 if (!ftree->Add(name.c_str())) return kFALSE;
371 }
372 Long64_t treeEvts = ftree->GetEntries();
373 LOG(info) << "Reading " << treeEvts << " entries.";
374 bool ok = finEv.Setup(ftree);
375
376 if (!ok) {
377 LOG(error) << "MuDISFilter: failed to bind one or more required branches";
378 return kFALSE;
379 }
380 LOG(info) << "MuDISFilter: Initialization successful.";
381 return kTRUE;
382 }
383 return kFALSE;
384}

392void MuDISFilter::process_file(const std::vector<std::string>& input,
393 const std::string& output) {
394 if (!fFilter && fFilterOption != 0 && !fUseDetectorAcceptance)
395 throw std::runtime_error(
396 "Filter options 1 and 2 require detector acceptance");
397 if (!fFilter && fUseDetectorAcceptance && !fPropagator)
398 throw std::runtime_error(
399 "Configure detector acceptance with geometry and field map first");
400
401 Bool_t treeOK = InitFiles(input, fstartEvt);
402
403 if (!treeOK) {
404 throw std::runtime_error("MuDISFilter: failed to initialize input files");
405 }
406
407 if (!fFilter && fUseDetectorAcceptance && fPropagator->HasMuonShieldField()) {
408 // A separate, vertex-only reader preserves the filtering chain's branch
409 // addresses and cache learning. Restrict the scan to the requested entries.
410 TChain vertices("MuonDIS");
411 for (const auto& name : input) vertices.Add(name.c_str());
412 TTreeReader reader(&vertices);
413 std::vector<std::unique_ptr<TTreeReaderValue<std::vector<double>>>> vz;
414 for (unsigned imat = 0; imat < nMats; ++imat)
415 vz.emplace_back(new TTreeReaderValue<std::vector<double>>(
416 reader, "mudis_DISvz_" + MatTypeStr[imat]));
417 const Long64_t end = fnEvts >= 0 ? std::min(Long64_t(fstartEvt) + fnEvts,
418 ftree->GetEntries())
419 : ftree->GetEntries();
420 double minimum = std::numeric_limits<double>::infinity(),
421 msMinimum = minimum;
422 bool valid = true;
423 if (fstartEvt < end) {
424 reader.SetEntriesRange(fstartEvt, end);
425 Long64_t scanned = 0;
426 while (reader.Next()) {
427 ++scanned;
428 for (unsigned imat = 0; imat < nMats; ++imat) {
429 const auto* values = vz[imat]->Get();
430 if (!values || vz[imat]->GetSetupStatus() < 0 ||
431 vz[imat]->GetReadStatus() !=
432 TTreeReaderValue<std::vector<double>>::kReadSuccess) {
433 valid = false;
434 continue;
435 }
436 for (double z : *values) {
437 if (!std::isfinite(z)) {
438 valid = false;
439 continue;
440 }
441 minimum = std::min(minimum, z);
442 if (MatTypeStr[imat] == "MS") msMinimum = std::min(msMinimum, z);
443 }
444 }
445 }
446 valid = valid && scanned == end - fstartEvt;
447 }
448 if (valid && std::isfinite(minimum)) {
449 const double detectorMinimum =
450 fFilterOption == 2 ? fDetectorVolumeZ.first
451 : fFilterOption == 1
452 ? *std::min_element(fStationZ.begin(), fStationZ.end())
453 : fDetectorZ;
454 minimum = std::min(minimum, detectorMinimum);
455 LOG(info) << "Muon shield field bounds: input MS minimum z = "
456 << msMinimum << " cm; conservative scan minimum z = " << minimum
457 << " cm";
458 fPropagator->SetMuonShieldMinZ(minimum);
459 } else {
460 // No usable vertices or failed reads: retain the complete field map.
461 fPropagator->SetMuonShieldMinZ(-std::numeric_limits<double>::infinity());
462 if (!valid)
463 LOG(warn)
464 << "Vertex range scan failed; using full muon shield field bounds";
465 }
466 }
467
468 // Histograms belong to this file, even if ROOT's automatic attachment is
469 // disabled. Clear borrowed pointers on success and on exception alike.
470 const auto closeOutput = [this](TFile* file) {
471 delete file;
472 fouttree = nullptr;
473 for (auto& hist : hist_all) hist = {};
474 for (auto& hist : hist_filt) hist = {};
475 fFilterEfficiency = {};
476 };
477 std::unique_ptr<TFile, decltype(closeOutput)> outfile(
478 TFile::Open(output.c_str(), "CREATE"), closeOutput);
479 if (!outfile || outfile->IsZombie()) {
480 throw std::runtime_error("MuDISFilter: cannot create output file " +
481 output);
482 }
483 outfile->cd();
484
485 for (unsigned imat = 0; imat < nMats; ++imat) {
486 TDirectory* dir = outfile->mkdir(MatTypeStr[imat].Data());
487
488 hist_all[imat] = BookHistograms(dir, MatTypeStr[imat]);
489 hist_filt[imat] = BookHistograms(dir, MatTypeStr[imat], "filtered");
490 }
491 BookFilterEfficiencyHistograms(outfile->mkdir("filter_efficiency"));
492
493 outfile->cd();
494 fouttree = new TTree(
495 "MuonDIS", "Muon information, DIS products and soft interaction tracks");
496 foutEv.InitTree(fouttree);
497
498 Long64_t n = ftree->GetEntries();
499 LOG(info) << " * input tree with " << n << " entries" << std::endl;
500
501 ProcessEvents();
502
503 outfile->cd();
504 if (outfile->Write() <= 0 || outfile->TestBit(TFile::kWriteError))
505 throw std::runtime_error("MuDISFilter: failed to write output file");
506 outfile->Close();
507}

520double MuDISFilter::Charge(const DISparticle& particle) const {
521 const auto* pdg = fPDG->GetParticle(particle.pid);
522 // Nuclear PDG codes encode Z in digits 5--7 (10LZZZAAAI).
523 if (!pdg && std::abs(particle.pid) >= 1000000000)
524 return ((std::abs(particle.pid) / 10000) % 1000) *
525 (particle.pid > 0 ? 1. : -1.);
526 return pdg ? pdg->Charge() / 3. : 0.;
527}

530 switch (particle.pid) {
531 case -11:

534 return 2;
535 case -13:
536 return 3;

559 }

561}

563void MuDISFilter::ConfigureFilterGeometry(
564 unsigned option, const MagneticTrackPropagator& propagator) {
565 if (option == 1) {
566 const std::array<const char*, 5> names = {"Tr1", "Tr2", "Tr3", "Tr4",
567 "TimingDetector"};
568 std::array<double, 5> planes;
569 std::array<std::array<double, 4>, 5> bounds;
570 for (unsigned i = 0; i < names.size(); ++i) {
571 planes[i] = propagator.GetPlaneZ(names[i]);
572 bounds[i] = propagator.GetVolumeExitFaceXY(names[i]);
573 }
574 fStationZ = planes;
575 fStationXY = bounds;
576 } else if (option == 2) {
577 const double front = propagator.GetVolumeZRange("Tr1").first;
578 const double end = propagator.GetVolumeZRange("TimingDetector").second;
579 if (front >= end)
580 throw std::runtime_error(
581 "Timing detector end must be downstream of the Tr1 front");
582 fDetectorVolumeZ = {front, end};
583 }
584}

589 if (fPropagator) ConfigureFilterGeometry(option, *fPropagator);
590 fFilterOption = option;

636std::vector<MuDISFilter::FilterCandidate> MuDISFilter::DecayDaughters(
637 const std::vector<DISparticle>& daughters, const TVector3& vertex) const {
638 if (!fUsePythiaDecays) {
639 std::vector<FilterCandidate> candidates;
640 candidates.reserve(daughters.size());
641 for (const auto& particle : daughters)
642 candidates.push_back({particle, vertex});
643 return candidates;
644 }
645 if (!fPythiaDecayer || !std::isfinite(fTr1Z))
646 throw std::runtime_error("MuDISFilter: Pythia decays require Tr1 geometry");
647
648 std::vector<FilterCandidate> candidates;
649 const double tr1Zmm = 10. * fTr1Z;
650 const auto addParticle = [&candidates](const Pythia8::Particle& particle) {
651 DISparticle daughter;
652 daughter.pid = particle.id();
653 daughter.px = particle.px();
654 daughter.py = particle.py();
655 daughter.pz = particle.pz();
656 daughter.E = particle.e();
657 candidates.push_back(
658 {daughter, TVector3(particle.xProd() / 10., particle.yProd() / 10.,
659 particle.zProd() / 10.)});
660 };
661 for (const auto& daughter : daughters) {
662 fPythiaDecayer->event.reset();
663 const double momentum2 = daughter.px * daughter.px +
664 daughter.py * daughter.py +
665 daughter.pz * daughter.pz;
666 const double mass2 = daughter.E * daughter.E - momentum2;
667 if (mass2 < -1.e-8) {
668 candidates.push_back({daughter, vertex});
669 continue;
670 }
671 const int root = fPythiaDecayer->event.append(
672 daughter.pid, 1, 0, 0, daughter.px, daughter.py, daughter.pz,
673 daughter.E, std::sqrt(std::max(0., mass2)));
674 fPythiaDecayer->event[root].vProd(10. * vertex.X(), 10. * vertex.Y(),
675 10. * vertex.Z(), 0.);
676 // Manually appended particles have tau = 0. Pythia samples lifetimes
677 // for its own decay products, but the original particle needs one here.
678 fPythiaDecayer->event[root].tau(fPythiaDecayer->event[root].tau0() *
679 fPythiaDecayer->rndm.exp());
680
681 std::function<void(int)> decay = [&](int index) {
682 const auto& particle = fPythiaDecayer->event[index];
683 if ((tr1Zmm - particle.zProd()) * particle.pz() <= 0.) {
684 addParticle(particle);
685 return;
686 }
687 if (!fPythiaDecayer->moreDecays(index) ||
688 fPythiaDecayer->event[index].isFinal() ||
689 ((tr1Zmm - fPythiaDecayer->event[index].zDec()) *
690 fPythiaDecayer->event[index].pz() <=
691 0)) {
692 addParticle(fPythiaDecayer->event[index]);
693 return;
694 }
695 for (int child : fPythiaDecayer->event[index].daughterList())
696 decay(child);
697 };
698 decay(root);
699 }
700 return candidates;
701}

734bool MuDISFilter::HitsTrackingAndTD(const FilterCandidate& candidate,
735 double charge) const {
736 const auto& particle = candidate.particle;
737 const auto& vertex = candidate.vertex;
738 TVector3 position = vertex, momentum(particle.px, particle.py, particle.pz);
739 std::array<unsigned, 5> order = {0, 1, 2, 3, 4};
740 std::sort(order.begin(), order.end(),
741 [this, &particle](unsigned a, unsigned b) {
742 return particle.pz >= 0. ? fStationZ[a] < fStationZ[b]
743 : fStationZ[a] > fStationZ[b];
744 });
745 bool firstPair = false, secondPair = false, TD = false;
746 for (unsigned station : order) {
747 if (!IsForwardToPlane(candidate, fStationZ[station])) continue;
748 TVector3 hit, nextMomentum;
749 if (!fPropagator->Extrapolate(charge, position, momentum,
750 fStationZ[station], hit, nextMomentum))
751 return false;
752 position = hit;
753 momentum = nextMomentum;
754 if (!InsideXY(hit, fStationXY[station])) continue;
755 if (station < 2)
756 firstPair = true;
757 else if (station < 4)
758 secondPair = true;
759 else
760 TD = true;
761 if (firstPair && secondPair && TD) return true;
762 }
763 return false;
764}

776bool MuDISFilter::PassChargedCandidates(
777 const std::vector<FilterCandidate>& candidates, unsigned minimum) const {
778 if (minimum == 0) return true;
779 if (candidates.size() < minimum) return false;
780 // Both charged filters count eligible candidates before expensive transport.
781 unsigned remaining = 0, accepted = 0;
782 std::vector<double> charges;
783 charges.reserve(candidates.size());
784 for (const auto& candidate : candidates) {
785 const double charge = ChargedCandidateCharge(candidate);
786 charges.push_back(charge);
787 if (charge != 0.) ++remaining;
788 }
789 for (std::size_t i = 0; i < candidates.size(); ++i) {
790 if (accepted + remaining < minimum) return false;
791 if (charges[i] == 0.) continue;
792 --remaining;
793 if (fUseDetectorAcceptance) {
794 const bool hits =
795 fFilterOption == 1
796 ? HitsTrackingAndTD(candidates[i], charges[i])
797 : HitsDetectorPlane(candidates[i], charges[i], fDetectorZ,
798 {-200., 200., -300., 300.});
799 if (!hits) continue;
800 }
801 if (++accepted >= minimum) return true;
802 }
803 return false;

876void MuDISFilter::FillDIS(Histograms& h, const MuonDISInBranches& br, int idis,
877 const std::vector<DISparticle>& daughters,
878 const std::vector<FilterCandidate>& candidates,
879 Histograms* filtered) {
880 unsigned charged = 0;
881 double totalP = 0., chargedP = 0., muonP = 0.;
882 for (const auto& p : daughters) {
883 const int category = DaughterCategory(p);
884 for (auto* target : {&h, filtered}) {
885 if (!target) continue;
886 FillWithFlow(target->dis_pdg, p.pid);
887 FillWithFlow(target->dis_pdgGrouped, category);
888 }
889 const double momentum = std::sqrt(p.px * p.px + p.py * p.py + p.pz * p.pz);
890 totalP += momentum;
891 if (IsCharged(p)) {
892 ++charged;
893 chargedP += momentum;
894 }
895 if (std::abs(p.pid) == 13) muonP += momentum;
896 }
897 unsigned chargedCandidates = 0;
898 double neutralTDEnergy = 0.;
899 for (const auto& candidate : candidates) {
900 if (IsCharged(candidate.particle)) {
901 ++chargedCandidates;
902 } else if (!IsNeutrino(candidate.particle) &&
903 HitsTimingDetector(candidate)) {
904 neutralTDEnergy += candidate.particle.E;
905 }
906 }
907 for (auto* target : {&h, filtered}) {
908 if (!target) continue;
909 FillWithFlow(target->dis_vxz, br.DISvz->at(idis), br.DISvx->at(idis));
910 FillWithFlow(target->dis_vyz, br.DISvz->at(idis), br.DISvy->at(idis));
911 FillWithFlow(target->dis_vxy, br.DISvx->at(idis), br.DISvy->at(idis));
912 FillWithFlow(target->dis_vr,
913 std::hypot(br.DISvx->at(idis), br.DISvy->at(idis)));
914 FillWithFlow(target->dis_vz, br.DISvz->at(idis));
915 FillWithFlow(target->dis_n, daughters.size());
916 FillWithFlow(target->dis_nCharged, charged);
917 FillWithFlow(target->dis_nChargedCandidates, chargedCandidates);
918 FillWithFlow(target->dis_eNeutralTD, neutralTDEnergy);
919 if (totalP > 0.) {
920 FillWithFlow(target->dis_pChargedFrac, chargedP / totalP);
921 FillWithFlow(target->dis_pMuFrac, muonP / totalP);
922 }
923 }
