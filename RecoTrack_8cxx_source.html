14RecoTrack::RecoTrack(const TVector3& position, const TVector3& momentum,
15 Int_t charge, const std::array<Double_t, 21>& covariance,
16 Double_t chi2, Double_t nDoF, UInt_t nMeasurements,
17 std::vector<Double_t> x_fits, std::vector<Double_t> y_fits,
18 std::vector<Double_t> z_fits,
19 std::vector<Double_t> residuals,
20 std::vector<Double_t> pulls)
21 : TObject(),
22 m_x(position.X()),
23 m_y(position.Y()),
24 m_z(position.Z()),
25 m_px(momentum.X()),
26 m_py(momentum.Y()),
27 m_pz(momentum.Z()),
28 m_charge(charge),
29 m_chi2(chi2),
30 m_nDoF(nDoF),
31 m_nMeasurements(nMeasurements),
32 m_residuals(std::move(residuals)),
33 m_x_fits(std::move(x_fits)),
34 m_y_fits(std::move(y_fits)),
35 m_z_fits(std::move(z_fits)),
36 m_pulls(std::move(pulls)),
37 m_cov_elements(covariance.begin(), covariance.end()) {
38 // Diagonal of the upper triangle of a 6x6 matrix, row-major
39 auto sigma = [&covariance](int index) {
40 return std::sqrt(std::max(0.0, covariance[index]));
41 };
42 m_err_x = sigma(0);
43 m_err_y = sigma(6);
44 m_err_z = sigma(11);
45 m_err_px = sigma(15);
46 m_err_py = sigma(18);
47 m_err_pz = sigma(20);
48}

50Bool_t RecoTrack::ExtrapolateToZ(Double_t z, TVector3& position) const {
51 if (std::abs(m_pz) < 1e-10) {
52 return false;
53 }
54 const Double_t lambda = (z - m_z) / m_pz;
55 position.SetXYZ(m_x + lambda * m_px, m_y + lambda * m_py, z);
56 return true;
57}

62void setExtrapolation(const TVector3& position, const TVector3& momentum,
63 const std::vector<Double_t>& covariance,
64 Double_t pathLength, Bool_t& valid,
65 std::array<Double_t*, 7> values,
66 std::vector<Double_t>& cov) {
67 valid = true;
68 const std::array<Double_t, 7> input{position.X(), position.Y(), position.Z(),
69 momentum.X(), momentum.Y(), momentum.Z(),
70 pathLength};
71 for (std::size_t i = 0; i < values.size(); ++i) {
72 *values[i] = input[i];
73 }
74 constexpr std::size_t kCovSize = 21;
75 if (covariance.size() == kCovSize) {
76 cov = covariance;
77 } else {
78 cov.clear(); // no covariance: empty, as when the setter is never called
79 }
80}
