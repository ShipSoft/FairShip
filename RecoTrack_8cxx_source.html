13RecoTrack::RecoTrack(const TVector3& position, const TVector3& momentum,
14 Int_t charge, const std::array<Double_t, 21>& covariance,
15 Double_t chi2, Double_t nDoF, UInt_t nMeasurements,
16 std::vector<Double_t> residuals,
17 std::vector<Double_t> pulls)
18 : TObject(),
19 m_x(position.X()),
20 m_y(position.Y()),
21 m_z(position.Z()),
22 m_px(momentum.X()),
23 m_py(momentum.Y()),
24 m_pz(momentum.Z()),
25 m_charge(charge),
26 m_chi2(chi2),
27 m_nDoF(nDoF),
28 m_nMeasurements(nMeasurements),
29 m_residuals(std::move(residuals)),
30 m_pulls(std::move(pulls)),
31 m_cov_elements(covariance.begin(), covariance.end()) {
32 // Diagonal of the upper triangle of a 6x6 matrix, row-major
33 auto sigma = [&covariance](int index) {
34 return std::sqrt(std::max(0.0, covariance[index]));
35 };
36 m_err_x = sigma(0);
37 m_err_y = sigma(6);
38 m_err_z = sigma(11);
39 m_err_px = sigma(15);
40 m_err_py = sigma(18);
41 m_err_pz = sigma(20);
42}

44Bool_t RecoTrack::ExtrapolateToZ(Double_t z, TVector3& position) const {
45 if (std::abs(m_pz) < 1e-10) {
46 return false;
47 }
48 const Double_t lambda = (z - m_z) / m_pz;
49 position.SetXYZ(m_x + lambda * m_px, m_y + lambda * m_py, z);
50 return true;
51}
