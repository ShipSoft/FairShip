30// The gas volume, which defines the detector ID, is placed with the straw
31// copy number plus 1000 (see strawtubes::ConstructGeometry).
32constexpr int kGasCopyOffset = 1000;

35std::shared_ptr<const Acts::HomogeneousSurfaceMaterial> strawMaterial() {
36 const auto gas = Acts::Material::fromMassDensity(
37 30000.0 * u::mm, 50000.0 * u::mm, 30.0, 15.0, 0.0018 * u::g / u::cm3);
38 const auto wall = Acts::Material::fromMassDensity(
39 300.0 * u::mm, 400.0 * u::mm, 27.0, 13.0, 1.4 * u::g / u::cm3);
40 const Acts::MaterialSlab gasSlab(gas, 10.0 * u::mm);
41 const Acts::MaterialSlab wallSlab(wall, 0.036 * u::mm);
42 auto slab = Acts::MaterialSlab::combineLayers(wallSlab, gasSlab);
43 slab = Acts::MaterialSlab::combineLayers(slab, wallSlab);
44 return std::make_shared<Acts::HomogeneousSurfaceMaterial>(slab);
45}

47Acts::Transform3 toTransform(const TGeoHMatrix& matrix) {
48 const Double_t* r = matrix.GetRotationMatrix();
49 const Double_t* t = matrix.GetTranslation();
50 Acts::RotationMatrix3 rotation;
51 rotation << r[0], r[1], r[2], r[3], r[4], r[5], r[6], r[7], r[8];
52 Acts::Transform3 transform = Acts::Transform3::Identity();
53 transform.linear() = rotation;
54 transform.translation() = Acts::Vector3(t[0], t[1], t[2]);
55 return transform;
56}

59Acts::GeometryIdentifier geometryId(int detectorId) {
60 const int station = detectorId / 1000000;
61 const int view = detectorId / 100000 % 10;
62 const int layer = detectorId / 10000 % 10;
63 const int straw = detectorId % 1000;
64 return Acts::GeometryIdentifier()
65 .withVolume(1)
66 .withLayer(8 * station + 2 * view + layer)
67 .withSensitive(straw);
68}

70struct Builder {
71 std::unordered_map<int, std::shared_ptr<Acts::Surface>>& surfaces;
72 std::unordered_map<int, StrawWire>& wires;
73 std::shared_ptr<const Acts::HomogeneousSurfaceMaterial> material;
74
75 void visit(const TGeoVolume& volume, const TGeoHMatrix& parent) {
76 for (int i = 0; i < volume.GetNdaughters(); ++i) {
77 const TGeoNode* node = volume.GetNode(i);
78 TGeoHMatrix global = parent;
79 global.Multiply(node->GetMatrix());
80 const TGeoVolume* daughter = node->GetVolume();
81 if (std::string_view(daughter->GetName()) == "straw") {
82 addStraw(*node, global);
83 } else {
84 visit(*daughter, global);
85 }
86 }
87 }
88
89 void addStraw(const TGeoNode& node, const TGeoHMatrix& global) {
90 const auto* tube =
91 dynamic_cast<const TGeoTube*>(node.GetVolume()->GetShape());
92 if (tube == nullptr) {
93 throw std::runtime_error("ActsStrawGeometry: straw is not a TGeoTube");
94 }
95 const int detectorId = node.GetNumber() + kGasCopyOffset;
96 auto bounds = std::make_shared<const Acts::LineBounds>(
97 tube->GetRmax() * ActsFrame::kShipLength,
98 tube->GetDZ() * ActsFrame::kShipLength);
99 auto surface = Acts::Surface::makeShared<Acts::StrawSurface>(
100 ActsFrame::transformToActs(toTransform(global)), bounds);
101 surface->assignSurfaceMaterial(material);
102 surface->assignGeometryId(geometryId(detectorId));
103 surfaces.emplace(detectorId, std::move(surface));
104
105 const Double_t* t = global.GetTranslation();
106 const Double_t* r = global.GetRotationMatrix();
107 // The tube axis is the local z axis, i.e. the third column
108 wires.emplace(
109 detectorId,
110 StrawWire{detectorId, detectorId / 1000000, detectorId / 100000 % 10,
111 detectorId / 10000 % 10, t[0], t[1], t[2], r[2], r[5], r[8],
112 tube->GetRmin(), tube->GetDZ()});
113 }
114};

118ActsStrawGeometry::ActsStrawGeometry(const TGeoManager& geometry) {
119 const TGeoVolume* top = geometry.GetTopVolume();
120 if (top == nullptr) {
121 throw std::invalid_argument("ActsStrawGeometry: no geometry loaded");
122 }
123 Builder builder{m_surfaces, m_wires, strawMaterial()};
124 builder.visit(*top, TGeoHMatrix());
125 if (m_surfaces.empty()) {
126 throw std::runtime_error("ActsStrawGeometry: no straws in the geometry");
127 }
128}
