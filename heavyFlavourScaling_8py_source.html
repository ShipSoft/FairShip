4"""Target-composition scaling of charm/beauty over min-bias cross-section ratios.
5
6Shared by ``macro/makeDecay.py`` and ``macro/run_fixedTarget.py`` so the physics
7constants and the ``chicc``/``chibb`` derivation live in a single place.
8"""
9
10from collections import namedtuple
11
12# Reference values for Molybdenum (historical defaults). Cross-section ratios
13# scale heavy-flavour ~ A and mbias ~ A^(0.71).
14heavyflavour_Ascale = 1
15mbias_Ascale = 0.71
16
17A_REF = 98.0
18CHICC_REF = 1.7e-3 # prob to produce primary ccbar pair/pot on Mo
19CHIBB_REF = 1.6e-7 # prob to produce primary bbbar pair/pot on Mo
20TARGET_A = {"W": 184.0, "Mo": 98.0}
21
22CrossSections = namedtuple("CrossSections", ["chicc", "chibb", "A", "scale"])
23
24
25def derive_cross_sections(target_composition=None, A=None, chicc=None, chibb=None):
26 """Return default-scaled chicc/chibb, honouring explicit overrides.
27
28 ``A`` overrides the ``target_composition`` preset; ``chicc``/``chibb``
29 override the target-derived value when not ``None``. Supply either
30 ``target_composition`` (a known preset) or an explicit ``A``.
31 """
32 if A is None:
33 if target_composition is None:
34 raise ValueError(f"Provide either target_composition (one of {sorted(TARGET_A)}) or an explicit A.")
35 A = TARGET_A[target_composition]
36 if A <= 0:
37 raise ValueError(f"Invalid target mass number A={A}. Must be > 0.")
38 scale = (A / A_REF) ** (heavyflavour_Ascale - mbias_Ascale)
39 chicc = chicc if chicc is not None else CHICC_REF * scale
40 chibb = chibb if chibb is not None else CHIBB_REF * scale
41 return CrossSections(chicc, chibb, A, scale)
42
43
44def check_run_type_override(is_beauty, chicc, chibb):
45 """Reject a cross-section override that does not match the run type.
46
47 A run is either charm or beauty, so at most one of ``chicc``/``chibb`` is
48 meaningful. Passing the wrong one (or both) is a mistake and raises
49 ``ValueError`` with a clear message rather than being silently ignored.
50 """
51 if chicc is not None and chibb is not None:
52 raise ValueError("Set only one of --chicc / --chibb; a run is either charm or beauty.")
53 if is_beauty and chicc is not None:
54 raise ValueError("This is a beauty run: pass --chibb, not --chicc.")
55 if not is_beauty and chibb is not None:
56 raise ValueError("This is a charm run: pass --chicc, not --chibb.")

59def format_summary(cs, target_composition=None):
60 """One human-readable summary of the derived cross-section ratios."""
61 label = target_composition or "custom (A given)"

65 f"Derived cross-section ratios: chicc={cs.chicc:.3e}, chibb={cs.chibb:.3e}"
