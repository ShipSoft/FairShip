11def correlated_sample(seed, n_events, p_active=0.3, n_clones=20):
12 """Events with n_clones branches of weight 1/n_clones around a common x."""
13 rng = np.random.default_rng(seed)
14 events: list[tuple[np.ndarray, np.ndarray] | None] = []
15 for _ in range(n_events):
16 if rng.random() < p_active:
17 x0 = rng.uniform(0.0, 10.0)
18 events.append((x0 + rng.normal(0.0, 0.3, n_clones), np.full(n_clones, 1.0 / n_clones)))
19 else:
20 events.append(None)
21 return events

24def fill(h, events, run=1):
25 for i, ev in enumerate(events):
26 if ev is None:
27 h.add_empty_events(1)
28 else:
29 h.fill_event(*ev, key=make_event_key(run, i))

32def fill_cpp(h, events, run=1):
33 for i, ev in enumerate(events):
34 if ev is None:
35 h.AddEmptyEvents(1)
36 continue
37 for x, w in zip(*ev):
38 h.Fill(x, w)
39 h.EndEvent(make_event_key(run, i))

42def test_matches_per_event_formula():
43 events = correlated_sample(1, 3000)
44 h = EventHistogram("x", EDGES)
45 fill(h, events)
46 t = np.zeros((len(events), len(EDGES) - 1))
47 for i, ev in enumerate(events):
48 if ev is not None:
49 t[i] = np.histogram(ev[0], EDGES, weights=ev[1])[0]
50 n = len(events)
51 assert h.n_events == n
52 assert h.values() == pytest.approx(t.sum(0) / n)
53 assert h.errors() == pytest.approx(t.std(0, ddof=1) / np.sqrt(n))

63def test_merge_equals_single_accumulator():
64 a, b = correlated_sample(3, 1000), correlated_sample(4, 1500)
65 ha = EventHistogram("x", EDGES, covariance=True, n_replicas=50)
66 hb = EventHistogram("x", EDGES, covariance=True, n_replicas=50)
67 hab = EventHistogram("x", EDGES, covariance=True, n_replicas=50)
68 fill(ha, a, run=1)
69 fill(hb, b, run=2)
70 fill(hab, a, run=1)
71 fill(hab, b, run=2)
72 ha += hb

157def test_root_round_trip_and_cpp_agree(root_with_shipdata, tmp_path):
158 ROOT = root_with_shipdata
159 events = correlated_sample(7, 2000)
160 h = EventHistogram("x", EDGES, covariance=True, n_replicas=20)
161 fill(h, events)
162 cpp = ROOT.EventHistogram("x", "", len(EDGES) - 1, 0.0, 10.0, True, 20)
163 fill_cpp(cpp, events)
164 result = cpp.MakeResult()
165 assert [result.GetBinContent(b) for b in range(1, 11)] == pytest.approx(h.values())
166 assert [result.GetBinError(b) for b in range(1, 11)] == pytest.approx(h.errors())
167 assert list(cpp.GetBootstrapErrors()) == pytest.approx(h.bootstrap_errors())
168 for key in (0, 1, make_event_key(7, 123456)):
169 assert [cpp.ReplicaWeight(key, r) for r in range(30)] == list(replica_weights(key, 30))
170
171 # Python writes, C++ reads; C++ writes, Python reads.
172 path = str(tmp_path / "eh.root")
173 f = ROOT.TFile(path, "RECREATE")
174 h.to_root(f)
175 f.Close()
176 f = ROOT.TFile(path)
177 back = ROOT.EventHistogram.Read(f, "x")
178 assert back.GetNEvents() == h.n_events
179 res = back.MakeResult()
180 assert [res.GetBinError(b) for b in range(1, 11)] == pytest.approx(h.errors())
181 f.Close()
182 path2 = str(tmp_path / "eh_cpp.root")
183 f = ROOT.TFile(path2, "RECREATE")
184 cpp.Write(f)
185 f.Close()
186 f = ROOT.TFile(path2)
187 again = EventHistogram.from_root(f, "x")
188 assert again.errors() == pytest.approx(h.errors())
189 assert again.covariance() == pytest.approx(h.covariance())
190 f.Close()
191
192 # Both refuse an incompatible merge and an open event.
193 other = ROOT.EventHistogram("x", "", len(EDGES) - 1, 0.0, 10.0, False, 20)
194 with pytest.raises(Exception, match="covariance"):
195 cpp.Add(other)
196 other.Fill(1.0)
197 with pytest.raises(Exception, match="open event"):
198 other.MakeResult()
