50class FakeTracker:
51 """Records what runTracking hands to SHiP::ActsStrawTracker.
52
53 FitCandidate drops the last measurement of each candidate, as the real
54 tracker drops measurements that do not fit the track.
55 """
56
57 def __init__(self, found=(), charges=(), unreachable_z=()):
58 self.measurements = []
59 self.candidates = []
60 self.tracks = []
61 self.vertices = []
62 self._found = found
63 self._charges = charges
64 self._unreachable_z = unreachable_z
65 self.extrapolations = []

67 def ClearEvent(self):
68 self.measurements.clear()

71 def AddMeasurement(self, detector_id, drift, resolution):
72 if detector_id == UNKNOWN_STRAW:
73 return -1
74 self.measurements.append((detector_id, drift, resolution))
75 return len(self.measurements) - 1

77 def FindTracks(self):
78 self.tracks = [list(t) for t in self._found]
79 return len(self.tracks)

81 def FitCandidate(self, measurements):
82 self.candidates.append(list(measurements))
83 self.tracks.append(list(measurements)[:-1])
84 return len(self.tracks) - 1

98 def ExtrapolateToZ(self, index, z, from_last_hit, position, momentum, covariance, path_length):
99 self.extrapolations.append((index, z, from_last_hit))
100 if z in self._unreachable_z:
101 return False
102 position.SetXYZ(0.0, 0.0, z)
103 path_length.value = z / 10.0
104 return True

111def test_candidate_hits_are_ordered_in_z_with_one_per_layer():
112 # Hit 0 is an unknown straw, so measurement indices are hit indices - 1.
113 # The last hit shares a straw layer with hit 2 but sits at smaller z, so
114 # it is kept and hit 2 is not.
115 hits = [make_hit(UNKNOWN_STRAW, 1, 0, 0, z=0.0)]
116 hits += [make_hit(1000 + i, 1, i // 2, i % 2, z=10.0 * i) for i in range(MIN_HITS_PER_TRACK + 1)]
117 hits.append(make_hit(9999, 1, 0, 1, z=5.0))
118 tracker = FakeTracker()
119 tracks, vertices, hit_indices = runTracking(
120 tracker, [{"indices": list(reversed(range(len(hits))))}], hits, 0.012, fit_vertex=False
121 )
122
123 assert len(tracks) == 1
124 assert vertices == []
125 used = [m + 1 for m in tracker.candidates[0]]
126 assert used == sorted(used, key=lambda i: hits[i].z)
127 assert 0 not in used
128 assert len({hits[i].layer_key for i in used}) == len(used)
129 assert len(hits) - 1 in used and 2 not in used
130 # The hit indices of the track are those the tracker kept
131 assert hit_indices[0] == used[:-1]

165def test_extrapolations_and_track_length_are_stored():
166 # The UBT is reached backwards from the first hit, the timing detector
167 # forwards from the last; a plane the track cannot reach is not stored
168 hits = [make_hit(1000 + i, 1, 0, 0, z=float(i)) for i in range(2)]
169 tracker = FakeTracker(found=[[0], [1]], unreachable_z=(9000.0,))
170 tracks, _, _ = runTracking(tracker, None, hits, 0.012, fit_vertex=False, ubt_z=3000.0, td_z=9000.0)
171 assert tracker.extrapolations == [(0, 3000.0, False), (0, 9000.0, True), (1, 3000.0, False), (1, 9000.0, True)]
172 assert [t.track_length for t in tracks] == [100.0, 101.0]
173 assert all(t.ubt == (3000.0, 300.0) for t in tracks)
174 assert not any(hasattr(t, "td") for t in tracks)
