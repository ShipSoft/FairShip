13def test_hits_follow_measurements_grouped_by_straw():
14 # Straws A, B, C; A has two hits. The measurements are grouped by straw,
15 # in a different order than the hits, and keep hit order within a straw.
16 hit_gids = ["A", "B", "A", "C"]
17 measurement_gids = ["C", "A", "A", "B"]
18 assert match_hits_to_measurements(hit_gids, measurement_gids) == {0: 1, 2: 2, 1: 3, 3: 0}
