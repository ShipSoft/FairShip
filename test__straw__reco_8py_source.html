41class FakeTracker:
42 """Records what runTracking hands to SHiP::ActsStrawTracker.
43
44 FitCandidate drops the last measurement of each candidate, as the real
45 tracker drops measurements that do not fit the track.
46 """
47
48 def __init__(self, found=(), charges=()):
49 self.measurements = []
50 self.candidates = []
51 self.tracks = []
52 self.vertices = []
53 self._found = found
54 self._charges = charges
55
56 def ClearEvent(self):
57 self.measurements.clear()
58 self.tracks.clear()
59
60 def AddMeasurement(self, detector_id, drift, resolution):
61 if detector_id == UNKNOWN_STRAW:
62 return -1
63 self.measurements.append((detector_id, drift, resolution))
64 return len(self.measurements) - 1
65
66 def FindTracks(self):
67 self.tracks = [list(t) for t in self._found]
68 return len(self.tracks)
69
70 def FitCandidate(self, measurements):
71 self.candidates.append(list(measurements))
72 self.tracks.append(list(measurements)[:-1])
73 return len(self.tracks) - 1
74
75 def GetNTracks(self):
76 return len(self.tracks)
77
78 def GetTrack(self, index):
79 return FakeTrack(self._charges[index] if self._charges else 1)
80
81 def GetTrackMeasurements(self, index):
82 return self.tracks[index]
83
84 def FitVertex(self, tracks, vertex):
85 self.vertices.append(list(tracks))
86 return True

89def test_candidate_hits_are_ordered_in_z_with_one_per_layer():
90 # Hit 0 is an unknown straw, so measurement indices are hit indices - 1.
91 # The last hit shares a straw layer with hit 2 but sits at smaller z, so
92 # it is kept and hit 2 is not.
93 hits = [make_hit(UNKNOWN_STRAW, 1, 0, 0, z=0.0)]
94 hits += [make_hit(1000 + i, 1, i // 2, i % 2, z=10.0 * i) for i in range(MIN_HITS_PER_TRACK + 1)]
95 hits.append(make_hit(9999, 1, 0, 1, z=5.0))
96 tracker = FakeTracker()
97 tracks, vertices, hit_indices = runTracking(
98 tracker, [{"indices": list(reversed(range(len(hits))))}], hits, 0.012, fit_vertex=False
99 )
100
101 assert len(tracks) == 1
102 assert vertices == []
103 used = [m + 1 for m in tracker.candidates[0]]
104 assert used == sorted(used, key=lambda i: hits[i].z)
105 assert 0 not in used
106 assert len({hits[i].layer_key for i in used}) == len(used)
107 assert len(hits) - 1 in used and 2 not in used
108 # The hit indices of the track are those the tracker kept
109 assert hit_indices[0] == used[:-1]
