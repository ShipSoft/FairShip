78def test_measurements_follow_hits_in_z_with_one_per_layer():
79 # Hit 0 is an unknown straw, so measurement indices are hit indices - 1.
80 # The last hit shares a straw layer with hit 2 but sits at smaller z, so
81 # it is kept and hit 2 is not.
82 hits = [make_hit(UNKNOWN_STRAW, 1, 0, 0, z=0.0)]
83 hits += [make_hit(1000 + i, 1, i // 2, i % 2, z=10.0 * i) for i in range(MIN_HITS_PER_TRACK + 1)]
84 hits.append(make_hit(9999, 1, 0, 1, z=5.0))
85 tracker = FakeTracker()
86 tracks, vertices, hit_indices = runTracking(
87 tracker, [candidate(list(reversed(range(len(hits)))))], hits, 0.012, fit_vertex=False
88 )
89
90 assert len(tracks) == 1
91 assert vertices == []
92 used = hit_indices[0]
93 assert used == sorted(used, key=lambda i: hits[i].z)
94 assert 0 not in used
95 assert len({hits[i].layer_key for i in used}) == len(used)
96 assert len(hits) - 1 in used and 2 not in used
97 # The fitter sees the measurements of exactly those hits
98 assert tracker.fits[0] == [i - 1 for i in used]
99 assert [tracker.measurements[m][0] for m in tracker.fits[0]] == [hits[i].detector_id for i in used]
