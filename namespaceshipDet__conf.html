327def _axis_cells(low: float, high: float, size: float) -> list[tuple[float, float]]:
328 """(centre, size) of the cells of ``size`` centred on the beam axis, cut at [low, high]."""
329 first = math.floor((low + size / 2.0) / size + 1.0e-9)

287 "y": (ubt_geo.BoxCenterY - ubt_geo.BoxY / 2.0, ubt_geo.BoxCenterY + ubt_geo.BoxY / 2.0),

293 f"The UBT map bounds in {axis}, {bounds[axis]} cm, differ from the UBT box of the geometry"

295 )

297 # The regions must tile the bounds: each inside them, no two overlapping,
298 # and together as large as them.
299 rectangles = []

301 half_x, half_y = region["size_x"] * u.cm / 2.0, region["size_y"] * u.cm / 2.0
302 x, y = region["x"] * u.cm, region["y"] * u.cm
303 rectangle = (x - half_x, x + half_x, y - half_y, y + half_y)
304 if (

306 or rectangle[1] > bounds["x"][1] + tolerance

350def generate_upstream_tagger_map(rule, ubt_geo) -> dict[str, Any]:
351 """Generate the UBT regions on the UBT box of the geometry configuration.
352
353 See the note in geometry/UpstreamTagger_config.yaml for the rule. The map
354 has the same form as the one stored with the geometry: kind, units, the
355 bounds of the box and the regions, all in cm.
356 """
357 if rule["kind"] != "simulation" or rule["units"] != "cm":
358 raise ValueError("The UBT map must be a simulation map expressed in cm")
359 bounds = {
360 "x": [(ubt_geo.BoxCenterX - ubt_geo.BoxX / 2.0) / u.cm, (ubt_geo.BoxCenterX + ubt_geo.BoxX / 2.0) / u.cm],
361 "y": [(ubt_geo.BoxCenterY - ubt_geo.BoxY / 2.0) / u.cm, (ubt_geo.BoxCenterY + ubt_geo.BoxY / 2.0) / u.cm],
362 }
363 size = rule["region_size"]
364 areas = _small_tile_areas(rule)
365 tolerance = 1.0e-9 # cm
366 regions = []
367 for y, size_y in _axis_cells(bounds["y"][0], bounds["y"][1], size):
368 for x, size_x in _axis_cells(bounds["x"][0], bounds["x"][1], size):
369 # A region lying inside a small-tile area has small tiles; an area
370 # covering only part of a region would cut it in two.
371 low_x, high_x, low_y, high_y = x - size_x / 2.0, x + size_x / 2.0, y - size_y / 2.0, y + size_y / 2.0
372 in_area = False
373 for area in areas:
374 overlap_x = min(high_x, area[1]) - max(low_x, area[0])
375 overlap_y = min(high_y, area[3]) - max(low_y, area[2])
376 if overlap_x <= tolerance or overlap_y <= tolerance:
377 continue
378 if overlap_x < size_x - tolerance or overlap_y < size_y - tolerance:
379 raise ValueError(
380 f"The UBT small-tile area x {list(area[:2])}, y {list(area[2:])} cm covers only part of the"
381 f" region at ({x}, {y}) cm: its edges must lie on region edges"
382 )
383 in_area = True
384 tile = rule["small_tile_size"] if in_area else rule["large_tile_size"]
385 for centre, extent, axis in ((x, size_x, "x"), (y, size_y, "y")):
386 for edge in (centre - extent / 2.0, centre + extent / 2.0):
387 tiles = (edge - rule["tile_edge_offset"]) / tile
388 if abs(tiles - round(tiles)) > 1.0e-6:
389 raise ValueError(
390 f"The UBT region at ({x}, {y}) cm has an edge at {axis} = {edge} cm, off the grid of its"
391 f" {tile} cm tiles: the UBT box, {bounds} cm, must end on a tile edge"
392 )
393 regions.append(
394 {
395 "id": len(regions),
396 "x": x,
397 "y": y,
398 "size_x": size_x,
399 "size_y": size_y,
400 "constituent_tile_size": tile,
401 }
402 )
403 return {"kind": rule["kind"], "units": rule["units"], "bounds": bounds, "regions": regions}
