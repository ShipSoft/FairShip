21def test_w_preset_scales_up():
22 cs = derive_cross_sections("W")
23 assert cs.A == 184.0
24 assert cs.scale == pytest.approx((184.0 / 98.0) ** 0.29)
25 assert cs.chicc == pytest.approx(CHICC_REF * cs.scale)
