140// Bind only on the current file's tree, never on TChain's persistent address
141// list. This checks both presence and type when a chain changes files.
142template <class T>
143bool BindInputBranch(TTree* tree, const TString& name, T* address,
144 bool required = true) {
145 auto* branch = tree->GetBranch(name);
146 if (!branch) {
147 if (required) LOG(error) << "Missing required input branch " << name;
148 return !required;
149 }
150 if constexpr (std::is_arithmetic_v<T>) {
151 // Reject object branches before ROOT attempts to bind a scalar address.
152 if (branch->GetClassName()[0] != '\0') {
153 LOG(error) << "Expected scalar input branch " << name;
154 return false;
155 }
156 }
157 branch->SetAutoDelete(false);
158 return tree->SetBranchAddress(name, address) >= 0;
159}

161struct CBMSimBranches {
162 std::vector<ShipMCTrack>* MCTrack = nullptr;
163 std::vector<vetoPoint>* sbtPt = nullptr;
164 std::vector<UpstreamTaggerPoint>* ubtPt = nullptr;
165 std::vector<strawtubesPoint>* sstPt = nullptr;
166 std::vector<TimeDetPoint>* tdPt = nullptr;
167
168 bool Setup(TChain* tree) {
169 fTreeNumber = -1;
170 Clear();
171 return tree->GetEntries() == 0 || PrepareEntry(tree, 0);
172 }
173
174 bool PrepareEntry(TTree* tree, Long64_t entry) {
175 Clear();
176 if (tree->LoadTree(entry) < 0) return false;
177 if (tree->GetTreeNumber() != fTreeNumber) {
178 fTreeNumber = tree->GetTreeNumber();
179 auto* current = tree->GetTree();
180 fValid = BindInputBranch(current, "MCTrack", &MCTrack);
181 fValid &= BindInputBranch(current, "TimeDetPoint", &tdPt);
182 fValid &= BindInputBranch(current, "vetoPoint", &sbtPt);
183 fValid &= BindInputBranch(current, "UpstreamTaggerPoint", &ubtPt);
184 fValid &= BindInputBranch(current, "strawtubesPoint", &sstPt);
185 }
186 return fValid;
187 }
188
189 private:
190 void Clear() {
191 fTracks.clear();
192 fSBT.clear();
193 fUBT.clear();
194 fSST.clear();
195 fTD.clear();
196 MCTrack = &fTracks;
197 sbtPt = &fSBT;
198 ubtPt = &fUBT;
199 sstPt = &fSST;
200 tdPt = &fTD;
201 }
202 std::vector<ShipMCTrack> fTracks;
203 std::vector<vetoPoint> fSBT;
204 std::vector<UpstreamTaggerPoint> fUBT;
205 std::vector<strawtubesPoint> fSST;
206 std::vector<TimeDetPoint> fTD;
207 int fTreeNumber = -1;
208 bool fValid = false;

211// For reading back the DIS tree
212struct MuonDISInBranches {
213 int nDISevts = 0; // per input muon per volume
214 int nDISevtsGenerated = -1;
215 double pPythia = -1.;
216 double wDIS = 1; // per input muon per volume
217 std::vector<double>* DISxsec = nullptr; // per input muon per DIS
218 std::vector<bool>* DIStarget =
219 nullptr; // per input muon per DIS, p=true, n=false
220 std::vector<double>* DISvx = nullptr;
221 std::vector<double>* DISvy = nullptr;
222 std::vector<double>* DISvz = nullptr;
223 std::vector<double>* DISvt = nullptr;
224 std::vector<int>* nDISdau = nullptr; // per DIS event muon
225 std::vector<DISparticle>* DISparticles = nullptr; // all DIS events together.
226
227 bool IsValid() const {
228 if (nDISevts < 0 || !DISxsec || !DIStarget || !DISvx || !DISvy || !DISvz ||
229 !DISvt || !nDISdau || !DISparticles)
230 return false;
231 const auto size = static_cast<std::size_t>(nDISevts);
232 if (DISxsec->size() != size || DIStarget->size() != size ||
233 DISvx->size() != size || DISvy->size() != size ||
234 DISvz->size() != size || DISvt->size() != size ||
235 nDISdau->size() != size)
236 return false;
237 std::size_t total = 0;
238 for (int count : *nDISdau) {
239 if (count < 0 ||
240 static_cast<std::size_t>(count) > DISparticles->size() - total)
241 return false;
242 total += count;
243 }
244 return total == DISparticles->size();
245 }
246
247 void Clear() {
248 nDISevts = 0;
249 nDISevtsGenerated = -1;
250 pPythia = -1.;
251 wDIS = 1.;
252 fStorage.initEvent(0);
253 DISxsec = &fStorage.DISxsec;
254 DIStarget = &fStorage.DIStarget;
255 DISvx = &fStorage.DISvx;
256 DISvy = &fStorage.DISvy;
257 DISvz = &fStorage.DISvz;
258 DISvt = &fStorage.DISvt;
259 nDISdau = &fStorage.nDISdau;
260 DISparticles = &fStorage.DISparticles;
261 }
262
263 bool SetupTree(TTree* tree, const TString& label) {
264 Clear();
265 bool ok = BindInputBranch(tree, "muon_nDISevt_" + label, &nDISevts);
266 ok &= BindInputBranch(tree, "muon_nDISGenerated_" + label,
267 &nDISevtsGenerated);
268 ok &= BindInputBranch(tree, "muon_pPythia_" + label, &pPythia);
269 ok &= BindInputBranch(tree, "muon_wDIS_" + label, &wDIS);
270 ok &= BindInputBranch(tree, "mudis_DISxsec_" + label, &DISxsec);
271 ok &= BindInputBranch(tree, "mudis_DIStarget_" + label, &DIStarget);
272 ok &= BindInputBranch(tree, "mudis_DISvx_" + label, &DISvx);
273 ok &= BindInputBranch(tree, "mudis_DISvy_" + label, &DISvy);
274 ok &= BindInputBranch(tree, "mudis_DISvz_" + label, &DISvz);
275 ok &= BindInputBranch(tree, "mudis_DISvt_" + label, &DISvt);
276 ok &= BindInputBranch(tree, "mudis_nDISdaughters_" + label, &nDISdau);
277 ok &= BindInputBranch(tree, "mudis_DISproducts_" + label, &DISparticles);
278 return ok;
279 }
