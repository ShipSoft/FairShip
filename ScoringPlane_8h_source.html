66 Bool_t ProcessHits(FairVolume* /*vol*/ = nullptr) override {
67 // Score each track once, on its first step in the plane.
68 if (!gMC->IsTrackEntering()) {
69 return kFALSE;
70 }
71 Int_t copyNo = 0;
72 gMC->CurrentVolID(copyNo);
73 TParticle* p = gMC->GetStack()->GetCurrentTrack();
74 const Int_t pdgCode = p->GetPdgCode();
75 const std::vector<Int_t>& accepted = fPlanes[copyNo - 1].absPdgCodes;
76 if (!accepted.empty() && std::find(accepted.begin(), accepted.end(),
77 std::abs(pdgCode)) == accepted.end()) {
78 return kFALSE;
79 }
80
81 this->fEventID = gMC->CurrentEvent();
82 this->fTrackID = gMC->GetStack()->GetCurrentTrackNumber();
83 this->fVolumeID = copyNo;
84 this->fTime = gMC->TrackTime() * 1.0e09; // ns
85 this->fLength = gMC->TrackLength();
86 gMC->TrackPosition(this->fPos);
87 gMC->TrackMomentum(this->fMom);
88 this->AddHit(this->fEventID, this->fTrackID, this->fVolumeID,
89 TVector3(this->fPos.X(), this->fPos.Y(), this->fPos.Z()),
90 TVector3(this->fMom.Px(), this->fMom.Py(), this->fMom.Pz()),
91 this->fTime, this->fLength, 0., pdgCode);
92
93 // Count a point for this track. ShipStack only keeps a secondary because
94 // of its points when it is configured with a minimum number of points
95 // (MCTracksWithHitsOnly / MCTracksWithHitsOrEnergyCut in
96 // run_simScript.py). With the default MCTracksWithEnergyCutOnly,
97 // secondaries below the energy cut are dropped and their points get track
98 // ID -2.
99 if (fPlanes[copyNo - 1].keepTracks) {
100 ShipStack* stack = dynamic_cast<ShipStack*>(gMC->GetStack());
101 stack->AddPoint(static_cast<DetectorId>(this->GetDetId()));
102 }
103 return kTRUE;
104 }
105
106 void ConstructGeometry() override {
107 TGeoVolume* top = gGeoManager->GetTopVolume();
108 ShipGeo::InitMedium("vacuum");
109 TGeoMedium* vacuum = gGeoManager->GetMedium("vacuum");
110 if (!vacuum) {
111 LOG(fatal) << this->GetName() << ": medium 'vacuum' not found.";
112 }
113 if (fPlanes.empty()) {
114 LOG(fatal) << this->GetName() << ": no scoring planes.";
115 }
116
117 for (size_t i = 0; i < fPlanes.size(); ++i) {
118 const Plane& pl = fPlanes[i];
119 if (pl.sizeX <= 0. || pl.sizeY <= 0. || pl.sizeZ <= 0.) {
120 LOG(fatal) << this->GetName() << ": invalid dimensions of " << pl.name;
121 }
122 TGeoVolume* plane = gGeoManager->MakeBox(
123 pl.name.c_str(), vacuum, pl.sizeX / 2., pl.sizeY / 2., pl.sizeZ / 2.);
124 plane->SetLineColor(fLineColor);
125 this->AddSensitiveVolume(plane);
126 // The copy number, i + 1, is the point detector ID: node <name>_<i + 1>.
127 top->AddNode(plane, i + 1, new TGeoTranslation(pl.x, pl.y, pl.z));
128 this->fDetector = plane;
129 LOG(info) << pl.name << ": " << pl.sizeX << " x " << pl.sizeY << " x "
130 << pl.sizeZ << " cm3 centred at (" << pl.x << ", " << pl.y
131 << ", " << pl.z << ") cm"
132 << (pl.absPdgCodes.empty() ? "" : ", PDG-filtered");
133 }
134 }
