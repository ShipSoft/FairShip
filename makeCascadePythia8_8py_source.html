4"""Pythia8 version of makeCascade.py: generate ccbar or bbbar in the hadronic cascade of the target.
5
6Same algorithm and output ntuple as makeCascade.py, so the output can be used as input file of
7run_fixedTarget.py. Every particle on the cascade stack produces a signal event with probability
8chi/chi_max, chi = K * sigma(signal) / sigma(total), and a minimum-bias event whose hadrons above
9threshold are added to the stack. Pythia8 changes the beam energy event by event only for soft QCD,
10so signal events are generated in batches in 10% momentum bins, in the centre-of-mass frame, and
11boosted to the lab along the projectile.
12
13Unlike makeCascade.py, elastic scattering does not increase the cascade depth, so depth 1 (the
14normalisation of run_fixedTarget.py) covers the beam proton up to its first inelastic interaction.
15
16With --cascade-model nucleus (default), every interaction of a cascade hadron is a hadron-nucleus
17collision built from hadron-nucleon subcollisions as in Pythia's PythiaCascade: a geometric number
18of subcollisions with the average of PythiaCascade for the target nucleus, the later ones by the
19leading hadron of the previous one. The heavy-flavour probability is then per hadron-nucleus
20collision, A sigma(signal) / sigma(hA), consistent with the normalisation of run_fixedTarget.py,
21which is per proton-nucleus interaction. --cascade-model nucleon follows makeCascade.py, with
22single collisions on free nucleons, which counts the subcollisions inside a nucleus twice.
23
24With --charm-production inclusive, charm signal events are inclusive inelastic events (SoftQCD:inelastic)
25generated at the exact momentum and direction of the projectile and kept if they contain charm, instead
26of forced HardQCD:hardccbar events. The FTFT tune describes charm production data with inclusive events;
27forced production has somewhat different x_F and pT^2 distributions. The normalisation (chi) is the same.
28"""

143# Forced-production charm K-factors equivalent to the FTFT inclusive normalisation (K = 2.48 for
144# nucleon and 2.02 for meson beams), K_incl * sigma_incl / sigma_forced, from Pythia 8.312 runs with
145# the FTFT tune. The ratio of inclusive to forced charm grows towards threshold, so the K-factor
146# depends on the beam momentum (GeV); values are interpolated in log(p) and kept flat outside the grid.
147# Measured for p p, p n, pi+ p and pi- p from 34 GeV, the charm threshold of the cascade, to 400 GeV
148# (statistical precision 1-2% above 60 GeV, 2-5% below); other beams use isospin
149# (n p = p n, n n = p p, pi+ n = pi- p, pi- n = pi+ p), antinucleons the nucleon and kaons the mean
150# pion values.
151KFORCED_CHARM_P = [34.0, 40.0, 47.0, 60.0, 100.0, 160.0, 250.0, 400.0]
152KFORCED_CHARM = {
153 (2212, 2212): [35.52, 25.24, 17.48, 13.99, 10.06, 9.11, 8.10, 7.30],
154 (2212, 2112): [38.08, 24.22, 18.93, 13.90, 10.56, 8.94, 8.25, 7.06],
155 (211, 2212): [12.93, 10.76, 9.92, 8.30, 6.49, 5.87, 5.33, 5.16],
156 (-211, 2212): [10.89, 9.22, 7.66, 7.03, 5.79, 5.30, 5.12, 5.12],

165def kforced_charm(beam_id, nucleon_id, p):
166 """Forced-production charm K-factor for beam_id on target nucleon nucleon_id at momentum p."""
167 abs_id = abs(beam_id)
168 if abs_id in (2212, 2112):
169 # n on p behaves as p on n, and n on n as p on p
170 key = (2212, nucleon_id) if abs_id == 2212 else (2212, 2112 if nucleon_id == 2212 else 2212)
171 kfactors = KFORCED_CHARM[key]
172 elif abs_id == 211:
173 # pi+ n behaves as pi- p, and pi- n as pi+ p
174 sign = 1 if beam_id > 0 else -1
175 key = (sign * 211 if nucleon_id == 2212 else -sign * 211, 2212)
176 kfactors = KFORCED_CHARM[key]
177 else:
178 kfactors = [0.5 * (a + b) for a, b in zip(KFORCED_CHARM[211, 2212], KFORCED_CHARM[-211, 2212])]
179 return float(np.interp(math.log(p), np.log(KFORCED_CHARM_P), kfactors))

203def ecm(beam_id, nucleon_id, p):
204 """Centre-of-mass energy of beam_id with momentum p on nucleon_id at rest."""
205 m_beam, m_target = mass(beam_id), mass(nucleon_id)
206 return math.sqrt(m_beam**2 + m_target**2 + 2.0 * m_target * math.sqrt(p**2 + m_beam**2))

209def n_coll_avg(a, sigma_hn, model):
210 """Average number of subcollisions in a hadron-nucleus collision (PythiaCascade::nCollAvg)."""
211
212 def n_more(i):
213 low = model.slope_lo[i] * sigma_hn
214 high = model.offset[i] + model.slope[i] * sigma_hn
215 if model.sigma_border is None:
216 return min(low, high)
217 return low if sigma_hn < model.sigma_border else high
218
219 for i, a_tab in enumerate(NCOLL_A):
220 if a == a_tab:
221 return 1.0 + n_more(i)
222 if a < a_tab:
223 w = (a_tab - a) / (a_tab - NCOLL_A[i - 1])
224 return (
225 1.0 + w * (a / NCOLL_A[i - 1]) ** (2 / 3) * n_more(i - 1) + (1 - w) * (a / a_tab) ** (2 / 3) * n_more(i)
226 )
227 raise ValueError(f"nucleus A = {a} outside 1-208")

230def nuclear_collision(coll_pythia, decayer, model, beam_id, px, py, pz, z, a, p_min):
231 """Hadron-nucleus collision as a sequence of hadron-nucleon subcollisions, as in PythiaCascade::nextColl.
232
233 coll_pythia are minimum-bias instances on proton and neutron with hadron decays off, so that a leading
234 resonance collides again itself; the decays are done by decayer once the collision is complete.
235
236 Returns the process code of the first non-elastic subcollision (102 if all were elastic), the
237 number of subcollisions, and the final particles (id, px, py, pz, p) of all subcollisions, without
238 the hadrons that collided again.
239 Subcollisions stop once the leading hadron is below p_min, which no longer adds particles above it.
240 """
241 p = math.sqrt(px**2 + py**2 + pz**2)
242 sigma_hn = coll_pythia[0].getSigmaTotal(beam_id, 2212, ecm(beam_id, 2212, p))
243 prob_more = 1.0 - 1.0 / n_coll_avg(a, sigma_hn, model)
244 direction = (px / p, py / p, pz / p)
245 n_p, n_n = z, a - z
246 final, latest = [], []
247 projectile = (beam_id, px, py, pz)
248 first_code, n_coll = 102, 0
249 for i_coll in range(1, a + 1):
250 proc_type = 0
251 if i_coll > 1:
252 if random.random() > prob_more:
253 break
254 hadrons = [f for f in latest if f[5]]
255 if not hadrons:
256 break
257 lead = max(hadrons, key=lambda f: f[1] * direction[0] + f[2] * direction[1] + f[3] * direction[2])
258 if lead[4] < p_min:
259 break
260 final.remove(lead)
261 projectile = lead[:4]
262 if ecm(lead[0], 2212, lead[4]) > model.sd_min_ecm:
263 proc_type = 4 if random.random() < model.prob_sd else model.other_process
264 on_proton = random.random() < n_p / (n_p + n_n)
265 if on_proton:
266 n_p -= 1
267 else:
268 n_n -= 1
269 pythia = coll_pythia[0 if on_proton else 1]
270 pythia.setBeamIDs(projectile[0], TARGET_NUCLEONS[0 if on_proton else 1])
271 pythia.setKinematics(*projectile[1:], 0.0, 0.0, 0.0)
272 for _ in range(100):
273 if pythia.next(proc_type) if proc_type else pythia.next():
274 break
275 else:
276 raise RuntimeError(f"Pythia8 failed to generate a subcollision for {projectile[0]}")
277 if first_code == 102:
278 first_code = pythia.infoPython().code()
279 n_coll += 1
280 event = pythia.event
281 latest = [
282 (event[i].id(), event[i].px(), event[i].py(), event[i].pz(), event[i].pAbs(), event[i].isHadron())
283 + (event[i].e(), event[i].m())
284 for i in range(event.size())
285 if event[i].isFinal()
286 ]
287 final += latest
288 # decays of the final particles, the cascade species staying stable
289 decayer.event.reset()
290 for part_id, part_px, part_py, part_pz, _, _, part_e, part_m in final:
291 decayer.event.append(part_id, 1, 0, 0, part_px, part_py, part_pz, part_e, part_m)
292 if not decayer.moreDecays():
293 raise RuntimeError("Pythia8 failed to decay the final particles of a hadron-nucleus collision")
294 event = decayer.event
295 final = [
296 (event[i].id(), event[i].px(), event[i].py(), event[i].pz(), event[i].pAbs())
297 for i in range(event.size())
298 if event[i].isFinal()
299 ]
300 return first_code, n_coll, final

314class PythiaFactory:
315 """Create initialised Pythia8 instances with the tune, their own seed and stable cascade and signal
316 particles. The seeds are drawn from a generator of their own, so that they do not follow the draws
317 of the cascade; Pythia8 accepts seeds up to 900000000."""
318
319 def __init__(self, seed, tune_settings, stable_ids):
320 self.seeds = random.Random(f"pythia8-{seed}")
321 self.tune_settings = tune_settings
322 self.stable_ids = stable_ids
323 self.n_instances = 0
324
325 def __call__(self, settings):
326 self.n_instances += 1
327 pythia = pythia8.Pythia("", False)
328 for setting in [*self.tune_settings, *settings, "Print:quiet = on", "Random:setSeed = on"]:
329 pythia.readString(setting)
330 pythia.readString(f"Random:seed = {self.seeds.randint(1, 900000000)}")
331 for pdg_id in self.stable_ids:
332 pythia.readString(f"{pdg_id}:mayDecay = off")
333 if not pythia.init():
334 raise RuntimeError(f"Pythia8 initialisation failed for {settings}")
335 return pythia

338def next_event(pythia):
339 """Generate the next event, giving up after 100 failed attempts."""
340 for _ in range(100):
341 if pythia.next():
342 return
343 raise RuntimeError(
344 f"Pythia8 failed to generate an event for beams {pythia.infoPython().idA()} on {pythia.infoPython().idB()}"
345 )

348class SignalEvents:
349 """Signal events in momentum bins p_beam / SIGNAL_BIN_RATIO^i, generated in batches of increasing
350 size, one Pythia8 instance per batch."""
351
352 def __init__(self, new_pythia, process, signal_ids, p_beam):
353 self.new_pythia = new_pythia
354 self.process = process
355 self.signal_ids = signal_ids
356 self.p_beam = p_beam
357 self.buffers = {}
358
359 def next(self, beam_id, nucleon_id, p):
360 """Return process code and signal hadrons (id, px, py, pz, E, m) in the CM frame of an event at
361 momentum p."""
362 p_bin = round(math.log(self.p_beam / p) / math.log(SIGNAL_BIN_RATIO))
363 buffer, n_fill = self.buffers.setdefault((beam_id, nucleon_id, p_bin), ([], 5))
364 if not buffer:
365 n_fill = min(2 * n_fill, 1000)
366 self.buffers[beam_id, nucleon_id, p_bin] = (buffer, n_fill)
367 pythia = self.new_pythia(
368 [self.process, *cm_beam_settings(beam_id, nucleon_id, self.p_beam / SIGNAL_BIN_RATIO**p_bin)]
369 )
370 for _ in range(n_fill):
371 next_event(pythia)
372 event = pythia.event
373 hadrons = [
374 event[i] for i in range(event.size()) if event[i].idAbs() in self.signal_ids and event[i].isFinal()
375 ]
376 buffer.append(
377 (pythia.infoPython().code(), [(h.id(), h.px(), h.py(), h.pz(), h.e(), h.m()) for h in hadrons])
378 )
379 return buffer.pop()

382# safety limit on the inclusive events tried for one charm signal event; near the 34 GeV threshold
383# about 1 in 3e5 inelastic events contains charm
384MAX_INCLUSIVE_TRIES = 100_000_000

387class InclusiveSignalEvents:
388 """Inclusive inelastic events with signal hadrons, at the exact projectile momentum, one Pythia8 instance
389 per target nucleon. Events are hadronised only if they contain a heavy quark, which saves most of the
390 time spent on events without one."""
391
392 def __init__(self, new_pythia, signal_ids, heavy_quark, p_beam):
393 self.signal_ids = signal_ids
394 self.heavy_quark = heavy_quark
395 self.pythia = [
396 new_pythia(
397 ["SoftQCD:inelastic = on", "HadronLevel:all = off"]
398 + ["Beams:frameType = 3", "Beams:allowVariableEnergy = on", "Beams:allowIDAswitch = on"]
399 + [f"Beams:idB = {nucleon_id}", f"Beams:pzA = {p_beam}", "Beams:pzB = 0."]
400 )
401 for nucleon_id in TARGET_NUCLEONS
402 ]
403 self.n_tried = 0
404
405 def next(self, beam_id, i_nucleon, px, py, pz):
406 """Return process code and signal hadrons (id, px, py, pz, E, m) in the lab frame."""
407 pythia = self.pythia[i_nucleon]
408 pythia.setBeamIDs(beam_id, TARGET_NUCLEONS[i_nucleon])
409 pythia.setKinematics(px, py, pz, 0.0, 0.0, 0.0)
410 for _ in range(MAX_INCLUSIVE_TRIES):
411 next_event(pythia)
412 self.n_tried += 1
413 event = pythia.event
414 if not any(event[i].idAbs() == self.heavy_quark for i in range(event.size())):
415 continue
416 if not pythia.forceHadronLevel():
417 continue
418 event = pythia.event
419 hadrons = [
420 event[i] for i in range(event.size()) if event[i].idAbs() in self.signal_ids and event[i].isFinal()
421 ]
422 if hadrons:
423 return pythia.infoPython().code(), [(h.id(), h.px(), h.py(), h.pz(), h.e(), h.m()) for h in hadrons]
424 raise RuntimeError(
425 f"no charm in {MAX_INCLUSIVE_TRIES} inclusive events of {beam_id} on {TARGET_NUCLEONS[i_nucleon]}"
426 f" at p = ({px:.1f}, {py:.1f}, {pz:.1f}) GeV"
427 )

506 args = parse_args()
507 signal = SIGNAL[args.heavy_quark]

521 ["SoftQCD:all = on", "Beams:frameType = 3", "Beams:allowVariableEnergy = on", "Beams:allowIDAswitch = on"]
522 + [f"Beams:idB = {nucleon_id}", f"Beams:pzA = {args.p_beam}", "Beams:pzB = 0."]
523 )

528 # The grid is logarithmic and chi is interpolated in log(chi) vs log(p): chi varies by orders of
529 # magnitude over the range and most of that variation sits close to the threshold.
530 p_grid = np.geomspace(p_threshold, args.p_beam, args.n_momentum_points)
531 log_p_grid = np.log(p_grid)

533 for pdg_id in CASCADE_IDS:
534 for beam_id in sorted({pdg_id, -pdg_id}):
535 if not PDG.GetParticle(beam_id):

540 pythia = new_pythia(
541 [signal_process, "PartonLevel:all = off", "HadronLevel:all = off"]
542 + cm_beam_settings(beam_id, nucleon_id, p)
543 )
544 for _ in range(args.n_sigma_events):
545 next_event(pythia)
546 sigma[beam_id, i_nucleon].append(
547 signal_kfactor(kfactors, beam_id, nucleon_id, p) * pythia.infoPython().sigmaGen()
548 )
549 sigma_total = [
550 mbias_pythia[i_nucleon].getSigmaTotal(beam_id, nucleon_id, ecm(beam_id, nucleon_id, p))
551 for p in p_grid
552 ]
553 log_chi[beam_id, i_nucleon] = np.log([s / t for s, t in zip(sigma[beam_id, i_nucleon], sigma_total)])
554 print(
555 f"K*sigma and chi at {args.p_beam} GeV for {beam_id} on {nucleon_id}: "
556 f"{sigma[beam_id, i_nucleon][-1]:.3e} mb, {math.exp(log_chi[beam_id, i_nucleon][-1]):.3e}"
557 )

559 version = round(mbias_pythia[0].settings.parm("Pythia:versionNumber"), 3) # codespell:ignore parm
560 model = nucleus_model(version)
561 coll_pythia, decayer = [], None
562 if args.cascade_model == "nucleus":
563 # subcollisions with hadron decays off, and the decays once a collision is complete
564 coll_pythia = [
565 new_pythia(
566 ["SoftQCD:all = on", "HadronLevel:Decay = off", "Beams:frameType = 3"]
567 + ["Beams:allowVariableEnergy = on", "Beams:allowIDAswitch = on", f"Beams:idB = {nucleon_id}"]
568 + [f"Beams:pzA = {args.p_beam}", "Beams:pzB = 0."]
569 )
570 for nucleon_id in TARGET_NUCLEONS
571 ]
572 decayer = new_pythia(["ProcessLevel:all = off"])
573 # heavy-flavour probability per hadron-nucleus collision: A sigma_QQ(hN) / sigma(hA), with
574 # sigma(hA) = A sigma_tot(hp) / <n_coll> as in PythiaCascade
575 for beam_id in {b for b, _ in log_chi}:
576 chi_nucleus = []
577 for i, p in enumerate(p_grid):
578 sigma_tot = mbias_pythia[0].getSigmaTotal(beam_id, 2212, ecm(beam_id, 2212, p))
579 sigma_n = proton_fraction * sigma[beam_id, 0][i] + (1.0 - proton_fraction) * sigma[beam_id, 1][i]
580 chi_nucleus.append(n_coll_avg(a_target, sigma_tot, model) * sigma_n / sigma_tot)
581 log_chi[beam_id, 0] = log_chi[beam_id, 1] = np.log(chi_nucleus)
582 chi_max = max(np.exp(c).max() for c in log_chi.values())
583 # cross section per nucleon of the target composition, at the beam energy, for the normalisation
584 sigma_QQ = proton_fraction * sigma[2212, 0][-1] + (1.0 - proton_fraction) * sigma[2212, 1][-1]
585 print(f"K*sigma per nucleon at {args.p_beam} GeV: {1e3 * sigma_QQ:.2f} ub")
586
587 signal_events = SignalEvents(new_pythia, signal_process, signal_ids, args.p_beam)
588 inclusive_events = (
589 InclusiveSignalEvents(new_pythia, signal_ids, args.heavy_quark, args.p_beam)
590 if args.charm_production == "inclusive" and args.heavy_quark == 4
591 else None
592 )

594 output_file = ROOT.TFile.Open(args.output, "RECREATE")
595 ntuple = ROOT.TNtuple(
596 "pythia6",
597 "pythia8 heavy flavour",
598 "id:px:py:pz:E:M:mid:mpx:mpy:mpz:mE:mM:k:a0:a1:a2:a3:a4:a5:a6:a7:a8:a9:a10:a11:a12:a13:a14:a15:\
599s0:s1:s2:s3:s4:s5:s6:s7:s8:s9:s10:s11:s12:s13:s14:s15:n_hadrons",
600 )
601 # number of signal particles per cascade depth, used by FixedTargetGenerator for the normalisation
602 depth_hist = ROOT.TH1F("2", "nr signal per cascade depth", 50, 0.5, 50.5)
603 # cross section per nucleon [mb] this file was generated with, read back by run_fixedTarget.py to
604 # scale chicc/chibb, so that the normalisation follows the beam energy, tune and target of the file
605 # hadd adds TParameters by default; keep the value of the first file when cascade files are merged
606 sigma_parameter = ROOT.TParameter("double")("sigma_QQ", sigma_QQ) # type: ignore[missing-attribute]
607 sigma_parameter.SetMergeMode("f")
608 sigma_parameter.Write()

612 if i_pot % 1000 == 0:
613 print("Generate event ", i_pot)
614 # stack: PID, px, py, pz, cascade depth, ancestors, interaction processes
615 stack: list[tuple[int, float, float, float, int, list[int], list[int]]] = []
616 stack.append((2212, 0.0, 0.0, args.p_beam, 1, [2212] + 99 * [0], 100 * [0]))
617 while stack:
618 beam_id, px, py, pz, depth, ancestors, processes = stack.pop()
619 p = math.sqrt(px**2 + py**2 + pz**2)
620 i_nucleon = 0 if random.random() < proton_fraction else 1
621 nucleon_id = TARGET_NUCLEONS[i_nucleon]
622 if math.exp(np.interp(math.log(p), log_p_grid, log_chi[beam_id, i_nucleon])) / chi_max > random.random():

624 beam = ROOT.TLorentzVector(px, py, pz, math.sqrt(p**2 + beam_mass**2))
625 if inclusive_events is not None:
626 code, hadrons = inclusive_events.next(beam_id, i_nucleon, px, py, pz)
627 else:
628 code, hadrons = signal_events.next(beam_id, nucleon_id, p)

630 n_processes = min(depth - 1, 15)

632 hadron = ROOT.TLorentzVector(*p4)
633 if inclusive_events is None: # forced events are generated in the CM frame
634 hadron.RotateUz(beam.Vect().Unit()) # type: ignore[missing-attribute]
635 hadron.Boost(boost) # type: ignore[missing-attribute]

637 row += [beam_id, px, py, pz, beam.E(), beam_mass, depth]
638 row += ancestors[:16] + processes[:n_processes] + [code] + (15 - n_processes) * [0]
639 # all signal hadrons of the event, read back together by FixedTargetGenerator
640 row += [len(hadrons)]
641 ntuple.Fill(array("f", row))
642 depth_hist.Fill(depth)
643 if args.cascade_model == "nucleus":
644 # hadron-nucleus collision to add new cascade particles to the stack
645 code, _, final = nuclear_collision(
646 coll_pythia, decayer, model, beam_id, px, py, pz, z_target, a_target, p_threshold
647 )
648 if code == 102:
649 # only elastic subcollisions: the cascade generation is kept
650 leading = max(final, key=lambda f: f[4])
651 if leading[4] > p_threshold and len(stack) < 999:
652 stack.append((beam_id, *leading[1:4], depth, ancestors, processes))
653 continue
654 new_depth = min(depth + 1, 98)
655 if depth == 1: # first inelastic interaction process of the beam proton
656 processes = [code] + processes[1:]
657 for part_id, part_px, part_py, part_pz, part_p in final:
658 if abs(part_id) in CASCADE_IDS and part_p > p_threshold and len(stack) < 999:
659 new_ancestors = ancestors[: new_depth - 1] + [part_id] + ancestors[new_depth:]
660 new_processes = processes[: new_depth - 1] + [code] + processes[new_depth:]
661 stack.append((part_id, part_px, part_py, part_pz, new_depth, new_ancestors, new_processes))
662 continue
663 # minimum-bias event to add new cascade particles to the stack
664 i_nucleon = 0 if random.random() < proton_fraction else 1
665 pythia = mbias_pythia[i_nucleon]
666 pythia.setBeamIDs(beam_id, TARGET_NUCLEONS[i_nucleon])
667 pythia.setKinematics(px, py, pz, 0.0, 0.0, 0.0)
668 next_event(pythia)
669 code = pythia.infoPython().code()
670 if code == 102:
671 # Elastic scattering does not start a new cascade generation: the scattered hadron keeps
672 # its depth. Depth 1 is then everything the beam proton produces up to and including its
673 # first inelastic interaction, which is what chicc/chibb in run_fixedTarget.py normalise.
674 final = [pythia.event[i] for i in range(pythia.event.size()) if pythia.event[i].isFinal()]
675 leading = max(final, key=lambda part: part.pAbs())
676 if leading.pAbs() > p_threshold and len(stack) < 999:
677 stack.append((beam_id, leading.px(), leading.py(), leading.pz(), depth, ancestors, processes))
678 continue
679 new_depth = min(depth + 1, 98)
680 if depth == 1: # first inelastic interaction process of the beam proton
681 processes = [code] + processes[1:]
682 for i in range(pythia.event.size()):
683 part = pythia.event[i]
684 if part.idAbs() in CASCADE_IDS and part.isFinal() and part.pAbs() > p_threshold and len(stack) < 999:
685 new_ancestors = ancestors[: new_depth - 1] + [part.id()] + ancestors[new_depth:]
686 new_processes = processes[: new_depth - 1] + [code] + processes[new_depth:]
687 stack.append((part.id(), part.px(), part.py(), part.pz(), new_depth, new_ancestors, new_processes))
688
689 print(f"Generated {args.n_pot} p.o.t. in {time.time() - t0} s with {new_pythia.n_instances} Pythia8 instances")
690 if inclusive_events is not None:
691 print(f"{inclusive_events.n_tried} inclusive events generated for the charm signal")

693 output_file.Close()
