4"""Pythia8 version of makeCascade.py: generate ccbar or bbbar in the hadronic cascade of the target.
5
6Same algorithm and output ntuple as makeCascade.py, so the output can be used as input file of
7run_fixedTarget.py. Every particle on the cascade stack produces a signal event with probability
8chi/chi_max, chi = K * sigma(signal) / sigma(total), and a minimum-bias event whose hadrons above
9threshold are added to the stack. Pythia8 changes the beam energy event by event only for soft QCD,
10so signal events are generated in batches in 10% momentum bins, in the centre-of-mass frame, and
11boosted to the lab along the projectile.
12
13Unlike makeCascade.py, elastic scattering does not increase the cascade depth, so depth 1 (the
14normalisation of run_fixedTarget.py) covers the beam proton up to its first inelastic interaction.
15"""

77# Forced-production charm K-factors equivalent to the FTFT inclusive normalisation (K = 2.48 for
78# nucleon and 2.02 for meson beams), K_incl * sigma_incl / sigma_forced, from Pythia 8.312 runs with
79# the FTFT tune. The ratio of inclusive to forced charm grows towards threshold, so the K-factor
80# depends on the beam momentum (GeV); values are interpolated in log(p) and kept flat outside the grid.
81# Measured for p p, p n, pi+ p and pi- p from 34 GeV, the charm threshold of the cascade, to 400 GeV
82# (statistical precision 1-2% above 60 GeV, 2-5% below); other beams use isospin
83# (n p = p n, n n = p p, pi+ n = pi- p, pi- n = pi+ p), antinucleons the nucleon and kaons the mean
84# pion values.
85KFORCED_CHARM_P = [34.0, 40.0, 47.0, 60.0, 100.0, 160.0, 250.0, 400.0]
86KFORCED_CHARM = {
87 (2212, 2212): [35.52, 25.24, 17.48, 13.99, 10.06, 9.11, 8.10, 7.30],
88 (2212, 2112): [38.08, 24.22, 18.93, 13.90, 10.56, 8.94, 8.25, 7.06],
89 (211, 2212): [12.93, 10.76, 9.92, 8.30, 6.49, 5.87, 5.33, 5.16],
90 (-211, 2212): [10.89, 9.22, 7.66, 7.03, 5.79, 5.30, 5.12, 5.12],
91}

99def kforced_charm(beam_id, nucleon_id, p):
100 """Forced-production charm K-factor for beam_id on target nucleon nucleon_id at momentum p."""
101 abs_id = abs(beam_id)
102 if abs_id in (2212, 2112):
103 # n on p behaves as p on n, and n on n as p on p
104 key = (2212, nucleon_id) if abs_id == 2212 else (2212, 2112 if nucleon_id == 2212 else 2212)
105 kfactors = KFORCED_CHARM[key]
106 elif abs_id == 211:
107 # pi+ n behaves as pi- p, and pi- n as pi+ p
108 sign = 1 if beam_id > 0 else -1
109 key = (sign * 211 if nucleon_id == 2212 else -sign * 211, 2212)
110 kfactors = KFORCED_CHARM[key]
111 else:
112 kfactors = [0.5 * (a + b) for a, b in zip(KFORCED_CHARM[211, 2212], KFORCED_CHARM[-211, 2212])]
113 return float(np.interp(math.log(p), np.log(KFORCED_CHARM_P), kfactors))

116def signal_kfactor(kfactors, beam_id, nucleon_id, p):
117 """K-factor applied to the forced signal cross section of beam_id on nucleon_id at momentum p.
118
119 kfactors: (nucleon beams, meson beams), or None for the momentum-dependent charm K-factors.
120 """
121 if kfactors is None:
122 return kforced_charm(beam_id, nucleon_id, p)
123 return kfactors[0] if abs(beam_id) in (2212, 2112) else kfactors[1]

132def ecm(beam_id, nucleon_id, p):
133 """Centre-of-mass energy of beam_id with momentum p on nucleon_id at rest."""
134 m_beam, m_target = mass(beam_id), mass(nucleon_id)
135 return math.sqrt(m_beam**2 + m_target**2 + 2.0 * m_target * math.sqrt(p**2 + m_beam**2))

149class PythiaFactory:
150 """Create initialised Pythia8 instances with the tune, their own seed and stable cascade and signal
151 particles. The seeds are drawn from a generator of their own, so that they do not follow the draws
152 of the cascade; Pythia8 accepts seeds up to 900000000."""
153
154 def __init__(self, seed, tune_settings, stable_ids):
155 self.seeds = random.Random(f"pythia8-{seed}")
156 self.tune_settings = tune_settings
157 self.stable_ids = stable_ids
158 self.n_instances = 0
159
160 def __call__(self, settings):
161 self.n_instances += 1
162 pythia = pythia8.Pythia("", False)
163 for setting in [*self.tune_settings, *settings, "Print:quiet = on", "Random:setSeed = on"]:
164 pythia.readString(setting)
165 pythia.readString(f"Random:seed = {self.seeds.randint(1, 900000000)}")
166 for pdg_id in self.stable_ids:
167 pythia.readString(f"{pdg_id}:mayDecay = off")
168 if not pythia.init():
169 raise RuntimeError(f"Pythia8 initialisation failed for {settings}")
170 return pythia

173def next_event(pythia):
174 """Generate the next event, giving up after 100 failed attempts."""
175 for _ in range(100):
176 if pythia.next():
177 return
178 raise RuntimeError(
179 f"Pythia8 failed to generate an event for beams {pythia.infoPython().idA()} on {pythia.infoPython().idB()}"
180 )

183class SignalEvents:
184 """Signal events in momentum bins p_beam / SIGNAL_BIN_RATIO^i, generated in batches of increasing
185 size, one Pythia8 instance per batch."""
186
187 def __init__(self, new_pythia, process, signal_ids, p_beam):
188 self.new_pythia = new_pythia
189 self.process = process
190 self.signal_ids = signal_ids
191 self.p_beam = p_beam
192 self.buffers = {}
193
194 def next(self, beam_id, nucleon_id, p):
195 """Return process code and signal hadrons (id, px, py, pz, E, m) in the CM frame of an event at
196 momentum p."""
197 p_bin = round(math.log(self.p_beam / p) / math.log(SIGNAL_BIN_RATIO))
198 buffer, n_fill = self.buffers.setdefault((beam_id, nucleon_id, p_bin), ([], 5))
199 if not buffer:
200 n_fill = min(2 * n_fill, 1000)
201 self.buffers[beam_id, nucleon_id, p_bin] = (buffer, n_fill)
202 pythia = self.new_pythia(
203 [self.process, *cm_beam_settings(beam_id, nucleon_id, self.p_beam / SIGNAL_BIN_RATIO**p_bin)]
204 )
205 for _ in range(n_fill):
206 next_event(pythia)
207 event = pythia.event
208 hadrons = [
209 event[i] for i in range(event.size()) if event[i].idAbs() in self.signal_ids and event[i].isFinal()
210 ]
211 buffer.append(
212 (pythia.infoPython().code(), [(h.id(), h.px(), h.py(), h.pz(), h.e(), h.m()) for h in hadrons])
213 )
214 return buffer.pop()

279def main():
280 args = parse_args()
281 signal = SIGNAL[args.heavy_quark]
282 p_threshold, signal_ids, signal_process = signal.p_threshold, signal.ids, signal.process
283 proton_fraction = PROTON_FRACTION[args.target_composition]
284 if args.pythia8_tune == "FTFT":
285 tune_settings, kfactors = FTFT_SETTINGS, signal.kfactors
286 else:
287 tune_settings, kfactors = [], (1.0, 1.0)
288
289 random.seed(args.seed)
290 new_pythia = PythiaFactory(args.seed, tune_settings, [*CASCADE_IDS, *signal_ids])
291
292 # minimum-bias events with variable beam particle and energy, one instance per target nucleon
293 mbias_pythia = [
294 new_pythia(
295 ["SoftQCD:all = on", "Beams:frameType = 3", "Beams:allowVariableEnergy = on", "Beams:allowIDAswitch = on"]
296 + [f"Beams:idB = {nucleon_id}", f"Beams:pzA = {args.p_beam}", "Beams:pzB = 0."]
297 )
298 for nucleon_id in TARGET_NUCLEONS
299 ]
300
301 # chi = K * sigma(signal) / sigma(total) vs momentum for all beam and target particles.
302 # The grid is logarithmic and chi is interpolated in log(chi) vs log(p): chi varies by orders of
303 # magnitude over the range and most of that variation sits close to the threshold.
304 p_grid = np.geomspace(p_threshold, args.p_beam, args.n_momentum_points)
305 log_p_grid = np.log(p_grid)
306 log_chi, sigma = {}, {}
307 for pdg_id in CASCADE_IDS:
308 for beam_id in sorted({pdg_id, -pdg_id}):
309 if not PDG.GetParticle(beam_id):
310 continue
311 for i_nucleon, nucleon_id in enumerate(TARGET_NUCLEONS):
312 sigma[beam_id, i_nucleon] = []
313 for p in p_grid:
314 pythia = new_pythia(
315 [signal_process, "PartonLevel:all = off", "HadronLevel:all = off"]
316 + cm_beam_settings(beam_id, nucleon_id, p)
317 )
318 for _ in range(args.n_sigma_events):
319 next_event(pythia)
320 sigma[beam_id, i_nucleon].append(
321 signal_kfactor(kfactors, beam_id, nucleon_id, p) * pythia.infoPython().sigmaGen()
322 )
323 sigma_total = [
324 mbias_pythia[i_nucleon].getSigmaTotal(beam_id, nucleon_id, ecm(beam_id, nucleon_id, p))
325 for p in p_grid
326 ]
327 log_chi[beam_id, i_nucleon] = np.log([s / t for s, t in zip(sigma[beam_id, i_nucleon], sigma_total)])
328 print(
329 f"K*sigma and chi at {args.p_beam} GeV for {beam_id} on {nucleon_id}: "
330 f"{sigma[beam_id, i_nucleon][-1]:.3e} mb, {math.exp(log_chi[beam_id, i_nucleon][-1]):.3e}"
331 )
332 chi_max = max(np.exp(c).max() for c in log_chi.values())
333 # cross section per nucleon of the target composition, at the beam energy, for the normalisation
334 sigma_QQ = proton_fraction * sigma[2212, 0][-1] + (1.0 - proton_fraction) * sigma[2212, 1][-1]
335 print(f"K*sigma per nucleon at {args.p_beam} GeV: {1e3 * sigma_QQ:.2f} ub")
336
337 signal_events = SignalEvents(new_pythia, signal_process, signal_ids, args.p_beam)
338
339 output_file = ROOT.TFile.Open(args.output, "RECREATE")
340 ntuple = ROOT.TNtuple(
341 "pythia6",
342 "pythia8 heavy flavour",
343 "id:px:py:pz:E:M:mid:mpx:mpy:mpz:mE:mM:k:a0:a1:a2:a3:a4:a5:a6:a7:a8:a9:a10:a11:a12:a13:a14:a15:\
344s0:s1:s2:s3:s4:s5:s6:s7:s8:s9:s10:s11:s12:s13:s14:s15",
345 )
346 # number of signal particles per cascade depth, used by FixedTargetGenerator for the normalisation
347 depth_hist = ROOT.TH1F("2", "nr signal per cascade depth", 50, 0.5, 50.5)
348 # cross section per nucleon [mb] this file was generated with, read back by run_fixedTarget.py to
349 # scale chicc/chibb, so that the normalisation follows the beam energy, tune and target of the file
350 ROOT.TParameter("double")("sigma_QQ", sigma_QQ).Write() # type: ignore[missing-attribute]
351
352 t0 = time.time()
353 for i_pot in range(args.n_pot):
354 if i_pot % 1000 == 0:
355 print("Generate event ", i_pot)
356 # stack: PID, px, py, pz, cascade depth, ancestors, interaction processes
357 stack: list[tuple[int, float, float, float, int, list[int], list[int]]] = []
358 stack.append((2212, 0.0, 0.0, args.p_beam, 1, [2212] + 99 * [0], 100 * [0]))
359 while stack:
360 beam_id, px, py, pz, depth, ancestors, processes = stack.pop()
361 p = math.sqrt(px**2 + py**2 + pz**2)
362 i_nucleon = 0 if random.random() < proton_fraction else 1
363 nucleon_id = TARGET_NUCLEONS[i_nucleon]
364 if math.exp(np.interp(math.log(p), log_p_grid, log_chi[beam_id, i_nucleon])) / chi_max > random.random():
365 code, hadrons = signal_events.next(beam_id, nucleon_id, p)
366 beam_mass = mass(beam_id)
367 beam = ROOT.TLorentzVector(px, py, pz, math.sqrt(p**2 + beam_mass**2))
368 boost = (beam + ROOT.TLorentzVector(0, 0, 0, mass(nucleon_id))).BoostVector() # type: ignore[missing-attribute]
369 n_processes = min(depth - 1, 15)
370 for hadron_id, *p4, hadron_mass in hadrons:
371 hadron = ROOT.TLorentzVector(*p4)
372 hadron.RotateUz(beam.Vect().Unit()) # type: ignore[missing-attribute]
373 hadron.Boost(boost) # type: ignore[missing-attribute]
374 row = [hadron_id, hadron.Px(), hadron.Py(), hadron.Pz(), hadron.E(), hadron_mass]
375 row += [beam_id, px, py, pz, beam.E(), beam_mass, depth]
376 row += ancestors[:16] + processes[:n_processes] + [code] + (15 - n_processes) * [0]
377 ntuple.Fill(array("f", row))
378 depth_hist.Fill(depth)
379 # minimum-bias event to add new cascade particles to the stack
380 i_nucleon = 0 if random.random() < proton_fraction else 1
381 pythia = mbias_pythia[i_nucleon]
382 pythia.setBeamIDs(beam_id, TARGET_NUCLEONS[i_nucleon])
383 pythia.setKinematics(px, py, pz, 0.0, 0.0, 0.0)
384 next_event(pythia)
385 code = pythia.infoPython().code()
386 if code == 102:
387 # Elastic scattering does not start a new cascade generation: the scattered hadron keeps
388 # its depth. Depth 1 is then everything the beam proton produces up to and including its
389 # first inelastic interaction, which is what chicc/chibb in run_fixedTarget.py normalise.
390 final = [pythia.event[i] for i in range(pythia.event.size()) if pythia.event[i].isFinal()]
391 leading = max(final, key=lambda part: part.pAbs())
392 if leading.pAbs() > p_threshold and len(stack) < 999:
393 stack.append((beam_id, leading.px(), leading.py(), leading.pz(), depth, ancestors, processes))
394 continue
395 new_depth = min(depth + 1, 98)
396 if depth == 1: # first inelastic interaction process of the beam proton
397 processes = [code] + processes[1:]
398 for i in range(pythia.event.size()):
399 part = pythia.event[i]
400 if part.idAbs() in CASCADE_IDS and part.isFinal() and part.pAbs() > p_threshold and len(stack) < 999:
401 new_ancestors = ancestors[: new_depth - 1] + [part.id()] + ancestors[new_depth:]
402 new_processes = processes[: new_depth - 1] + [code] + processes[new_depth:]
403 stack.append((part.id(), part.px(), part.py(), part.pz(), new_depth, new_ancestors, new_processes))
404
405 print(f"Generated {args.n_pot} p.o.t. in {time.time() - t0} s with {new_pythia.n_instances} Pythia8 instances")
406 output_file.Write()
407 output_file.Close()
408 print(f"Output file is {args.output}")
