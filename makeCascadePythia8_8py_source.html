4"""Pythia8 version of makeCascade.py: generate ccbar or bbbar in the hadronic cascade of the target.
5
6Same algorithm and output ntuple as makeCascade.py, so the output can be used as input file of
7run_fixedTarget.py. Every particle on the cascade stack produces a signal event with probability
8chi/chi_max, chi = K * sigma(signal) / sigma(total), and a minimum-bias event whose hadrons above
9threshold are added to the stack. Pythia8 changes the beam energy event by event only for soft QCD,
10so signal events are generated in batches in 10% momentum bins, in the centre-of-mass frame, and
11boosted to the lab along the projectile.
12
13Unlike makeCascade.py, elastic scattering does not increase the cascade depth, so depth 1 (the
14normalisation of run_fixedTarget.py) covers the beam proton up to its first inelastic interaction.
15
16With --charm-production inclusive, charm signal events are inclusive inelastic events (SoftQCD:inelastic)
17generated at the exact momentum and direction of the projectile and kept if they contain charm, instead
18of forced HardQCD:hardccbar events. The FTFT tune describes charm production data with inclusive events;
19forced production has somewhat different x_F and pT^2 distributions. The normalisation (chi) is the same.
20"""

82# Forced-production charm K-factors equivalent to the FTFT inclusive normalisation (K = 2.48 for
83# nucleon and 2.02 for meson beams), K_incl * sigma_incl / sigma_forced, from Pythia 8.312 runs with
84# the FTFT tune. The ratio of inclusive to forced charm grows towards threshold, so the K-factor
85# depends on the beam momentum (GeV); values are interpolated in log(p) and kept flat outside the grid.
86# Measured for p p, p n, pi+ p and pi- p from 34 GeV, the charm threshold of the cascade, to 400 GeV
87# (statistical precision 1-2% above 60 GeV, 2-5% below); other beams use isospin
88# (n p = p n, n n = p p, pi+ n = pi- p, pi- n = pi+ p), antinucleons the nucleon and kaons the mean
89# pion values.
90KFORCED_CHARM_P = [34.0, 40.0, 47.0, 60.0, 100.0, 160.0, 250.0, 400.0]
91KFORCED_CHARM = {
92 (2212, 2212): [35.52, 25.24, 17.48, 13.99, 10.06, 9.11, 8.10, 7.30],
93 (2212, 2112): [38.08, 24.22, 18.93, 13.90, 10.56, 8.94, 8.25, 7.06],
94 (211, 2212): [12.93, 10.76, 9.92, 8.30, 6.49, 5.87, 5.33, 5.16],
95 (-211, 2212): [10.89, 9.22, 7.66, 7.03, 5.79, 5.30, 5.12, 5.12],
96}

104def kforced_charm(beam_id, nucleon_id, p):
105 """Forced-production charm K-factor for beam_id on target nucleon nucleon_id at momentum p."""
106 abs_id = abs(beam_id)
107 if abs_id in (2212, 2112):
108 # n on p behaves as p on n, and n on n as p on p
109 key = (2212, nucleon_id) if abs_id == 2212 else (2212, 2112 if nucleon_id == 2212 else 2212)
110 kfactors = KFORCED_CHARM[key]
111 elif abs_id == 211:
112 # pi+ n behaves as pi- p, and pi- n as pi+ p
113 sign = 1 if beam_id > 0 else -1
114 key = (sign * 211 if nucleon_id == 2212 else -sign * 211, 2212)
115 kfactors = KFORCED_CHARM[key]
116 else:
117 kfactors = [0.5 * (a + b) for a, b in zip(KFORCED_CHARM[211, 2212], KFORCED_CHARM[-211, 2212])]
118 return float(np.interp(math.log(p), np.log(KFORCED_CHARM_P), kfactors))

121def signal_kfactor(kfactors, beam_id, nucleon_id, p):
122 """K-factor applied to the forced signal cross section of beam_id on nucleon_id at momentum p.
123
124 kfactors: (nucleon beams, meson beams), or None for the momentum-dependent charm K-factors.
125 """
126 if kfactors is None:
127 return kforced_charm(beam_id, nucleon_id, p)
128 return kfactors[0] if abs(beam_id) in (2212, 2112) else kfactors[1]

131def mass(pdg_id):

137def ecm(beam_id, nucleon_id, p):
138 """Centre-of-mass energy of beam_id with momentum p on nucleon_id at rest."""
139 m_beam, m_target = mass(beam_id), mass(nucleon_id)
140 return math.sqrt(m_beam**2 + m_target**2 + 2.0 * m_target * math.sqrt(p**2 + m_beam**2))

154class PythiaFactory:
155 """Create initialised Pythia8 instances with the tune, their own seed and stable cascade and signal
156 particles. The seeds are drawn from a generator of their own, so that they do not follow the draws
157 of the cascade; Pythia8 accepts seeds up to 900000000."""
158
159 def __init__(self, seed, tune_settings, stable_ids):
160 self.seeds = random.Random(f"pythia8-{seed}")
161 self.tune_settings = tune_settings
162 self.stable_ids = stable_ids
163 self.n_instances = 0
164
165 def __call__(self, settings):
166 self.n_instances += 1
167 pythia = pythia8.Pythia("", False)
168 for setting in [*self.tune_settings, *settings, "Print:quiet = on", "Random:setSeed = on"]:
169 pythia.readString(setting)
170 pythia.readString(f"Random:seed = {self.seeds.randint(1, 900000000)}")
171 for pdg_id in self.stable_ids:
172 pythia.readString(f"{pdg_id}:mayDecay = off")
173 if not pythia.init():
174 raise RuntimeError(f"Pythia8 initialisation failed for {settings}")
175 return pythia

178def next_event(pythia):
179 """Generate the next event, giving up after 100 failed attempts."""
180 for _ in range(100):
181 if pythia.next():
182 return
183 raise RuntimeError(
184 f"Pythia8 failed to generate an event for beams {pythia.infoPython().idA()} on {pythia.infoPython().idB()}"
185 )

188class SignalEvents:
189 """Signal events in momentum bins p_beam / SIGNAL_BIN_RATIO^i, generated in batches of increasing
190 size, one Pythia8 instance per batch."""
191
192 def __init__(self, new_pythia, process, signal_ids, p_beam):
193 self.new_pythia = new_pythia
194 self.process = process
195 self.signal_ids = signal_ids
196 self.p_beam = p_beam
197 self.buffers = {}
198
199 def next(self, beam_id, nucleon_id, p):
200 """Return process code and signal hadrons (id, px, py, pz, E, m) in the CM frame of an event at
201 momentum p."""
202 p_bin = round(math.log(self.p_beam / p) / math.log(SIGNAL_BIN_RATIO))
203 buffer, n_fill = self.buffers.setdefault((beam_id, nucleon_id, p_bin), ([], 5))
204 if not buffer:
205 n_fill = min(2 * n_fill, 1000)
206 self.buffers[beam_id, nucleon_id, p_bin] = (buffer, n_fill)
207 pythia = self.new_pythia(
208 [self.process, *cm_beam_settings(beam_id, nucleon_id, self.p_beam / SIGNAL_BIN_RATIO**p_bin)]
209 )
210 for _ in range(n_fill):
211 next_event(pythia)
212 event = pythia.event
213 hadrons = [
214 event[i] for i in range(event.size()) if event[i].idAbs() in self.signal_ids and event[i].isFinal()
215 ]
216 buffer.append(
217 (pythia.infoPython().code(), [(h.id(), h.px(), h.py(), h.pz(), h.e(), h.m()) for h in hadrons])
218 )
219 return buffer.pop()

223# about 1 in 3e5 inelastic events contains charm
224MAX_INCLUSIVE_TRIES = 100_000_000

227class InclusiveSignalEvents:
228 """Inclusive inelastic events with signal hadrons, at the exact projectile momentum, one Pythia8 instance
229 per target nucleon. Events are hadronised only if they contain a heavy quark, which saves most of the
230 time spent on events without one."""
231
232 def __init__(self, new_pythia, signal_ids, heavy_quark, p_beam):
233 self.signal_ids = signal_ids
234 self.heavy_quark = heavy_quark
235 self.pythia = [
236 new_pythia(
237 ["SoftQCD:inelastic = on", "HadronLevel:all = off"]
238 + ["Beams:frameType = 3", "Beams:allowVariableEnergy = on", "Beams:allowIDAswitch = on"]
239 + [f"Beams:idB = {nucleon_id}", f"Beams:pzA = {p_beam}", "Beams:pzB = 0."]
240 )
241 for nucleon_id in TARGET_NUCLEONS
242 ]
243 self.n_tried = 0
244
245 def next(self, beam_id, i_nucleon, px, py, pz):
246 """Return process code and signal hadrons (id, px, py, pz, E, m) in the lab frame."""
247 pythia = self.pythia[i_nucleon]
248 pythia.setBeamIDs(beam_id, TARGET_NUCLEONS[i_nucleon])
249 pythia.setKinematics(px, py, pz, 0.0, 0.0, 0.0)
250 for _ in range(MAX_INCLUSIVE_TRIES):
251 next_event(pythia)
252 self.n_tried += 1
253 event = pythia.event
254 if not any(event[i].idAbs() == self.heavy_quark for i in range(event.size())):
255 continue
256 if not pythia.forceHadronLevel():
257 continue
258 event = pythia.event
259 hadrons = [
260 event[i] for i in range(event.size()) if event[i].idAbs() in self.signal_ids and event[i].isFinal()
261 ]
262 if hadrons:
263 return pythia.infoPython().code(), [(h.id(), h.px(), h.py(), h.pz(), h.e(), h.m()) for h in hadrons]
264 raise RuntimeError(
265 f"no charm in {MAX_INCLUSIVE_TRIES} inclusive events of {beam_id} on {TARGET_NUCLEONS[i_nucleon]}"
266 f" at p = ({px:.1f}, {py:.1f}, {pz:.1f}) GeV"
267 )

339def main():
340 args = parse_args()
341 signal = SIGNAL[args.heavy_quark]
342 p_threshold, signal_ids, signal_process = signal.p_threshold, signal.ids, signal.process
343 proton_fraction = PROTON_FRACTION[args.target_composition]
344 if args.pythia8_tune == "FTFT":
345 tune_settings, kfactors = FTFT_SETTINGS, signal.kfactors
346 else:
347 tune_settings, kfactors = [], (1.0, 1.0)
348
349 random.seed(args.seed)
350 new_pythia = PythiaFactory(args.seed, tune_settings, [*CASCADE_IDS, *signal_ids])
351
352 # minimum-bias events with variable beam particle and energy, one instance per target nucleon
353 mbias_pythia = [
354 new_pythia(
355 ["SoftQCD:all = on", "Beams:frameType = 3", "Beams:allowVariableEnergy = on", "Beams:allowIDAswitch = on"]
356 + [f"Beams:idB = {nucleon_id}", f"Beams:pzA = {args.p_beam}", "Beams:pzB = 0."]
357 )
358 for nucleon_id in TARGET_NUCLEONS
359 ]
360
361 # chi = K * sigma(signal) / sigma(total) vs momentum for all beam and target particles.
362 # The grid is logarithmic and chi is interpolated in log(chi) vs log(p): chi varies by orders of
363 # magnitude over the range and most of that variation sits close to the threshold.
364 p_grid = np.geomspace(p_threshold, args.p_beam, args.n_momentum_points)
365 log_p_grid = np.log(p_grid)
366 log_chi, sigma = {}, {}
367 for pdg_id in CASCADE_IDS:
368 for beam_id in sorted({pdg_id, -pdg_id}):
369 if not PDG.GetParticle(beam_id):
370 continue
371 for i_nucleon, nucleon_id in enumerate(TARGET_NUCLEONS):
372 sigma[beam_id, i_nucleon] = []
373 for p in p_grid:
374 pythia = new_pythia(
375 [signal_process, "PartonLevel:all = off", "HadronLevel:all = off"]
376 + cm_beam_settings(beam_id, nucleon_id, p)
377 )
378 for _ in range(args.n_sigma_events):
379 next_event(pythia)
380 sigma[beam_id, i_nucleon].append(
381 signal_kfactor(kfactors, beam_id, nucleon_id, p) * pythia.infoPython().sigmaGen()
382 )
383 sigma_total = [
384 mbias_pythia[i_nucleon].getSigmaTotal(beam_id, nucleon_id, ecm(beam_id, nucleon_id, p))
385 for p in p_grid
386 ]
387 log_chi[beam_id, i_nucleon] = np.log([s / t for s, t in zip(sigma[beam_id, i_nucleon], sigma_total)])
388 print(
389 f"K*sigma and chi at {args.p_beam} GeV for {beam_id} on {nucleon_id}: "
390 f"{sigma[beam_id, i_nucleon][-1]:.3e} mb, {math.exp(log_chi[beam_id, i_nucleon][-1]):.3e}"
391 )
392 chi_max = max(np.exp(c).max() for c in log_chi.values())
393 # cross section per nucleon of the target composition, at the beam energy, for the normalisation
394 sigma_QQ = proton_fraction * sigma[2212, 0][-1] + (1.0 - proton_fraction) * sigma[2212, 1][-1]
395 print(f"K*sigma per nucleon at {args.p_beam} GeV: {1e3 * sigma_QQ:.2f} ub")
396
397 signal_events = SignalEvents(new_pythia, signal_process, signal_ids, args.p_beam)
398 inclusive_events = (
399 InclusiveSignalEvents(new_pythia, signal_ids, args.heavy_quark, args.p_beam)
400 if args.charm_production == "inclusive" and args.heavy_quark == 4
401 else None
402 )
403
404 output_file = ROOT.TFile.Open(args.output, "RECREATE")
405 ntuple = ROOT.TNtuple(
406 "pythia6",
407 "pythia8 heavy flavour",
408 "id:px:py:pz:E:M:mid:mpx:mpy:mpz:mE:mM:k:a0:a1:a2:a3:a4:a5:a6:a7:a8:a9:a10:a11:a12:a13:a14:a15:\
409s0:s1:s2:s3:s4:s5:s6:s7:s8:s9:s10:s11:s12:s13:s14:s15:n_hadrons",
410 )
411 # number of signal particles per cascade depth, used by FixedTargetGenerator for the normalisation
412 depth_hist = ROOT.TH1F("2", "nr signal per cascade depth", 50, 0.5, 50.5)
413 # cross section per nucleon [mb] this file was generated with, read back by run_fixedTarget.py to
414 # scale chicc/chibb, so that the normalisation follows the beam energy, tune and target of the file
415 # hadd adds TParameters by default; keep the value of the first file when cascade files are merged
416 sigma_parameter = ROOT.TParameter("double")("sigma_QQ", sigma_QQ) # type: ignore[missing-attribute]
417 sigma_parameter.SetMergeMode("f")
418 sigma_parameter.Write()
419
420 t0 = time.time()
421 for i_pot in range(args.n_pot):
422 if i_pot % 1000 == 0:
423 print("Generate event ", i_pot)
424 # stack: PID, px, py, pz, cascade depth, ancestors, interaction processes
425 stack: list[tuple[int, float, float, float, int, list[int], list[int]]] = []
426 stack.append((2212, 0.0, 0.0, args.p_beam, 1, [2212] + 99 * [0], 100 * [0]))
427 while stack:
428 beam_id, px, py, pz, depth, ancestors, processes = stack.pop()
429 p = math.sqrt(px**2 + py**2 + pz**2)
430 i_nucleon = 0 if random.random() < proton_fraction else 1
431 nucleon_id = TARGET_NUCLEONS[i_nucleon]
432 if math.exp(np.interp(math.log(p), log_p_grid, log_chi[beam_id, i_nucleon])) / chi_max > random.random():
433 beam_mass = mass(beam_id)
434 beam = ROOT.TLorentzVector(px, py, pz, math.sqrt(p**2 + beam_mass**2))
435 if inclusive_events is not None:
436 code, hadrons = inclusive_events.next(beam_id, i_nucleon, px, py, pz)
437 else:
438 code, hadrons = signal_events.next(beam_id, nucleon_id, p)
439 boost = (beam + ROOT.TLorentzVector(0, 0, 0, mass(nucleon_id))).BoostVector() # type: ignore[missing-attribute]
440 n_processes = min(depth - 1, 15)
441 for hadron_id, *p4, hadron_mass in hadrons:
442 hadron = ROOT.TLorentzVector(*p4)
443 if inclusive_events is None: # forced events are generated in the CM frame
444 hadron.RotateUz(beam.Vect().Unit()) # type: ignore[missing-attribute]
445 hadron.Boost(boost) # type: ignore[missing-attribute]
446 row = [hadron_id, hadron.Px(), hadron.Py(), hadron.Pz(), hadron.E(), hadron_mass]
447 row += [beam_id, px, py, pz, beam.E(), beam_mass, depth]
448 row += ancestors[:16] + processes[:n_processes] + [code] + (15 - n_processes) * [0]
449 # all signal hadrons of the event, read back together by FixedTargetGenerator
450 row += [len(hadrons)]
451 ntuple.Fill(array("f", row))
452 depth_hist.Fill(depth)
453 # minimum-bias event to add new cascade particles to the stack
454 i_nucleon = 0 if random.random() < proton_fraction else 1
455 pythia = mbias_pythia[i_nucleon]
456 pythia.setBeamIDs(beam_id, TARGET_NUCLEONS[i_nucleon])
457 pythia.setKinematics(px, py, pz, 0.0, 0.0, 0.0)
458 next_event(pythia)
459 code = pythia.infoPython().code()
460 if code == 102:
461 # Elastic scattering does not start a new cascade generation: the scattered hadron keeps
462 # its depth. Depth 1 is then everything the beam proton produces up to and including its
463 # first inelastic interaction, which is what chicc/chibb in run_fixedTarget.py normalise.
464 final = [pythia.event[i] for i in range(pythia.event.size()) if pythia.event[i].isFinal()]
465 leading = max(final, key=lambda part: part.pAbs())
466 if leading.pAbs() > p_threshold and len(stack) < 999:
467 stack.append((beam_id, leading.px(), leading.py(), leading.pz(), depth, ancestors, processes))
468 continue
469 new_depth = min(depth + 1, 98)
470 if depth == 1: # first inelastic interaction process of the beam proton
471 processes = [code] + processes[1:]
472 for i in range(pythia.event.size()):
473 part = pythia.event[i]
474 if part.idAbs() in CASCADE_IDS and part.isFinal() and part.pAbs() > p_threshold and len(stack) < 999:
475 new_ancestors = ancestors[: new_depth - 1] + [part.id()] + ancestors[new_depth:]
476 new_processes = processes[: new_depth - 1] + [code] + processes[new_depth:]
477 stack.append((part.id(), part.px(), part.py(), part.pz(), new_depth, new_ancestors, new_processes))
478
479 print(f"Generated {args.n_pot} p.o.t. in {time.time() - t0} s with {new_pythia.n_instances} Pythia8 instances")
480 if inclusive_events is not None:
481 print(f"{inclusive_events.n_tried} inclusive events generated for the charm signal")
482 output_file.Write()
483 output_file.Close()
484 print(f"Output file is {args.output}")

488 main()
