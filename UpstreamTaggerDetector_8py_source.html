32 def __init__(self, name, intree, outtree=None, mcBranchName=None) -> None:
33 # Before creating any output branch, so that a UBT that cannot be
34 # digitized leaves none.
35 self.validate_stored_geometry(ROOT.gGeoManager, global_variables.ShipGeo.UpstreamTagger)
36 super().__init__(name, intree, mcBranchName=mcBranchName, outtree=outtree)
37 # Parallel to the MC link: for each hit, the unrounded ADC signal of
38 # each of its MC points, in the same order as the point indices.
39 self.MCadc = None
40 if mcBranchName:
41 self.MCadc = ROOT.std.vector("std::vector<float>")()
42 self.outtree.Branch(f"{mcBranchName}ADC", self.MCadc)
43
44 # The UBT parameters are the same for every MC point of the run: read
45 # them once here.
46 ubt_geo = global_variables.ShipGeo.UpstreamTagger
47 # Region ID -> constituent tile side. Some ROOT/Python serialization
48 # paths stringify the dictionary keys.
49 self._tile_sizes = {int(region): float(size) for region, size in ubt_geo.RegionTileSize.items()}
50 self._grid_x = (ubt_geo.TileGridOriginX, ubt_geo.TileGridEndX)
51 self._grid_y = (ubt_geo.TileGridOriginY, ubt_geo.TileGridEndY)
52 self._z = ubt_geo.z
53 self._time_res = ubt_geo.TimeResolution
54 self._trigger_threshold = ubt_geo.ADCTriggerThreshold
55
56 # Tile-ID grid: cells of the smallest tile side over the tile grid, and
57 # for each tile side the offset of its ID namespace (see tile_id).
58 self._grid_size = min(self._tile_sizes.values())
59 if self._grid_size <= 0.0:
60 raise ValueError(f"Invalid UBT tile-ID grid size: {self._grid_size}")
61 self._columns = round((self._grid_x[1] - self._grid_x[0]) / self._grid_size)
62 self._rows = round((self._grid_y[1] - self._grid_y[0]) / self._grid_size)
63 self._id_offsets = {}
64 for tile_size in set(self._tile_sizes.values()):
65 multiple = round(tile_size / self._grid_size)
66 if multiple <= 0 or not math.isclose(tile_size, multiple * self._grid_size, rel_tol=0.0, abs_tol=1.0e-9):
67 raise ValueError(
68 f"UBT tile size {tile_size} cm is not an integer multiple of the tile-ID base cell size"
69 f" {self._grid_size} cm"
70 )
71 self._id_offsets[tile_size] = (multiple - 1) * self._columns * self._rows
72
73 self._adc_response_maps = self._load_adc_response_maps(
74 ubt_geo.ADCResponseMapFile, set(self._tile_sizes.values())
75 )

77 @classmethod
78 def validate_stored_geometry(cls, geo_manager, ubt_geo) -> None:
79 """Raise UBTGeometryMismatch if the UBT points of this geometry cannot be digitized.
80
81 The decision is taken from the simulated geometry, not from the
82 configuration, which configure replaces for old geometry files. A
83 geometry without any UBT region volume predates the scintillator UBT:
84 its UBT was a vacuum box, whose detector IDs are not region IDs.
85 Otherwise every region of the UBT map must have its volume, with the
86 same tile size.
87 """
88 if not geo_manager:
89 raise RuntimeError("Load the geometry of the simulation (FAIRGeom) before digitizing the UBT")
90 volumes = {} # region ID -> tile size suffix, e.g. "4cm"
91 for volume in geo_manager.GetListOfVolumes():
92 name = volume.GetName()
93 if name.startswith(cls.REGION_VOLUME_PREFIXREGION_VOLUME_PREFIX):
94 region, _, size = name[len(cls.REGION_VOLUME_PREFIXREGION_VOLUME_PREFIX) :].partition("_")
95 volumes[int(region)] = size
96 if not volumes:
97 raise UBTGeometryMismatch(
98 "the geometry has no UBT region volumes: it predates the scintillator UBT (a vacuum box)"
99 )
100 expected = {int(region): f"{round(float(size))}cm" for region, size in ubt_geo.RegionTileSize.items()}
101 differing = sorted(r for r in expected.keys() | volumes.keys() if expected.get(r) != volumes.get(r))
102 if differing:
103 raise UBTGeometryMismatch(
104 f"{len(differing)} UBT regions of the geometry differ from the UBT map, e.g. {differing[:5]}"
105 )

113 @staticmethod
114 def _load_adc_response_maps(path: str, tile_sizes: set[float]) -> dict[float, ROOT.TH2D]:
115 """Load the ADC/MeV vs local position (mm) TH2D for each tile size (cm)."""
116 # PyROOT raises OSError when the file cannot be opened, also when it is
117 # not a ROOT file, e.g. a Git LFS pointer that was not fetched.
118 try:
119 response_file = ROOT.TFile.Open(path)
120 except OSError as error:
121 raise OSError(
122 f"Cannot open the UBT ADC response map file {path}; it is stored with Git LFS, so run `git lfs pull`"
123 " if it has not been fetched"
124 ) from error
125 maps = {}
126 for tile_size in tile_sizes:
127 name = f"adc_per_mev_{round(10.0 * tile_size)}mm"
128 histogram = response_file.Get(name)
129 if not isinstance(histogram, ROOT.TH2D):
130 raise ValueError(f"UBT ADC response map {path} has no TH2D '{name}'")
131 histogram.SetDirectory(ROOT.gROOT)
132 maps[tile_size] = histogram
133 response_file.Close()
134 return maps

140 @staticmethod
141 def _tile_center(coordinate: float, grid_origin: float, grid_end: float, tile_size: float) -> float:
142 """Return the globally aligned tile center containing coordinate."""
143 coordinate = min(max(coordinate, grid_origin), math.nextafter(grid_end, grid_origin))
144 tile_count = round((grid_end - grid_origin) / tile_size)
145 if tile_count <= 0:
146 raise ValueError(f"Invalid UBT tile grid [{grid_origin}, {grid_end}] with tile size {tile_size}")
147 # At the positive boundary, floating-point subtraction and division
148 # can round nextafter(grid_end, grid_origin) back to an exact quotient
149 # of tile_count. Cap the result so the last valid tile is selected.
150 tile_index = min(math.floor((coordinate - grid_origin) / tile_size), tile_count - 1)
151 return grid_origin + (tile_index + 0.5) * tile_size

153 def digitized_position(self, point, tile_size: float) -> ROOT.TVector3:
154 """Map an MC point to the center of its 2x2 or 4x4 cm2 tile."""
155 x = self._tile_center(point.GetX(), self._grid_x[0], self._grid_x[1], tile_size)
156 y = self._tile_center(point.GetY(), self._grid_y[0], self._grid_y[1], tile_size)
157 return ROOT.TVector3(x, y, self._z)

159 def tile_id(self, position: ROOT.TVector3, tile_size: float) -> int:
160 """Return a stable unique ID for a constituent tile.
161
162 The tile is identified by its lower-left cell on the tile-ID grid, the
163 grid of the smallest tile side (2 cm) over the tile grid bounds, with
164 ``columns`` x ``rows`` cells:
165
166 tile ID = (tile side / grid size - 1) * columns * rows + row * columns + column
167
168 So each tile side has its own range of ``columns * rows`` IDs, and a
169 2 cm and a 4 cm tile with the same lower-left cell get different IDs.
170 For the default geometry (340 x 504 cm2, 170 x 252 cells), the range
171 of 2 cm tiles is [0, 42840) and that of 4 cm tiles [42840, 85680). The
172 stride follows the UBT box, so decoding an ID needs the grid of its
173 geometry:
174
175 side_index, cell = divmod(tile_id, columns * rows)
176 row, column = divmod(cell, columns)
177 """
178 column = round((position.X() - tile_size / 2.0 - self._grid_x[0]) / self._grid_size)
179 row = round((position.Y() - tile_size / 2.0 - self._grid_y[0]) / self._grid_size)
180 if not (0 <= column < self._columns and 0 <= row < self._rows):
181 raise ValueError(
182 f"UBT tile at ({position.X()}, {position.Y()}) cm with size {tile_size} cm "
183 f"maps to cell ({column}, {row}), outside the {self._columns} x {self._rows} tile-ID grid; "
184 f"x bounds={self._grid_x} cm, y bounds={self._grid_y} cm, base cell size={self._grid_size} cm"
185 )
186 return self._id_offsets[tile_size] + row * self._columns + column

188 def adc_signal(self, point, tile_center: ROOT.TVector3, tile_size: float) -> float:
189 """Return the (unrounded) ADC signal of a point from its energy and position in the tile."""
190 # FairShip positions are in cm and the response-map coordinates are mm.
191 local_x_mm = 10.0 * (point.GetX() - tile_center.X())
192 local_y_mm = 10.0 * (point.GetY() - tile_center.Y())
193 response_map = self._adc_response_maps[tile_size]
194 # Clamp to the outermost bins so hits on a tile edge do not land in
195 # the under/overflow bins.
196 x_axis = response_map.GetXaxis()
197 y_axis = response_map.GetYaxis()
198 x_bin = min(max(x_axis.FindFixBin(local_x_mm), 1), x_axis.GetNbins())
199 y_bin = min(max(y_axis.FindFixBin(local_y_mm), 1), y_axis.GetNbins())
200 adc_per_mev = response_map.GetBinContent(x_bin, y_bin)
201 deposited_energy_mev = 1000.0 * point.GetEnergyLoss()
202 return deposited_energy_mev * adc_per_mev

204 def digitize(self) -> None:
205 """Digitize UBT MC points into one hit per fired constituent tile.
206
207 An MC point covers a (track, region) pair and a region holds 25 or 100
208 tiles, so a tile can collect points of several tracks, e.g. a muon and
209 a delta ray. Their ADC signals, each taken at its own position in the
210 tile, are summed before rounding and before the trigger threshold is
211 applied. The hit time is that of the first point in the tile.
212 """
213 tiles: dict[int, _FiredTile] = {} # by tile ID, in the order of their first point
214 for index, aMCPoint in enumerate(self.intree.UpstreamTaggerPoint):
215 tile_size = self._tile_sizes[aMCPoint.GetDetectorID()]
216 position = self.digitized_position(aMCPoint, tile_size)
217 tile_id = self.tile_id(position, tile_size)
218 tile = tiles.get(tile_id)
219 if tile is None:
220 tile = tiles[tile_id] = _FiredTile(aMCPoint, position)
221 elif aMCPoint.GetTime() < tile.first.GetTime():
222 tile.first = aMCPoint
223 tile.points.append(index)
224 tile.signals.append(self.adc_signal(aMCPoint, position, tile_size))
225
226 for tile_id, tile in tiles.items():
227 adc_counts = max(0, round(sum(tile.signals)))
228 aHit = ROOT.UpstreamTaggerHit(tile.first, self.intree.t0, tile.position, self._time_res)
229 aHit.SetTileID(tile_id)
230 aHit.SetADC(adc_counts)
231 aHit.SetTriggered(adc_counts >= self._trigger_threshold)
232 self.det.push_back(aHit)
233 if self.MCdet is not None:
234 self.MCdet.push_back(ROOT.std.vector("int")(tile.points))
235 if self.MCadc is not None:
236 self.MCadc.push_back(ROOT.std.vector("float")(tile.signals))
