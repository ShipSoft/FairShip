175// Test vector operations
176template <typename T>
177bool test_vector_operations(const char* class_name) {
178 std::cout << "Testing " << class_name << "... ";
179
180 // Test 1: Can we create a vector?
181 if (!test_copyable<T>()) {
182 std::cout << "FAIL: Not copyable (private copy constructor?)" << std::endl;
183 return false;
184 }
185
186 try {
187 // Test 2: Create default objects
188 std::vector<T> vec;
189 vec.reserve(10);
190
191 // Test 3: Emplace back (uses default constructor)
192 vec.emplace_back();
193
194 // Test 4: Push back (uses copy constructor)
195 T obj;
196 vec.push_back(obj);
197
198 // Test 5: Access elements with const reference
199 const T& const_ref = vec[0];
200 test_const_access(const_ref);
201
202 // Test 6: Vector resize (uses copy constructor)
203 vec.resize(5);
204
205 std::cout << "PASS" << std::endl;
206 return true;
207 } catch (const std::exception& e) {
208 std::cout << "FAIL: " << e.what() << std::endl;
209 return false;
210 }
211}
