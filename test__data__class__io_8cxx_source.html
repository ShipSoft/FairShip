190// Test vector operations
191template <typename T>
192bool test_vector_operations(const char* class_name) {
193 std::cout << "Testing " << class_name << "... ";
194
195 // Test 1: Can we create a vector?
196 if (!test_copyable<T>()) {
197 std::cout << "FAIL: Not copyable (private copy constructor?)" << std::endl;
198 return false;
199 }
200
201 try {
202 // Test 2: Create default objects
203 std::vector<T> vec;
204 vec.reserve(10);
205
206 // Test 3: Emplace back (uses default constructor)
207 vec.emplace_back();
208
209 // Test 4: Push back (uses copy constructor)
210 T obj;
211 vec.push_back(obj);
212
213 // Test 5: Access elements with const reference
214 const T& const_ref = vec[0];
215 test_const_access(const_ref);
216
217 // Test 6: Vector resize (uses copy constructor)
218 vec.resize(5);
219
220 std::cout << "PASS" << std::endl;
221 return true;
222 } catch (const std::exception& e) {
223 std::cout << "FAIL: " << e.what() << std::endl;
224 return false;
225 }
226}

228// Write RecoTracks to a TTree, as the reconstruction does, and read them back:
229// one with every member set, including the extrapolations and the track
230// length, and one with none of those, which must keep reading as missing
231bool test_recotrack_tree_roundtrip() {
232 std::cout << "Testing SHiP::RecoTrack TTree round trip... ";
233 bool ok = true;
234 const auto fail = [&ok](const char* what) {
235 if (ok) {
236 std::cout << "FAIL" << std::endl;
237 }
238 std::cout << " " << what << std::endl;
239 ok = false;
240 };
241 const auto same = [](const std::vector<Double_t>& a,
242 const std::vector<Double_t>& b) { return a == b; };
243
244 std::array<Double_t, 21> covariance{};
245 for (std::size_t i = 0; i < covariance.size(); ++i) {
246 covariance[i] = 0.01 * static_cast<Double_t>(i + 1);
247 }
248 std::vector<Double_t> ubtCov(21), tdCov(21);
249 for (std::size_t i = 0; i < 21; ++i) {
250 ubtCov[i] = 0.1 * static_cast<Double_t>(i + 1);
251 tdCov[i] = -0.2 * static_cast<Double_t>(i + 1);
252 }
253 const std::vector<Double_t> xFits{1.1, 1.2, 1.3};
254 const std::vector<Double_t> yFits{-2.1, -2.2, -2.3};
255 const std::vector<Double_t> zFits{8400., 8402., 8410.};
256 const std::vector<Double_t> residuals{0.01, -0.02, 0.03};
257 const std::vector<Double_t> pulls{0.5, -1.0, 1.5};
258
259 SHiP::RecoTrack full(TVector3(1., 2., 8400.), TVector3(0.1, -0.2, 10.), -1,
260 covariance, 12.5, 7., 3, xFits, yFits, zFits, residuals,
261 pulls);
262 // The UBT is upstream of the first hit: a negative path length
263 full.SetUBTExtrapolation(TVector3(0.5, 1.5, 3287.), TVector3(0.1, -0.2, 10.),
264 ubtCov, -5113.4);
265 full.SetTDExtrapolation(TVector3(3., -4., 9614.), TVector3(0.2, -0.3, 9.9),
266 tdCov, 1204.6);
267 full.SetTrackLength(2405.7);
268 const SHiP::RecoTrack bare(TVector3(), TVector3(0., 0., 1.), 1, covariance,
269 1., 1., 2, {}, {}, {}, {}, {});
270
271 const char* filename = "test_RecoTrack_roundtrip.root";
272 {
273 TFile file(filename, "RECREATE");
274 TTree tree("tracks", "RecoTrack round trip");
275 std::vector<SHiP::RecoTrack> tracks{full, bare};
276 tree.Branch("RecoTracks", &tracks, 32000, -1);
277 tree.Fill();
278 file.Write();
279 }
280
281 std::vector<SHiP::RecoTrack>* read = nullptr;
282 TFile file(filename, "READ");
283 auto* tree = file.Get<TTree>("tracks");
284 if (tree == nullptr) {
285 fail("tree not found");
286 } else {
287 tree->SetBranchAddress("RecoTracks", &read);
288 tree->GetEntry(0);
289 }
290 if (ok && (read == nullptr || read->size() != 2)) {
291 fail("two tracks read back");
292 }
293 if (ok) {
294 const SHiP::RecoTrack& t = (*read)[0];
295 if (t.x() != 1. || t.y() != 2. || t.z() != 8400. || t.px() != 0.1 ||
296 t.py() != -0.2 || t.pz() != 10. || t.charge() != -1) {
297 fail("state and charge");
298 }
299 if (t.chi2() != 12.5 || t.nDoF() != 7. || t.nMeasurements() != 3) {
300 fail("fit quality");
301 }
302 if (!same(t.GetCovarianceElements(),
303 std::vector<Double_t>(covariance.begin(), covariance.end()))) {
304 fail("track covariance");
305 }
306 if (!same(t.GetXFittedPoints(), xFits) ||
307 !same(t.GetYFittedPoints(), yFits) ||
308 !same(t.GetZFittedPoints(), zFits)) {
309 fail("fitted points");
310 }
311 if (!same(t.GetResiduals(), residuals) || !same(t.GetPulls(), pulls)) {
312 fail("residuals and pulls");
313 }
314 if (!t.HasUBTExtrapolation() ||
315 t.GetUBTPosition() != TVector3(0.5, 1.5, 3287.) ||
316 t.GetUBTMomentum() != TVector3(0.1, -0.2, 10.) ||
317 !same(t.GetUBTCovariance(), ubtCov) ||
318 t.GetUBTPathLength() != -5113.4) {
319 fail("UBT extrapolation (and the sign of its path length)");
320 }
321 if (!t.HasTDExtrapolation() ||
322 t.GetTDPosition() != TVector3(3., -4., 9614.) ||
323 t.GetTDMomentum() != TVector3(0.2, -0.3, 9.9) ||
324 !same(t.GetTDCovariance(), tdCov) || t.GetTDPathLength() != 1204.6) {
325 fail("timing-detector extrapolation");
326 }
327 if (t.GetTrackLength() != 2405.7) {
328 fail("track length");
329 }
330
331 const SHiP::RecoTrack& b = (*read)[1];
332 if (b.HasUBTExtrapolation() || b.HasTDExtrapolation()) {
333 fail("unset extrapolations read back as stored");
334 }
335 if (!std::isnan(b.GetUBTPosition().Z()) ||
336 !std::isnan(b.GetUBTPathLength()) ||
337 !std::isnan(b.GetTDMomentum().Z()) ||
338 !std::isnan(b.GetTDPathLength()) || !std::isnan(b.GetTrackLength())) {
339 fail("unset extrapolations and track length read back as not NaN");
340 }
341 if (!b.GetUBTCovariance().empty() || !b.GetTDCovariance().empty()) {
342 fail("unset extrapolation covariances read back as not empty");
343 }
344 if (!b.GetXFittedPoints().empty() || !b.GetResiduals().empty()) {
345 fail("empty fitted points read back as not empty");
346 }
347 }
348 file.Close();
349 std::remove(filename);
350
351 // An extrapolation stored without a covariance keeps it empty
352 SHiP::RecoTrack noCov = bare;
353 noCov.SetUBTExtrapolation(TVector3(), TVector3(0., 0., 1.), {}, -1.);
354 if (!noCov.HasUBTExtrapolation() || !noCov.GetUBTCovariance().empty()) {
355 fail("extrapolation without covariance");
356 }
357
358 if (ok) {
359 std::cout << "PASS" << std::endl;
360 }
361 return ok;
362}
