27Bool_t CaloScoringPlane::ProcessHits(FairVolume* /*vol*/) {
28 // Score each track once, on its first step inside the plane (this runs
29 // after that step, so the values below are those at its end: the exit face
30 // for a particle crossing the plane). Tracks created inside the (vacuum)
31 // plane are not scored.
32 if (!gMC->IsTrackEntering()) {
33 return kFALSE;
34 }
35
36 fEventID = gMC->CurrentEvent();
37 fTrackID = gMC->GetStack()->GetCurrentTrackNumber();
38 Int_t copyNo = 0;
39 gMC->CurrentVolID(copyNo);
40 fVolumeID = copyNo;
41
42 fTime = gMC->TrackTime() * 1.0e09; // ns
43 fLength = gMC->TrackLength();
44 gMC->TrackPosition(fPos);
45 gMC->TrackMomentum(fMom);
46
47 TParticle* p = gMC->GetStack()->GetCurrentTrack();
48 const Int_t pdgCode = p->GetPdgCode();
49
50 AddHit(fEventID, fTrackID, fVolumeID, TVector3(fPos.X(), fPos.Y(), fPos.Z()),
51 TVector3(fMom.Px(), fMom.Py(), fMom.Pz()), fTime, fLength, 0.,
52 pdgCode);
53
54 // Count a point for this track. Note that ShipStack only keeps a secondary
55 // because of its points when it is configured with a minimum number of
56 // points (MCTracksWithHitsOnly / MCTracksWithHitsOrEnergyCut in
57 // run_simScript.py). With the default MCTracksWithEnergyCutOnly, secondaries
58 // below the energy cut are dropped and their points get track ID -2.
59 ShipStack* stack = dynamic_cast<ShipStack*>(gMC->GetStack());
60 stack->AddPoint(kCaloScoringPlane);
61
62 return kTRUE;
63}

65void CaloScoringPlane::ConstructGeometry() {
66 TGeoVolume* top = gGeoManager->GetTopVolume();
67
68 ShipGeo::InitMedium("vacuum");
69 TGeoMedium* vacuum = gGeoManager->GetMedium("vacuum");
70 if (!vacuum) {
71 Fatal("ConstructGeometry", "Medium 'vacuum' not found.");
72 }
73
74 fDetector = gGeoManager->MakeBox("CaloScoringPlane", vacuum, fXSize / 2.,
75 fYSize / 2., fZSize / 2.);
76 fDetector->SetLineColor(kOrange);
77 AddSensitiveVolume(fDetector);
78
79 // Node name in the top volume is "CaloScoringPlane_1".
80 top->AddNode(fDetector, 1, new TGeoTranslation(0., 0., fZPos));
81 LOG(info) << "CaloScoringPlane placed at z = " << fZPos << " cm, size "
82 << fXSize << " x " << fYSize << " x " << fZSize << " cm";
83}
