4"""Check that every shipped field map is stored the way ShipBFieldMap reads it.
5
6The reader streams the ``Data`` tree sequentially and indexes it as
7``(iX*Ny + iY)*Nz + iZ``. Nothing in the file states that ordering, and when
8``Nx == Ny`` with matching x and y ranges a transposed map still has the right
9number of entries, so it is read mirrored about the x = y plane without a
10single complaint. That is what happened to the default spectrometer map.
11
12Two things pin the ordering down. The stored node coordinates say what the
13writer intended, and the divergence of the field says what is actually in the
14file: a physical field has div B = 0, and no transposition survives that.
15"""
16
17from pathlib import Path

23MAP_DIRECTORY = Path(__file__).resolve().parent.parent / "files"

25# Per map: the component that dominates on the beam axis, how much better the
26# divergence has to be than the transposed reading, and how large the
27# divergence may be next to the local gradient scale.
28#
29# The shield maps sit inside iron on a coarse grid, so their numerical
30# divergence is large and only weakly discriminating; their node coordinates
31# carry the check instead. The measured values are roughly 50x / 0.015 for the
32# spectrometer maps, 3.7x / 0.29 for the older one, and 2.2x / 0.9 for the
33# shield maps.
34EXPECTED = {
35 "2026_09_28_SHiP_SpectrometerField_ECN3_MgB2": ("Bx", 10.0, 0.1),
36 "2026_09_28_MainSpectrometerField_V13_3500": ("Bx", 10.0, 0.1),
37 "2026_09_28_MainSpectrometerField_V21_2000": ("Bx", 10.0, 0.1),
38 "2026_09_28_MainSpectrometerField_V21_3000": ("Bx", 10.0, 0.1),
39 "2026_07_02_MainSpectrometerField_V21_2455": ("Bx", 10.0, 0.1),
40 "MainSpectrometerField": ("Bx", 2.0, 0.5),
41 "TRY_2025": ("By", 1.4, 1.2),
42 "TRY_2026": ("By", 1.4, 1.2),
43}

46def _is_lfs_pointer(path):
47 with path.open("rb") as handle:
48 return handle.read(40).startswith(b"version https://git-lfs")

51def _discover():
52 """Every readable field map in files/.
53
54 Unfetched git-lfs pointers are dropped one by one rather than taken as a
55 reason to skip the lot, so a partly fetched checkout still checks the maps
56 it does have.
57 """
58 if not MAP_DIRECTORY.is_dir():
59 return []
60 candidates = sorted(MAP_DIRECTORY.glob("*.root"))
61 return [path for path in candidates if not _is_lfs_pointer(path) and fmt.is_field_map(path)]
62
63
64MAPS = _discover()

66pytestmark = pytest.mark.skipif(not MAPS, reason="no field maps in files/ (git-lfs objects not fetched?)")

75def expectation(stem):
76 """What this map should look like, or a skip if nobody has said."""
77 if stem not in EXPECTED:
78 pytest.skip(f"{stem} is not listed in EXPECTED (see test_map_is_expected)")
79 return EXPECTED[stem]

82@pytest.mark.parametrize("path", MAPS, ids=lambda path: path.stem)
83def test_map_is_expected(path):
84 """A new map has to be added to EXPECTED deliberately, not by accident."""
85 assert path.stem in EXPECTED, (

91@pytest.mark.parametrize("path", MAPS, ids=lambda path: path.stem)
92def test_binning_is_consistent(path):
93 rng = fmt.read_range(path)
94 assert min(rng.dx, rng.dy, rng.dz) > 0.0
95 assert min(rng.Nx, rng.Ny, rng.Nz) >= 2
96 # Limits that are not a whole number of bins apart would put every node
97 # away from where the interpolation looks for it.
98 assert max(rng.extent_mismatch()) < 1e-2
99 assert fmt.entry_count(path) == rng.N

102@pytest.mark.parametrize("path", MAPS, ids=lambda path: path.stem)
103def test_coordinates_follow_the_assumed_order(path):
104 """The regression test: a transposed map is off by tens of bins here."""
105 rng = fmt.read_range(path)
106 residuals = fmt.coord_residuals(path, rng)
107 assert residuals is not None, (
108 f"{path.name} stores no x,y,z branches, so its ordering cannot be "
109 "verified. Repack it with field/canonicaliseFieldMap.py."
110 )
111 # The builtin max() skips a NaN that is not first, so check explicitly.
112 assert np.all(np.isfinite(residuals)), f"{path.name} stores non-finite node coordinates: residuals {residuals}"
113 assert max(residuals) < 0.5, (
114 f"{path.name} is stored in {fmt.infer_data_order(path, rng)} order, not "
115 "the ascending z,y,x order ShipBFieldMap assumes."
116 )

119@pytest.mark.parametrize("path", MAPS, ids=lambda path: path.stem)
120def test_divergence_prefers_the_stored_order(path, loaded):
121 """div B = 0 only holds for the reading that matches the file."""
122 _, minimum_ratio, maximum_relative = expectation(path.stem)
123 rng, field = loaded[path.stem]
124 as_stored, scale = fmt.div_b_rms(field, rng)
125 transposed, _ = fmt.div_b_rms(fmt.transpose_xy(field), rng)
126 assert as_stored * minimum_ratio < transposed
127 assert as_stored < maximum_relative * scale

130@pytest.mark.parametrize("path", MAPS, ids=lambda path: path.stem)
131def test_expected_component_dominates_on_axis(path, loaded):
132 """Catches a Bx/By column swap, which leaves div B untouched."""
133 dominant = expectation(path.stem)[0]
134 rng, field = loaded[path.stem]
135 peaks = np.abs(fmt.on_axis(field, rng)).max(axis=0)
136 index = fmt.FIELD_BRANCHES.index(dominant)
137 others = np.delete(peaks, index).max()
138 assert peaks[index] > 5.0 * others
