91 // Hand the buffered clones of a splitting decay to the stack popper. They
92 // can only be popped during a step, so the first track after the decay is
93 // designated as their carrier and must not be stopped before it steps.
94 // The designation is shared by all instances: run_fixedTarget can add
95 // several sensitive planes and whichever is polled first would otherwise
96 // stop the carrier before the splitting instance ever sees it.
97 static constexpr Int_t kNoCarrier = -1;
98 static constexpr Int_t kCarrierRequested = -2;
99 static Int_t fgCarrierTrackID;
100
101 // Ask for the next track to carry the clones just added to the buffer.
102 static void RequestCloneCarrier() { fgCarrierTrackID = kCarrierRequested; }
103
104 // Tracks that an unsplit run would not have. A decay that the clones
105 // re-sample is replaced as a whole: its products, and everything they go on
106 // to make, would otherwise be counted on top of the same channels from the
107 // clones. A carrier that an unsplit run would have stopped only steps so
108 // that the clones get popped, and is replaced with all its descendants too.
109 // Shared by all instances for the same reason as the carrier: a sibling
110 // plane must not score them either.
111 static std::set<Int_t> fgReplacedDecays;
113 static std::set<Int_t> fgReplacedTracks;

117 // Weight of a per-step split track along its path. Geant4 hands a track's
118 // secondaries to the stack only once the track is done, and ShipStack then
119 // scales them by its final weight, although one made mid-track should carry
120 // the weight the track had when it made it. PreTrack() puts that right from
121 // this history. Shared by all instances, because the first plane to run
122 // PreTrack() for a secondary has to see it.
123 struct WeightHistory {
124 Double_t initial = 1.;
127 std::vector<std::pair<Double_t, Double_t>> splits;
131 Double_t WeightAt(Double_t t) const;
132 };
133 static std::map<Int_t, WeightHistory> fgWeightHistory;
134 static std::set<Int_t> fgWeightCorrected;

152 // Split set of a track that is being split, created on first use. The
153 // track is recorded on the stack as the survivor of its set; its clones
154 // and continuations join the same set. A continuation maps to the set of
155 // the track it continues, so a per-step split track keeps one set along its
156 // whole path. The set's weight is the track's weight when the set is
157 // created, before any split lowers it.
158 Int_t SplitSetOf(Int_t trackId, Double_t weight);

172 int32_t fNsplits;
173 // intermediate splits to use at each step before the particle decays
174 int32_t fIntermediateNsplits;
175 // Upper bound on the clone buffer, in both splitting modes. Per-step
176 // splitting is the pathological case: the buffer grows by
177 // fIntermediateNsplits on every qualifying step and is only drained in
178 // PreTrack(), so a bad split count could make it grow without bound within a
179 // single track. ~25k TrackBuffer records is about 2.5 MB, far above what any
180 // sane configuration reaches. Set via --max-split-buffer.
181 // This is a hard bound: SplitStep() stops accepting per-step clones early
182 // enough to leave room for the fNsplits endpoint clones PostTrack() appends,
183 // and Initialize() rejects a cap that fNsplits alone would exceed.
184 std::size_t fMaxSplitBuffer = 25'000;
185 // Upper bound on the projected event size: the stack size plus
186 // kShowerSafetyFactor entries for every pending clone. With the defaults this
187 // is the cap that stops per-step splitting, at about 10k pending clones.
188 // Set via --max-event-size.
189 std::size_t fMaxEventSize = 5'000'000;
190 // latches so each cap is reported at most once per event
191 Bool_t fSplitBufferLimitWarned = kFALSE;
192 Bool_t fEventSizeLimitWarned = kFALSE;
193 Bool_t fSplitOnce =
194 kTRUE; // determine if we want to split once (when the particle decays)
195 // or at every step (taking decay probabilities into account)
196
197 std::vector<TrackBuffer> fSecondaryBuffer;
198 std::set<Int_t> fCloneTracks;
199 // The last per-step split, so that PostTrack() can tell whether the step on
200 // which a track decayed was split.
201 Int_t fLastSplitTrackID = -1;
202 Int_t fLastSplitStep = -1;
203 // Split set per stack index of a split track or continuation, and the next

213 Int_t fLostCloneTracks = 0;
214 Double_t fLostCloneWeight = 0.;
215
216 TFile* fout;
217 TClonesArray* fElectrons;
218 Int_t index;
219};
