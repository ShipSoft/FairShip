27void RecoVertex::AddTrack(Int_t trackId, const TVector3& momentum,
28 const std::array<Double_t, 6>& momentumCovariance) {
29 m_trackIds.push_back(trackId);
30 m_trackPx.push_back(momentum.X());
31 m_trackPy.push_back(momentum.Y());
32 m_trackPz.push_back(momentum.Z());
33 m_trackCovP.insert(m_trackCovP.end(), momentumCovariance.begin(),
34 momentumCovariance.end());
35}

37std::array<Double_t, 6> RecoVertex::GetTrackMomentumCovariance(size_t i) const {
38 if (i >= m_trackIds.size()) {
39 throw std::out_of_range("RecoVertex: track index out of range");
40 }
41 std::array<Double_t, 6> covariance{};
42 std::copy_n(m_trackCovP.begin() + 6 * i, 6, covariance.begin());
43 return covariance;
44}
