118template <int N, std::size_t M = N * (N + 1) / 2>
119std::array<Double_t, M> upperTriangle(const Acts::SquareMatrix<N>& matrix) {
120 std::array<Double_t, M> elements{};
121 std::size_t k = 0;
122 for (int row = 0; row < N; ++row) {
123 for (int col = row; col < N; ++col) {
124 elements[k++] = matrix(row, col);
125 }
126 }
127 return elements;
128}

131Acts::Matrix<3, 3> momentumJacobianDirection(double qOverP) {
132 return Acts::SquareMatrix3::Identity() / std::abs(qOverP);
133}

135Acts::Vector3 momentumJacobianQOverP(const Acts::Vector3& direction,
136 double qOverP) {
137 return -direction / (qOverP * std::abs(qOverP));
138}

141Acts::SquareMatrix<6> shipCovariance(const Acts::BoundTrackParameters& params,
142 const Acts::GeometryContext& gctx) {
143 const auto& surface = params.referenceSurface();
144 const Acts::FreeVector free =
145 Acts::transformBoundToFreeParameters(surface, gctx, params.parameters());
146 const Acts::Vector3 direction = free.segment<3>(Acts::eFreeDir0);
147 const Acts::BoundToFreeMatrix boundToFree = surface.boundToFreeJacobian(
148 gctx, free.segment<3>(Acts::eFreePos0), direction);
149 const Acts::FreeMatrix freeCovariance =
150 boundToFree * params.covariance().value() * boundToFree.transpose();
151
152 const double qOverP = free[Acts::eFreeQOverP];
153 const Acts::RotationMatrix3 rotation = ActsFrame::toShipRotation();
154 Acts::Matrix<6, Acts::eFreeSize> freeToShip =
155 Acts::Matrix<6, Acts::eFreeSize>::Zero();
156 freeToShip.block<3, 3>(0, Acts::eFreePos0) =
157 rotation / ActsFrame::kShipLength;
158 freeToShip.block<3, 3>(3, Acts::eFreeDir0) =
159 rotation * momentumJacobianDirection(qOverP) / u::GeV;
160 freeToShip.block<3, 1>(3, Acts::eFreeQOverP) =
161 rotation * momentumJacobianQOverP(direction, qOverP) / u::GeV;
162 return freeToShip * freeCovariance * freeToShip.transpose();
163}

166Acts::SquareMatrix3 shipMomentumCovariance(
167 const Acts::BoundTrackParameters& params) {
168 const double phi = params.parameters()[Acts::eBoundPhi];
169 const double theta = params.parameters()[Acts::eBoundTheta];
170 const double qOverP = params.parameters()[Acts::eBoundQOverP];
171 const double p = 1. / std::abs(qOverP);
172 const Acts::Vector3 direction = params.direction();
173 Acts::Matrix<3, Acts::eBoundSize> boundToMomentum =
174 Acts::Matrix<3, Acts::eBoundSize>::Zero();
175 boundToMomentum.col(Acts::eBoundPhi) =
176 Acts::Vector3(-std::sin(phi) * std::sin(theta),
177 std::cos(phi) * std::sin(theta), 0.) *
178 p;
179 boundToMomentum.col(Acts::eBoundTheta) =
180 Acts::Vector3(std::cos(phi) * std::cos(theta),
181 std::sin(phi) * std::cos(theta), -std::sin(theta)) *
182 p;
183 boundToMomentum.col(Acts::eBoundQOverP) =
184 momentumJacobianQOverP(direction, qOverP);
185 const Acts::Matrix<3, Acts::eBoundSize> jacobian =
186 ActsFrame::toShipRotation() * boundToMomentum / u::GeV;
187 return jacobian * params.covariance().value() * jacobian.transpose();
188}

217Acts::Vector3 closestPoint(const std::vector<Tangent>& lines) {
218 Acts::SquareMatrix3 a = Acts::SquareMatrix3::Zero();
219 Acts::Vector3 b = Acts::Vector3::Zero();
220 for (const auto& [position, direction] : lines) {
221 const Acts::SquareMatrix3 projection =
222 Acts::SquareMatrix3::Identity() - direction * direction.transpose();
223 a += projection;
224 b += projection * position;
225 }
226 return a.colPivHouseholderQr().solve(b);
227}

231Acts::Vector3 vertexSeed(const std::vector<Acts::BoundTrackParameters>& tracks,
232 const Acts::GeometryContext& gctx) {
233 Acts::Vector3 seed = closestPoint(tangents(tracks, gctx));
234 if (!seed.allFinite()) {
235 seed = Acts::Vector3(0.5 * (kVertexSeedMinX + kVertexSeedMaxX), 0., 0.);
236 }
237 seed.x() = std::clamp(seed.x(), kVertexSeedMinX, kVertexSeedMaxX);
238 return seed;
239}

252double lineDistance(const Acts::Vector3& p1, const Acts::Vector3& d1,
253 const Acts::Vector3& p2, const Acts::Vector3& d2) {
254 const Acts::Vector3 delta = p1 - p2;
255 const Acts::Vector3 normal = d1.cross(d2);
256 if (normal.norm() > 1e-12) {
257 return std::abs(delta.dot(normal)) / normal.norm();
258 }
259 return delta.cross(d1.normalized()).norm();
260}

266 : straws(geometry),
267 shipField(magField),
268 field(std::make_shared<ActsShipField>(magField)),
269 fitter(DirectPropagator(Stepper(field), Acts::DirectNavigator(),
270 Acts::getDefaultLogger("Propagator",
271 Acts::Logging::WARNING)),
272 Acts::getDefaultLogger("KalmanFitter", Acts::Logging::WARNING)),
273 extrapolator(
274 Stepper(field), Acts::VoidNavigator(),
275 Acts::getDefaultLogger("Extrapolator", Acts::Logging::WARNING)),
276 tracks(std::make_shared<Acts::VectorTrackContainer>(),
277 std::make_shared<Trajectory>()) {
278 extensions.surfaceAccessor.connect<&Impl::surfaceOf>(this);
279 extensions.calibrator.connect<&Impl::calibrate>(this);

282 extensions.smoother.connect<&Acts::MbfSmoother::operator()<Trajectory>>(

285 // The magnet sits between stations 2 and 3
286 for (const auto& [id, wire] : straws.wires()) {
287 if (wire.station == 2) {
288 magnetStart = std::max(magnetStart, wire.z);
289 } else if (wire.station == 3) {
290 magnetEnd = std::min(magnetEnd, wire.z);
291 }
292 }
293 LineSegment axis;
294 const auto [integral, centre] = fieldIntegral(axis);
295 patRec.emplace(StrawPatRec::Config{}, centre, integral);

313 std::pair<double, double> fieldIntegral(const LineSegment& line) const {
314 double integral = 0.;
315 double moment = 0.;
316 for (double z = magnetStart; z < magnetEnd; z += kFieldStep) {
317 const double zMid = z + kFieldStep / 2.;
318 const double x[3] = {line.x(zMid), line.y(zMid), zMid};
319 double b[3] = {0., 0., 0.};
320 shipField->Field(x, b);
321 // kGauss to T, cm to m
322 const double bx = 0.1 * b[0] * 0.01 * kFieldStep;
323 integral += bx;
324 moment += bx * zMid;
325 }
326 return {integral, integral != 0. ? moment / integral
327 : 0.5 * (magnetStart + magnetEnd)};
328 }

334 void calibrate(const Acts::GeometryContext& /*gctx*/,
335 const Acts::CalibrationContext& /*cctx*/,
336 const Acts::SourceLink& sourceLink,
337 Trajectory::TrackStateProxy state) const {
338 const auto& measurement = measurements.at(sourceLink.get<std::size_t>());
339 state.allocateCalibrated(1);
340 state.template calibrated<1>()(0) =
341 measurement.sign * measurement.driftRadius;
342 state.template calibratedCovariance<1>()(0, 0) = measurement.variance;
343 state.setProjectorSubspaceIndices(std::array{Acts::eBoundLoc0});
344 state.setUncalibratedSourceLink(Acts::SourceLink(sourceLink));
345 }

370 std::optional<std::pair<Acts::BoundTrackParameters, double>>
371 extrapolateWithLength(const Extrapolator& propagator,
372 const Acts::BoundTrackParameters& start,
373 const Acts::Surface& target) const {
374 // Try first the direction in which the straight line along the momentum
375 // meets the target. A propagation the wrong way only stops at the path
376 // limit, as the void navigator has no world boundary; the other
377 // direction stays as the fallback in case the field changes the answer.
378 const auto intersection =
379 target.intersect(gctx, start.position(gctx), start.direction())
380 .closest();
381 const bool behind =
382 intersection.isValid() && intersection.pathLength() < 0.;
383 const auto first =
384 behind ? Acts::Direction::Backward() : Acts::Direction::Forward();
385 const auto second =
386 behind ? Acts::Direction::Forward() : Acts::Direction::Backward();
387 for (const auto direction : {first, second}) {
388 Extrapolator::Options<> options(gctx, mctx);
389 options.pathLimit = kPathLimit;
390 options.direction = direction;
391 auto result = propagator.propagate(start, target, options);
392 if (result.ok() && result.value().endParameters.has_value()) {
393 // Take the sign from the direction rather than relying on the
394 // sign convention of the accumulated path length
395 const double sign = direction == Acts::Direction::Forward() ? 1. : -1.;
396 const double length = sign * std::abs(result.value().pathLength);
397 return std::make_pair(result.value().endParameters.value(), length);
398 }
399 }
400 return std::nullopt;
401 }
402
404 std::optional<Acts::BoundTrackParameters> extrapolate(
405 const Extrapolator& propagator, const Acts::BoundTrackParameters& start,
406 const Acts::Surface& target) const {
407 auto result = extrapolateWithLength(propagator, start, target);
408 if (!result) {
409 return std::nullopt;
410 }
411 return std::move(result->first);
412 }

416 Acts::BoundTrackParameters endSmoothedParameters(Int_t track,
417 bool last) const {
418 // Throws for a bad index, like the other track accessors
419 Acts::BoundTrackParameters params = trackParameters(track);
420 const auto proxy = tracks.getTrack(track);
421 // The states run from the last to the first measurement
422 for (const auto state : proxy.trackStatesReversed()) {
423 if (!isFittedMeasurement(state)) {
424 continue;
425 }
426 params = Acts::BoundTrackParameters(
427 state.referenceSurface().getSharedPtr(), state.smoothed(),
428 state.smoothedCovariance(), proxy.particleHypothesis());
429 if (last) {
430 break;
431 }
432 }
433 return params;
434 }

483Int_t ActsStrawTracker::AddMeasurement(Int_t detectorId, Double_t driftRadius,
484 Double_t resolution) {
485 const Acts::Surface* surface = fImpl->straws.surface(detectorId);
486 const StrawWire* wire = fImpl->straws.wire(detectorId);
487 if (surface == nullptr || wire == nullptr) {
488 return -1;
489 }
490 const double sigma = resolution * ActsFrame::kShipLength;
491 fImpl->measurements.push_back({surface, wire,
492 std::abs(driftRadius) * ActsFrame::kShipLength,
493 sigma * sigma});
494 return static_cast<Int_t>(fImpl->measurements.size() - 1);
495}

512Acts::BoundTrackParameters ActsStrawTracker::Impl::seed(
513 const TrackCandidate& candidate, const StrawMeasurement& first) const {
514 const LineSegment& line = candidate.upstream;
515 const double z = first.wire->z;
516 const Acts::Vector3 position =
517 ActsFrame::positionToActs(line.x(z), line.y(z), z);
518 const Acts::Vector3 direction =
519 (ActsFrame::toActsRotation() * Acts::Vector3(line.tx, line.ty, 1.))
520 .normalized();
521 const auto hypothesis = Acts::ParticleHypothesis::muon();
522 const double qOverP = hypothesis.qOverP(
523 candidate.momentum * u::GeV, static_cast<double>(candidate.charge));
524
525 // Local coordinates at the point of closest approach to the first straw
526 const Acts::Surface& surface = *first.surface;
527 const auto intersection =
528 surface.intersect(gctx, position, direction).closest();
529 const Acts::Vector3 onSurface =
530 intersection.isValid() ? intersection.position() : position;
531 auto local = surface.globalToLocal(gctx, onSurface, direction);
532 const Acts::Vector2 localPosition =
533 local.ok() ? local.value() : Acts::Vector2::Zero();
534
535 Acts::BoundVector parameters = Acts::BoundVector::Zero();
536 parameters[Acts::eBoundLoc0] = localPosition[0];
537 parameters[Acts::eBoundLoc1] = localPosition[1];
538 parameters[Acts::eBoundPhi] = Acts::VectorHelpers::phi(direction);
539 parameters[Acts::eBoundTheta] = Acts::VectorHelpers::theta(direction);
540 parameters[Acts::eBoundQOverP] = qOverP;
541 Acts::BoundVector sigmas;
542 sigmas << kSeedSigmaLoc0, kSeedSigmaLoc1, kSeedSigmaAngle, kSeedSigmaAngle,
543 kSeedRelativeSigmaP * std::abs(qOverP), kSeedSigmaTime;
544 const Acts::BoundMatrix covariance = sigmas.cwiseProduct(sigmas).asDiagonal();
545 return Acts::BoundTrackParameters(surface.getSharedPtr(), parameters,
546 covariance, hypothesis);
547}

549std::optional<std::size_t> ActsStrawTracker::Impl::fitOnce(
550 const std::vector<Int_t>& indices,
551 const Acts::BoundTrackParameters& start) {
552 std::vector<Acts::SourceLink> sourceLinks;
553 std::vector<const Acts::Surface*> surfaces;
554 for (const Int_t index : indices) {
555 sourceLinks.emplace_back(static_cast<std::size_t>(index));
556 surfaces.push_back(measurements[index].surface);
557 }
558 Acts::PropagatorPlainOptions propagatorOptions(gctx, mctx);
559 Acts::KalmanFitterOptions<Trajectory> options(gctx, mctx, std::cref(cctx),
560 extensions, propagatorOptions,
561 surfaces.front());
562 options.referenceSurfaceStrategy = Acts::TrackExtrapolationStrategy::first;
563
564 const std::size_t nTracksBefore = tracks.size();
565 auto result = fitter.fit(sourceLinks.begin(), sourceLinks.end(), start,
566 options, surfaces, tracks);
567 if (!result.ok() || !result.value().hasReferenceSurface()) {
568 // A failed fit may leave a partial track behind. Drop it, so that track
569 // indices stay contiguous.
570 while (tracks.size() > nTracksBefore) {
571 tracks.removeTrack(tracks.size() - 1);
572 }
573 return std::nullopt;
574 }
575 return result.value().index();
576}

578Int_t ActsStrawTracker::Impl::fit(const TrackCandidate& candidate,
579 const std::vector<PatRecHit>& hits) {
580 // The measurements, ordered in z, with the sides of the wires from the
581 // segment fits
582 std::vector<std::pair<double, Int_t>> ordered;
583 for (const LineSegment* segment :
584 {&candidate.upstream, &candidate.downstream}) {
585 for (std::size_t k = 0; k < segment->hits.size(); ++k) {
586 const PatRecHit& hit = hits[segment->hits[k]];
587 measurements[hit.index].sign = segment->signs[k];
588 ordered.emplace_back(hit.wire->z, hit.index);
589 }
590 }
591 std::sort(ordered.begin(), ordered.end());
592 std::vector<Int_t> indices;
593 for (const auto& [z, index] : ordered) {
594 indices.push_back(index);
595 }
596
597 // Momentum from the kink, with the field integral along this track
598 TrackCandidate refined = candidate;
599 const auto [integral, centre] = fieldIntegral(candidate.upstream);
600 const double kink = candidate.downstream.ty - candidate.upstream.ty;
601 if (integral != 0. && kink != 0.) {
602 const LineSegment& up = candidate.upstream;
603 const double pz =
604 std::min(std::abs(kKick * integral / kink), kMaxSeedMomentum);
605 refined.momentum = pz * std::sqrt(1. + up.tx * up.tx + up.ty * up.ty);
606 refined.charge = kink / integral > 0. ? 1 : -1;
607 }
608 Acts::BoundTrackParameters start =
609 seed(refined, measurements[indices.front()]);
610
611 for (int refit = 0;; ++refit) {
612 if (static_cast<int>(indices.size()) < kMinFitMeasurements) {
613 return -1;
614 }
615 const auto track = fitOnce(indices, start);
616 if (!track) {
617 return -1;
618 }
619 // Check each hit against the fitted track: take the side of the wire it
620 // passes on, and drop hits that are too far from it
621 bool changed = false;
622 std::vector<std::pair<double, Int_t>> kept;
623 const auto proxy = tracks.getTrack(*track);
624 for (const auto state : proxy.trackStatesReversed()) {
625 if (!isFittedMeasurement(state)) {
626 continue;
627 }
628 const auto index = static_cast<Int_t>(
629 state.getUncalibratedSourceLink().template get<std::size_t>());
630 auto& m = measurements[index];
631 const double predicted = state.smoothed()[Acts::eBoundLoc0];
632 const double sigma =
633 std::sqrt(m.variance + state.smoothedCovariance()(Acts::eBoundLoc0,
634 Acts::eBoundLoc0));
635 if (std::abs(std::abs(predicted) - m.driftRadius) >
636 kOutlierPull * sigma) {
637 changed = true;
638 continue;
639 }
640 const int side = predicted >= 0. ? 1 : -1;
641 if (side != m.sign && std::abs(predicted) > std::sqrt(m.variance)) {
642 m.sign = side;
643 changed = true;
644 }
645 kept.emplace_back(m.wire->z, index);
646 }
647 if (!changed || refit == kMaxRefits) {
648 trackMeasurements.push_back(indices);
649 fittedTrackMeasurements.push_back(fittedMeasurements(*track));
650 return static_cast<Int_t>(*track);
651 }
652 // Refit from the fitted parameters, with the seed uncertainties so the
653 // previous fit does not bias the next
654 auto params = proxy.createParametersAtReference();
655 start = Acts::BoundTrackParameters(
656 params.referenceSurface().getSharedPtr(), params.parameters(),
657 seed(refined, measurements[indices.front()]).covariance(),
658 params.particleHypothesis());
659 tracks.removeTrack(*track);
660 std::sort(kept.begin(), kept.end());
661 indices.clear();
662 for (const auto& [z, index] : kept) {
663 indices.push_back(index);
664 }
665 }

668Int_t ActsStrawTracker::FindTracks() {
669 auto& impl = *fImpl;
670 std::vector<Int_t> all(impl.measurements.size());
671 std::iota(all.begin(), all.end(), 0);
672 const auto hits = impl.patRecHits(all);
673 const auto candidates = impl.patRec->findTracks(hits);
674
675 // Fit every candidate, then resolve the ambiguities between the fitted
676 // tracks: best first, rejecting tracks that share too many hits with an
677 // accepted one or fit badly
678 struct Fitted {
679 std::size_t candidate;
680 double chi2PerNdf;
681 std::vector<Int_t> measurements;
682 };
683 std::vector<Fitted> fitted;
684 for (std::size_t i = 0; i < candidates.size(); ++i) {
685 const Int_t track = impl.fit(candidates[i], hits);
686 if (track < 0) {
687 continue;
688 }
689 const auto proxy = impl.tracks.getTrack(track);
690 if (proxy.nDoF() > 0) {
691 fitted.push_back(
692 {i, proxy.chi2() / proxy.nDoF(), impl.trackMeasurements[track]});
693 }
694 }
695 const auto rank = [](const Fitted& f) {
696 return f.chi2PerNdf -
697 kTrackHitWeight * static_cast<double>(f.measurements.size());
698 };
699 std::sort(fitted.begin(), fitted.end(),
700 [&](const auto& a, const auto& b) { return rank(a) < rank(b); });
701 std::vector<const Fitted*> accepted;
702 for (const auto& f : fitted) {
703 if (f.chi2PerNdf > kMaxTrackChi2PerNdf ||
704 f.measurements.size() < kMinTrackHits) {
705 continue;
706 }
707 const bool conflict =
708 std::any_of(accepted.begin(), accepted.end(), [&](const Fitted* a) {
709 int shared = 0;
710 for (const Int_t m : f.measurements) {
711 shared +=
712 std::count(a->measurements.begin(), a->measurements.end(), m);
713 }
714 const auto shorter =
715 std::min(f.measurements.size(), a->measurements.size());
716 return shared > kMaxSharedFraction * static_cast<double>(shorter);
717 });
718 if (!conflict) {
719 accepted.push_back(&f);
720 }
721 }
722 // Refit the accepted tracks into an empty container, in candidate order,
723 // so the track indices are contiguous
724 std::sort(accepted.begin(), accepted.end(),
725 [](const Fitted* a, const Fitted* b) {
726 return a->candidate < b->candidate;
727 });
728 impl.tracks.clear();
729 impl.trackMeasurements.clear();
730 impl.fittedTrackMeasurements.clear();
731 for (const Fitted* f : accepted) {
732 impl.fit(candidates[f->candidate], hits);
733 }
734 return GetNTracks();
735}

756RecoTrack ActsStrawTracker::GetTrack(Int_t track) const {
757 const auto& impl = *fImpl;
758 const auto params = impl.trackParameters(track);
759 const auto proxy = impl.tracks.getTrack(track);
760
761 std::vector<Double_t> residuals;
762 std::vector<Double_t> pulls;
763 std::vector<Double_t> xFittedPoints;
764 std::vector<Double_t> yFittedPoints;
765 std::vector<Double_t> zFittedPoints;
766 for (const auto state : proxy.trackStatesReversed()) {
767 if (!isFittedMeasurement(state)) {
768 continue;
769 }
770 const double measured = state.template calibrated<1>()(0);
771 const double variance = state.template calibratedCovariance<1>()(0, 0);
772 const double smoothed = state.smoothed()[Acts::eBoundLoc0];
773 const double smoothedVariance =
774 state.smoothedCovariance()(Acts::eBoundLoc0, Acts::eBoundLoc0);
775 const double residual = measured - smoothed;
776 // Residuals of smoothed states have variance V - HCH^T
777 const double residualVariance = variance - smoothedVariance;
778 residuals.push_back(residual / ActsFrame::kShipLength);
779 pulls.push_back(
780 residualVariance > 0. ? residual / std::sqrt(residualVariance) : 0.);
781 const Acts::Vector3 fittedPosition = ActsFrame::positionToShip(
782 tangent(state.referenceSurface(), state.smoothed(), impl.gctx).first);
783 xFittedPoints.push_back(fittedPosition.x());
784 yFittedPoints.push_back(fittedPosition.y());
785 zFittedPoints.push_back(fittedPosition.z());
786 }
787 std::reverse(residuals.begin(), residuals.end());
788 std::reverse(pulls.begin(), pulls.end());
789 std::reverse(xFittedPoints.begin(), xFittedPoints.end());
790 std::reverse(yFittedPoints.begin(), yFittedPoints.end());
791 std::reverse(zFittedPoints.begin(), zFittedPoints.end());
792
793 const Acts::Vector3 position =
794 ActsFrame::positionToShip(params.position(impl.gctx));
795 const Acts::Vector3 momentum = ActsFrame::momentumToShip(params.momentum());
796 return RecoTrack(TVector3(position.x(), position.y(), position.z()),
797 TVector3(momentum.x(), momentum.y(), momentum.z()),
798 static_cast<Int_t>(std::lround(params.charge())),
799 upperTriangle<6>(shipCovariance(params, impl.gctx)),
800 proxy.chi2(), proxy.nDoF(), proxy.nMeasurements(),
801 std::move(xFittedPoints), std::move(yFittedPoints),
802 std::move(zFittedPoints), std::move(residuals),
803 std::move(pulls));
804}

806Bool_t ActsStrawTracker::FitVertex(const std::vector<Int_t>& tracks,
807 RecoVertex& vertex) const {
808 const auto& impl = *fImpl;
809 if (tracks.size() < 2) {
810 return false;
811 }
812 std::vector<Acts::BoundTrackParameters> fitted;
813 for (const Int_t track : tracks) {
814 fitted.push_back(impl.trackParameters(track));
815 }
816
817 // As in the GenFit vertexing: start from the closest approach of the
818 // straight tracks, then extrapolate them through the field to the plane of
819 // constant z through the current estimate, and take the closest approach
820 // of their tangents there, until it no longer moves
821 Acts::Vector3 point = vertexSeed(fitted, impl.gctx);
822 std::vector<Acts::BoundTrackParameters> atPlane;
823 for (int iteration = 0; iteration < kVertexIterations; ++iteration) {
824 const auto plane = constantZPlane(point);
825 atPlane.clear();
826 for (const auto& params : fitted) {
827 auto extrapolated = impl.extrapolate(impl.extrapolator, params, *plane);
828 if (!extrapolated) {
829 return false;
830 }
831 atPlane.push_back(std::move(*extrapolated));
832 }
833 const Acts::Vector3 next = closestPoint(tangents(atPlane, impl.gctx));
834 if (!next.allFinite()) {
835 return false;
836 }
837 const double step = (next - point).norm();
838 point = next;
839 if (step < kVertexTolerance) {
840 break;
841 }
842 }
843
844 // Covariance of the vertex (and of the DOCA) from the covariances of the
845 // tangents' positions and directions on the plane, by linear propagation
846 // with numerical derivatives
847 const auto base = tangents(atPlane, impl.gctx);
848 const int nPar = 4; // loc0, loc1, phi, theta
849 const int n = static_cast<int>(atPlane.size());
850 Eigen::MatrixXd jacobian = Eigen::MatrixXd::Zero(4, nPar * n);
851 Eigen::MatrixXd covariance = Eigen::MatrixXd::Zero(nPar * n, nPar * n);
852 const bool twoTracks = n == 2;
853 const double baseDoca = twoTracks
854 ? lineDistance(base[0].first, base[0].second,
855 base[1].first, base[1].second)
856 : 0.;
857 for (int i = 0; i < n; ++i) {
858 const auto& cov = atPlane[i].covariance().value();
859 covariance.block(nPar * i, nPar * i, nPar, nPar) =
860 cov.topLeftCorner(nPar, nPar);
861 for (int k = 0; k < nPar; ++k) {
862 const double epsilon = std::max(1e-3 * std::sqrt(cov(k, k)), 1e-12);
863 Acts::BoundVector shifted = atPlane[i].parameters();
864 shifted[k] += epsilon;
865 auto lines = base;
866 lines[i] = tangent(atPlane[i].referenceSurface(), shifted, impl.gctx);
867 jacobian.block(0, nPar * i + k, 3, 1) =
868 (closestPoint(lines) - point) / epsilon;
869 if (twoTracks) {
870 jacobian(3, nPar * i + k) =
871 (lineDistance(lines[0].first, lines[0].second, lines[1].first,
872 lines[1].second) -
873 baseDoca) /
874 epsilon;
875 }
876 }
877 }
878 const Eigen::MatrixXd propagated =
879 jacobian * covariance * jacobian.transpose();
880 const Acts::SquareMatrix3 actsCovariance = propagated.topLeftCorner(3, 3);
881
882 const Acts::RotationMatrix3 rotation = ActsFrame::toShipRotation();
883 const Acts::Vector3 position = ActsFrame::positionToShip(point);
884 const Acts::SquareMatrix3 positionCovariance =
885 rotation * actsCovariance * rotation.transpose() /
886 (ActsFrame::kShipLength * ActsFrame::kShipLength);
887 // Fit quality: the DOCA against its uncertainty for two tracks
888 const double docaVariance = propagated(3, 3);
889 const double chi2 =
890 twoTracks && docaVariance > 0. ? baseDoca * baseDoca / docaVariance : 0.;
891 vertex = RecoVertex(TVector3(position.x(), position.y(), position.z()),
892 upperTriangle<3>(positionCovariance), chi2,
893 twoTracks ? 1. : 2. * n - 3.);
894
895 for (int i = 0; i < n; ++i) {
896 const Acts::Vector3 momentum =
897 ActsFrame::momentumToShip(atPlane[i].momentum());
898 vertex.AddTrack(tracks[i],
899 TVector3(momentum.x(), momentum.y(), momentum.z()),
900 upperTriangle<3>(shipMomentumCovariance(atPlane[i])));
901 }
902 if (twoTracks) {
903 vertex.SetDoca(baseDoca / ActsFrame::kShipLength);
904 }
905 return true;
906}

908Double_t ActsStrawTracker::DistanceToHit(Int_t track,
909 const TVector3& hit) const {
910 const auto& impl = *fImpl;
911 // Plane of constant z centred on the hit
912 const auto plane =
913 constantZPlane(ActsFrame::positionToActs(hit.X(), hit.Y(), hit.Z()));
914
915 const auto result =
916 impl.extrapolate(impl.extrapolator, impl.trackParameters(track), *plane);
917 if (!result) {
918 return -1.;
919 }
920 const auto& local = result->parameters();
921 return std::hypot(local[Acts::eBoundLoc0], local[Acts::eBoundLoc1]) /
922 ActsFrame::kShipLength;
923}

925Bool_t ActsStrawTracker::ExtrapolateToZ(Int_t track, Double_t z,
926 Bool_t fromLastHit, TVector3& position,
927 TVector3& momentum,
928 std::vector<Double_t>& covariance,
929 Double_t& pathLength) const {
930 const auto& impl = *fImpl;
931 const auto start = impl.endSmoothedParameters(track, fromLastHit);
932 const auto plane = constantZPlane(ActsFrame::positionToActs(0., 0., z));
933 const auto result =
934 impl.extrapolateWithLength(impl.extrapolator, start, *plane);
935 if (!result) {
936 return false;
937 }
938 const auto& [params, length] = *result;
939 const Acts::Vector3 pos =
940 ActsFrame::positionToShip(params.position(impl.gctx));
941 const Acts::Vector3 mom = ActsFrame::momentumToShip(params.momentum());
942 position.SetXYZ(pos.x(), pos.y(), pos.z());
943 momentum.SetXYZ(mom.x(), mom.y(), mom.z());
944 covariance.clear();
945 if (params.covariance().has_value()) {
946 const auto elements = upperTriangle<6>(shipCovariance(params, impl.gctx));
947 covariance.assign(elements.begin(), elements.end());
948 }
949 pathLength = length / ActsFrame::kShipLength;
950 return true;
951}

953Double_t ActsStrawTracker::TrackLength(Int_t track) const {
954 const auto& impl = *fImpl;
955 const auto first = impl.endSmoothedParameters(track, false);
956 const auto last = impl.endSmoothedParameters(track, true);
957 if (&first.referenceSurface() == &last.referenceSurface()) {
958 // a single fitted measurement
959 return std::numeric_limits<Double_t>::quiet_NaN();
960 }
961 const auto result = impl.extrapolateWithLength(impl.extrapolator, first,
962 last.referenceSurface());
963 if (!result) {
964 return std::numeric_limits<Double_t>::quiet_NaN();
965 }
966 return std::abs(result->second) / ActsFrame::kShipLength;
967}
