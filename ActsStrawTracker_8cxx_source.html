84template <int N, std::size_t M = N * (N + 1) / 2>
85std::array<Double_t, M> upperTriangle(const Acts::SquareMatrix<N>& matrix) {
86 std::array<Double_t, M> elements{};
87 std::size_t k = 0;
88 for (int row = 0; row < N; ++row) {
89 for (int col = row; col < N; ++col) {
90 elements[k++] = matrix(row, col);
91 }
92 }
93 return elements;
94}

98 return Acts::SquareMatrix3::Identity() / std::abs(qOverP);

101Acts::Vector3 momentumJacobianQOverP(const Acts::Vector3& direction,
102 double qOverP) {
103 return -direction / (qOverP * std::abs(qOverP));

107Acts::SquareMatrix<6> shipCovariance(const Acts::BoundTrackParameters& params,
108 const Acts::GeometryContext& gctx) {
109 const auto& surface = params.referenceSurface();
110 const Acts::FreeVector free =
111 Acts::transformBoundToFreeParameters(surface, gctx, params.parameters());
112 const Acts::Vector3 direction = free.segment<3>(Acts::eFreeDir0);
113 const Acts::BoundToFreeMatrix boundToFree = surface.boundToFreeJacobian(
114 gctx, free.segment<3>(Acts::eFreePos0), direction);
115 const Acts::FreeMatrix freeCovariance =
116 boundToFree * params.covariance().value() * boundToFree.transpose();
117
118 const double qOverP = free[Acts::eFreeQOverP];
119 const Acts::RotationMatrix3 rotation = ActsFrame::toShipRotation();
120 Acts::Matrix<6, Acts::eFreeSize> freeToShip =
121 Acts::Matrix<6, Acts::eFreeSize>::Zero();
122 freeToShip.block<3, 3>(0, Acts::eFreePos0) =
123 rotation / ActsFrame::kShipLength;
124 freeToShip.block<3, 3>(3, Acts::eFreeDir0) =
125 rotation * momentumJacobianDirection(qOverP) / u::GeV;
126 freeToShip.block<3, 1>(3, Acts::eFreeQOverP) =
127 rotation * momentumJacobianQOverP(direction, qOverP) / u::GeV;
128 return freeToShip * freeCovariance * freeToShip.transpose();
129}

132Acts::SquareMatrix3 shipMomentumCovariance(
133 const Acts::BoundTrackParameters& params) {
134 const double phi = params.parameters()[Acts::eBoundPhi];
135 const double theta = params.parameters()[Acts::eBoundTheta];
136 const double qOverP = params.parameters()[Acts::eBoundQOverP];
137 const double p = 1. / std::abs(qOverP);
138 const Acts::Vector3 direction = params.direction();
139 Acts::Matrix<3, Acts::eBoundSize> boundToMomentum =
140 Acts::Matrix<3, Acts::eBoundSize>::Zero();
141 boundToMomentum.col(Acts::eBoundPhi) =
142 Acts::Vector3(-std::sin(phi) * std::sin(theta),
143 std::cos(phi) * std::sin(theta), 0.) *
144 p;
145 boundToMomentum.col(Acts::eBoundTheta) =
146 Acts::Vector3(std::cos(phi) * std::cos(theta),
147 std::sin(phi) * std::cos(theta), -std::sin(theta)) *
148 p;
149 boundToMomentum.col(Acts::eBoundQOverP) =
150 momentumJacobianQOverP(direction, qOverP);
151 const Acts::Matrix<3, Acts::eBoundSize> jacobian =
152 ActsFrame::toShipRotation() * boundToMomentum / u::GeV;
153 return jacobian * params.covariance().value() * jacobian.transpose();
154}

157Acts::Vector3 vertexSeed(const std::vector<Acts::BoundTrackParameters>& tracks,
158 const Acts::GeometryContext& gctx) {
159 Acts::SquareMatrix3 a = Acts::SquareMatrix3::Zero();
160 Acts::Vector3 b = Acts::Vector3::Zero();
161 for (const auto& track : tracks) {
162 const Acts::Vector3 direction = track.direction();
163 const Acts::SquareMatrix3 projection =
164 Acts::SquareMatrix3::Identity() - direction * direction.transpose();
165 a += projection;
166 b += projection * track.position(gctx);
167 }
168 Acts::Vector3 seed = a.colPivHouseholderQr().solve(b);
169 if (!seed.allFinite()) {
170 seed = Acts::Vector3(0.5 * (kVertexSeedMinX + kVertexSeedMaxX), 0., 0.);
171 }
172 seed.x() = std::clamp(seed.x(), kVertexSeedMinX, kVertexSeedMaxX);
173 return seed;
174}
175
177double lineDistance(const Acts::Vector3& p1, const Acts::Vector3& d1,
178 const Acts::Vector3& p2, const Acts::Vector3& d2) {
179 const Acts::Vector3 delta = p1 - p2;
180 const Acts::Vector3 normal = d1.cross(d2);
181 if (normal.norm() > 1e-12) {
182 return std::abs(delta.dot(normal)) / normal.norm();
183 }
184 return delta.cross(d1.normalized()).norm();
185}

214 void calibrate(const Acts::GeometryContext& /*gctx*/,
215 const Acts::CalibrationContext& /*cctx*/,
216 const Acts::SourceLink& sourceLink,
217 Trajectory::TrackStateProxy state) const {
218 const auto& measurement = measurements.at(sourceLink.get<std::size_t>());
219 state.allocateCalibrated(1);
220 state.template calibrated<1>()(0) = measurement.driftRadius;
221 state.template calibratedCovariance<1>()(0, 0) = measurement.variance;
222 state.setProjectorSubspaceIndices(std::array{Acts::eBoundLoc0});
223 state.setUncalibratedSourceLink(Acts::SourceLink(sourceLink));

234 std::optional<Acts::BoundTrackParameters> extrapolate(
235 const Extrapolator& propagator, const Acts::BoundTrackParameters& start,
236 const Acts::Surface& target) const {
237 for (const auto direction :
238 {Acts::Direction::Forward(), Acts::Direction::Backward()}) {
239 Extrapolator::Options<> options(gctx, mctx);
240 options.pathLimit = kPathLimit;
241 options.direction = direction;
242 auto result = propagator.propagate(start, target, options);
243 if (result.ok() && result.value().endParameters.has_value()) {
244 return result.value().endParameters.value();
245 }
246 }
247 return std::nullopt;
248 }

280Int_t ActsStrawTracker::AddMeasurement(Int_t detectorId, Double_t driftRadius,
281 Double_t resolution) {
282 const Acts::Surface* surface = fImpl->straws.surface(detectorId);
283 if (surface == nullptr) {
284 return -1;
285 }
286 const double sigma = resolution * ActsFrame::kShipLength;
287 fImpl->measurements.push_back(
288 {surface, driftRadius * ActsFrame::kShipLength, sigma * sigma});
289 return static_cast<Int_t>(fImpl->measurements.size() - 1);
290}

292Int_t ActsStrawTracker::FitTrack(const std::vector<Int_t>& measurements,
293 const TVector3& seedPosition,
294 const TVector3& seedMomentum, Int_t charge) {
295 auto& impl = *fImpl;
296 if (measurements.empty() || charge == 0 || seedMomentum.Mag() <= 0.) {
297 return -1;
298 }
299 std::vector<Acts::SourceLink> sourceLinks;
300 std::vector<const Acts::Surface*> surfaces;
301 for (const Int_t index : measurements) {
302 if (index < 0 ||
303 static_cast<std::size_t>(index) >= impl.measurements.size()) {
304 throw std::out_of_range("ActsStrawTracker: measurement out of range");
305 }
306 sourceLinks.emplace_back(static_cast<std::size_t>(index));
307 surfaces.push_back(impl.measurements[index].surface);
308 }
309
310 // Seed on the surface of the first measurement
311 const Acts::Surface& seedSurface = *surfaces.front();
312 const Acts::Vector3 position = ActsFrame::positionToActs(
313 seedPosition.X(), seedPosition.Y(), seedPosition.Z());
314 const Acts::Vector3 momentum = ActsFrame::momentumToActs(
315 seedMomentum.X(), seedMomentum.Y(), seedMomentum.Z());
316 const Acts::Vector3 direction = momentum.normalized();
317 const auto hypothesis = Acts::ParticleHypothesis::muon();
318 const double qOverP =
319 hypothesis.qOverP(momentum.norm(), static_cast<double>(charge));
320
321 // The seed is generally not on the straw, so take its point of closest
322 // approach and the local coordinates there
323 const auto intersection =
324 seedSurface.intersect(impl.gctx, position, direction).closest();
325 const Acts::Vector3 onSurface =
326 intersection.isValid() ? intersection.position() : position;
327 auto local = seedSurface.globalToLocal(impl.gctx, onSurface, direction);
328 const Acts::Vector2 localPosition =
329 local.ok() ? local.value() : Acts::Vector2::Zero();
330
331 Acts::BoundVector parameters = Acts::BoundVector::Zero();
332 parameters[Acts::eBoundLoc0] = localPosition[0];
333 parameters[Acts::eBoundLoc1] = localPosition[1];
334 parameters[Acts::eBoundPhi] = Acts::VectorHelpers::phi(direction);
335 parameters[Acts::eBoundTheta] = Acts::VectorHelpers::theta(direction);
336 parameters[Acts::eBoundQOverP] = qOverP;
337 Acts::BoundVector sigmas;
338 sigmas << kSeedSigmaLoc0, kSeedSigmaLoc1, kSeedSigmaAngle, kSeedSigmaAngle,
339 kSeedRelativeSigmaP * std::abs(qOverP), kSeedSigmaTime;
340 const Acts::BoundMatrix covariance = sigmas.cwiseProduct(sigmas).asDiagonal();
341 const Acts::BoundTrackParameters seed(seedSurface.getSharedPtr(), parameters,
342 covariance, hypothesis);
343
344 Acts::PropagatorPlainOptions propagatorOptions(impl.gctx, impl.mctx);
345 Acts::KalmanFitterOptions<Trajectory> options(
346 impl.gctx, impl.mctx, std::cref(impl.cctx), impl.extensions,
347 propagatorOptions, &seedSurface);
348 options.referenceSurfaceStrategy = Acts::TrackExtrapolationStrategy::first;
349
350 const std::size_t nTracksBefore = impl.tracks.size();
351 auto result = impl.fitter.fit(sourceLinks.begin(), sourceLinks.end(), seed,
352 options, surfaces, impl.tracks);
353 if (!result.ok() || !result.value().hasReferenceSurface()) {
354 // A failed fit may leave a partial track behind. Drop it, so that track
355 // indices stay contiguous.
356 while (impl.tracks.size() > nTracksBefore) {
357 impl.tracks.removeTrack(impl.tracks.size() - 1);
358 }
359 return -1;
360 }
361 return static_cast<Int_t>(result.value().index());
362}

368RecoTrack ActsStrawTracker::GetTrack(Int_t track) const {
369 const auto& impl = *fImpl;
370 const auto params = impl.trackParameters(track);
371 const auto proxy = impl.tracks.getTrack(track);
372
373 std::vector<Double_t> residuals;
374 std::vector<Double_t> pulls;
375 for (const auto state : proxy.trackStatesReversed()) {
376 if (!state.hasCalibrated() || !state.hasSmoothed()) {
377 continue;
378 }
379 const double measured = state.template calibrated<1>()(0);
380 const double variance = state.template calibratedCovariance<1>()(0, 0);
381 const double smoothed = state.smoothed()[Acts::eBoundLoc0];
382 const double smoothedVariance =
383 state.smoothedCovariance()(Acts::eBoundLoc0, Acts::eBoundLoc0);
384 const double residual = measured - smoothed;
385 // Residuals of smoothed states have variance V - HCH^T
386 const double residualVariance = variance - smoothedVariance;
387 residuals.push_back(residual / ActsFrame::kShipLength);
388 pulls.push_back(
389 residualVariance > 0. ? residual / std::sqrt(residualVariance) : 0.);
390 }
391 std::reverse(residuals.begin(), residuals.end());
392 std::reverse(pulls.begin(), pulls.end());
393
394 const Acts::Vector3 position =
395 ActsFrame::positionToShip(params.position(impl.gctx));
396 const Acts::Vector3 momentum = ActsFrame::momentumToShip(params.momentum());
397 return RecoTrack(TVector3(position.x(), position.y(), position.z()),
398 TVector3(momentum.x(), momentum.y(), momentum.z()),
399 static_cast<Int_t>(std::lround(params.charge())),
400 upperTriangle<6>(shipCovariance(params, impl.gctx)),
401 proxy.chi2(), proxy.nDoF(), proxy.nMeasurements(),
402 std::move(residuals), std::move(pulls));
403}

405Bool_t ActsStrawTracker::FitVertex(const std::vector<Int_t>& tracks,
406 RecoVertex& vertex) const {
407 const auto& impl = *fImpl;
408 if (tracks.size() < 2) {
409 return false;
410 }
411 std::vector<Acts::BoundTrackParameters> fitted;
412 for (const Int_t track : tracks) {
413 fitted.push_back(impl.trackParameters(track));
414 }
415 const Acts::Vector3 seed = vertexSeed(fitted, impl.gctx);
416
417 // Where the field at the seed is negligible, the helical linearisation is
418 // ill-defined, so treat the tracks as straight lines there
419 std::shared_ptr<const Acts::MagneticFieldProvider> field = impl.field;
420 auto cache = field->makeCache(impl.mctx);
421 const auto seedField = field->getField(seed, cache);
422 if (seedField.ok() && seedField.value().norm() < kNegligibleField) {
423 field = std::make_shared<Acts::NullBField>();
424 }
425 auto propagator = std::make_shared<Extrapolator>(
426 Stepper(field), Acts::VoidNavigator(),
427 Acts::getDefaultLogger("VertexPropagator", Acts::Logging::WARNING));
428
429 // Start the fit from the track parameters at the seed
430 const auto seedSurface =
431 Acts::Surface::makeShared<Acts::PerigeeSurface>(seed);
432 std::vector<Acts::BoundTrackParameters> atSeed;
433 for (const auto& params : fitted) {
434 auto extrapolated = impl.extrapolate(*propagator, params, *seedSurface);
435 if (!extrapolated) {
436 return false;
437 }
438 atSeed.push_back(std::move(*extrapolated));
439 }
440
441 Acts::HelicalTrackLinearizer::Config linearizerConfig;
442 linearizerConfig.bField = field;
443 linearizerConfig.propagator = propagator;
444 linearizerConfig.targetTolerance = 1e-4;
445 const Acts::HelicalTrackLinearizer linearizer(linearizerConfig);
446
447 Acts::FullBilloirVertexFitter::Config fitterConfig;
448 fitterConfig.maxIterations = 200;
449 fitterConfig.extractParameters
450 .connect<&Acts::InputTrack::extractParameters>();
451 fitterConfig.trackLinearizer
452 .connect<&Acts::HelicalTrackLinearizer::linearizeTrack>(&linearizer);
453 const Acts::FullBilloirVertexFitter fitter(
454 fitterConfig,
455 Acts::getDefaultLogger("VertexFitter", Acts::Logging::WARNING));
456
457 std::vector<Acts::InputTrack> inputTracks;
458 for (const auto& params : atSeed) {
459 inputTracks.emplace_back(&params);
460 }
461 // A loose constraint at the seed only sets the starting point of the fit
462 Acts::Vertex constraint(seed);
463 constraint.setFullCovariance(Acts::SquareMatrix4::Identity() * 1e6);
464 const Acts::VertexingOptions vertexingOptions(impl.gctx, impl.mctx,
465 constraint);
466 auto fieldCache = field->makeCache(impl.mctx);
467 const auto result = fitter.fit(inputTracks, vertexingOptions, fieldCache);
468 if (!result.ok()) {
469 return false;
470 }
471 const Acts::Vertex& fittedVertex = result.value();
472
473 const Acts::RotationMatrix3 rotation = ActsFrame::toShipRotation();
474 const Acts::Vector3 position =
475 ActsFrame::positionToShip(fittedVertex.position());
476 const Acts::SquareMatrix3 positionCovariance =
477 rotation * fittedVertex.covariance() * rotation.transpose() /
478 (ActsFrame::kShipLength * ActsFrame::kShipLength);
479 const auto [chi2, nDoF] = fittedVertex.fitQuality();
480 vertex = RecoVertex(TVector3(position.x(), position.y(), position.z()),
481 upperTriangle<3>(positionCovariance), chi2, nDoF);
482
483 for (const auto& trackAtVertex : fittedVertex.tracks()) {
484 const auto* original =
485 trackAtVertex.originalParams.as<Acts::BoundTrackParameters>();
486 const auto index = static_cast<std::size_t>(original - atSeed.data());
487 if (original == nullptr || index >= atSeed.size()) {
488 return false;
489 }
490 const auto& params = trackAtVertex.fittedParams;
491 const Acts::Vector3 momentum = ActsFrame::momentumToShip(params.momentum());
492 vertex.AddTrack(tracks[index],
493 TVector3(momentum.x(), momentum.y(), momentum.z()),
494 upperTriangle<3>(shipMomentumCovariance(params)));
495 }
496
497 if (atSeed.size() == 2) {
498 const double doca =
499 lineDistance(atSeed[0].position(impl.gctx), atSeed[0].direction(),
500 atSeed[1].position(impl.gctx), atSeed[1].direction());
501 vertex.SetDoca(doca / ActsFrame::kShipLength);
502 }
503 return true;
504}

506Double_t ActsStrawTracker::DistanceToHit(Int_t track,
507 const TVector3& hit) const {
508 const auto& impl = *fImpl;
509 // Plane of constant z (SHiP), i.e. constant x (ACTS), centred on the hit
510 Acts::Transform3 transform = Acts::Transform3::Identity();
511 transform.linear().col(0) = Acts::Vector3::UnitY();
512 transform.linear().col(1) = Acts::Vector3::UnitZ();
513 transform.linear().col(2) = Acts::Vector3::UnitX();
514 transform.translation() =
515 ActsFrame::positionToActs(hit.X(), hit.Y(), hit.Z());
516 const auto plane = Acts::Surface::makeShared<Acts::PlaneSurface>(transform);
517
518 const auto result =
519 impl.extrapolate(impl.extrapolator, impl.trackParameters(track), *plane);
520 if (!result) {
521 return -1.;
522 }
523 const auto& local = result->parameters();
524 return std::hypot(local[Acts::eBoundLoc0], local[Acts::eBoundLoc1]) /
525 ActsFrame::kShipLength;
526}
