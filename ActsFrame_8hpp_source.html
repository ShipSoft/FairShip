56inline Acts::Transform3 transformToActs(const Acts::Transform3& shipTransform) {
57 Acts::Transform3 transform = Acts::Transform3::Identity();
58 transform.linear() = toActsRotation() * shipTransform.linear();
59 transform.translation() =
60 toActsRotation() * shipTransform.translation() * kShipLength;
61 return transform;
