61Int_t HistogramId(Int_t pdg) {
62 return 10000 + TMath::Abs(pdg) + (pdg < 0 ? 10000 : 0);

65// ShipStack::PushTrack multiplies the weight of a pushed track by that of the
66// track it names as parent, and leaves it alone for a primary. A clone that
67// should end up with weight w is therefore pushed with w divided by this.
68Double_t PushedWeightScale(Int_t parentId) {
69 if (parentId < 0) {
70 return 1.;
71 }
72 auto* stack = dynamic_cast<ShipStack*>(gMC->GetStack());
73 TParticle* parent = stack->GetParticle(parentId);
74 if (!parent || parent->GetWeight() <= 0.) {
75 return 1.;
76 }
77 return parent->GetWeight();
78}

127Bool_t exitHadronAbsorber::ProcessHits(FairVolume* vol) {
129 TString volName = gMC->CurrentVolName();
130
131 // Only this detector's own plane: with per-step splitting the volumes
132 // registered below the target include other exitHadronAbsorber planes, whose
133 // names share the prefix.
134 if (volName == fPlaneVolName) {
135 if (gMC->IsTrackEntering()) {
136 fTrackID = gMC->GetStack()->GetCurrentTrackNumber();
137 fEventID = gMC->CurrentEvent();
138 TParticle* p = gMC->GetStack()->GetCurrentTrack();
139 fUniqueID = p->GetUniqueID();
140 Int_t pdgCode = p->GetPdgCode();
141 Int_t motherId = p->GetFirstMother();
142 gMC->TrackMomentum(fMom);
143 if (!fOnlyMuons || TMath::Abs(pdgCode) == 13) {
144 fTime = gMC->TrackTime() * 1.0e09;
145 fLength = gMC->TrackLength();
146 gMC->TrackPosition(fPos);
147 if (((fMom.E() - fMom.M()) > EMax) &&
148 (fDecayedParentIDs.count(motherId) == 0)) {
149 AddHit(fEventID, fTrackID, 111,
150 TVector3(fPos.X(), fPos.Y(), fPos.Z()),
151 TVector3(fMom.Px(), fMom.Py(), fMom.Pz()), fTime, fLength, 0,
152 pdgCode, TVector3(p->Vx(), p->Vy(), p->Vz()),
153 TVector3(p->Px(), p->Py(), p->Pz()));
154 auto* stack = dynamic_cast<ShipStack*>(gMC->GetStack());
155 stack->AddPoint(kVETO);
156 }
157 }
158 }
159
160 if ((!fCylindricalPlane) && fzPos > 1E8) {
161 gMC->StopTrack();
162 }
163 }
164
165 // Both counts have to be positive: PostTrack() only terminates the ledger,
166 // handing what is left of it to the endpoint clones, when fNsplits > 0.
167 // run_fixedTarget.py pairs the two options for the same reason.
168 if (fNsplits > 0 && fIntermediateNsplits > 0 && (!fSplitOnce)) {
169 Int_t currentTrackId = gMC->GetStack()->GetCurrentTrackNumber();
170
171 if (fCloneTracks.count(currentTrackId) > 0) {
172 return kTRUE;
173 }
174
175 Int_t track_pid = gMC->TrackPid();
176 bool kaon_or_pion =
177 (TMath::Abs(track_pid) == 211 || TMath::Abs(track_pid) == 321);
178
179 if (kaon_or_pion) {
180 TParticle* part = gMC->GetStack()->GetCurrentTrack();
181 if (!part) return kTRUE;
182
183 Double_t delta_s = gMC->TrackStep();
184
185 if (delta_s > 0) {
186 TLorentzVector pos, mom;
187 gMC->TrackPosition(pos);
188 gMC->TrackMomentum(mom);
189
190 Double_t mass = gMC->TrackMass();
191 Double_t tau = gMC->ParticleLifeTime(track_pid); // in ns, unconverted
192 Double_t c_tau = tau * ShipUnit::c_light; // in cm
193
194 Double_t instantP = mom.P();
195
196 Double_t lambda_decay = (instantP / mass) * c_tau;
197 Double_t P_decay = 1.0 - TMath::Exp(-delta_s / lambda_decay);
198 if ((P_decay > 0.0) && ((mom.E() - mom.M()) > EMax)) {
199 double polX = 0, polY = 0, polZ = 0;
200 TVector3 polVector;
201 part->GetPolarisation(polVector);
202 polX = polVector.X();
203 polY = polVector.Y();
204 polZ = polVector.Z();
205 Int_t trueParentId = part->GetFirstMother();
206
207 // Reserve room for the fNsplits endpoint clones PostTrack() appends
208 // when the parent finally decays, so the buffer never exceeds
209 // fMaxSplitBuffer. All terms are on the left to keep the unsigned
210 // arithmetic from wrapping.
211 const std::size_t secondaryBufferSize = fSecondaryBuffer.size();
212 const std::size_t event_size = gMC->GetStack()->GetNtrack();
213
214 const std::size_t pending =
215 secondaryBufferSize +
216 static_cast<std::size_t>(fIntermediateNsplits) +
217 static_cast<std::size_t>(fNsplits);
218 const bool bufferFull = pending > fMaxSplitBuffer;
219 const std::size_t projected_size =
220 event_size + pending * kShowerSafetyFactor;
221 const bool eventFull = projected_size > fMaxEventSize;
222 if (bufferFull || eventFull) {
223 // Skip the split for this step instead of truncating the buffer.
224 // The track weight is the ledger: leaving it untouched
225 // means the weight we did not split off is still carried by the
226 // track, and reaches the natural-decay clones in PostTrack() if
227 // the track ends in a decay. The total weight the track
228 // contributes is unchanged, only the statistical boost is reduced.
229 if (bufferFull && !fSplitBufferLimitWarned) {
230 LOG(warning) << "exitHadronAbsorber: intermediate split buffer "
231 "reached "
232 << secondaryBufferSize
233 << " entries; skipping further per-step splitting "
234 "(reported once per event). Consider lowering "
235 "--intermediate-kaon-pion-splits or raising "
236 "--max-split-buffer.";
237 fSplitBufferLimitWarned = kTRUE;
238 }
239 if (eventFull && !fEventSizeLimitWarned) {
240 LOG(warning) << "exitHadronAbsorber: event size reached "
241 << event_size << " entries, projected size "
242 << projected_size
243 << " with the pending split clones; skipping "
244 "further per-step splitting (reported once per "
245 "event). Consider lowering "
246 "--intermediate-kaon-pion-splits or raising "
247 "--max-event-size.";
248 fEventSizeLimitWarned = kTRUE;
249 }
250 return kTRUE;
251 }
252
253 // The track's own weight is the ledger: it starts at whatever the
254 // track was pushed with and loses the decay branch at every split.
255 // The clones get exactly what the track loses, so track + clones
256 // keep the weight the track started with at every step, whatever
257 // that weight is, even if it differs from that of the track
258 // ShipStack scales the clones by (see PushedWeightScale). Lowering
259 // the track's weight also keeps everything it goes on to do in
260 // step: its hit at the sensitive plane, and the secondaries it
261 // makes if it ends by interacting rather than decaying. Geant4
262 // pushes a secondary onto the VMC stack when that secondary starts
263 // tracking, which is after the parent is done, so the secondaries
264 // pick this up on their own. This relies on
265 // /mcTracking/saveSecondariesInStep staying off (see
266 // gconfig/g4config.in); turning it on pushes secondaries during the
267 // step that makes them. A side effect is that secondaries made
268 // mid-track by processes that do not end it (delta rays, elastic
269 // recoils) get the parent's final weight, not its weight at the
270 // step that made them. That is negligible for the muon rate, but do
271 // not rely on their weights.
272 const Double_t trackWeight = part->GetWeight();
273 const Double_t cloneWeight = trackWeight * P_decay /
274 fIntermediateNsplits /
275 PushedWeightScale(trueParentId);
276 for (int i = 0; i < fIntermediateNsplits; ++i) {
277 TrackBuffer clone;
278 clone.pdg = track_pid;
279 clone.px = mom.Px();
280 clone.py = mom.Py();
281 clone.pz = mom.Pz();
282 clone.e = mom.E();
283 clone.x = pos.X();
284 clone.y = pos.Y();
285 clone.z = pos.Z();
286 clone.t = pos.T();
287 clone.polx = polX;
288 clone.poly = polY;
289 clone.polz = polZ;
290 clone.weight = cloneWeight;
291 clone.parentID = trueParentId;
292 fSecondaryBuffer.push_back(clone);
293 }
294 part->SetWeight(trackWeight * (1.0 - P_decay));
295 }
296 }
297 }
298 }
299 return kTRUE;
300}

302void exitHadronAbsorber::Initialize() {
303 SHiP::Detector<vetoPoint>::Initialize();
304 // PostTrack() buffers fNsplits endpoint clones for every splitting decay in
305 // both modes, without consulting the cap, so a cap below fNsplits breaks the
306 // hard bound. With per-step splitting it also makes the ProcessHits()
307 // reservation permanently unsatisfiable, disabling per-step splitting.
308 if (fNsplits > 0 && static_cast<std::size_t>(fNsplits) > fMaxSplitBuffer) {
309 LOG(fatal) << "exitHadronAbsorber: max split buffer (" << fMaxSplitBuffer
310 << ") must be at least the endpoint split count (" << fNsplits
311 << ")";
312 }
313 TSeqCollection* fileList = gROOT->GetListOfFiles();
314 fout = dynamic_cast<TFile*>(fileList->At(0));
315 // book hists for Genie neutrino momentum distribution, and for the other
316 // particles in kHistogrammed
317 TDatabasePDG* PDG = TDatabasePDG::Instance();
318 for (const auto& particle : kHistogrammed) {
319 for (Int_t sign : {1, -1}) {
320 if (sign < 0 && !particle.withAntiparticle) {
321 continue;
322 }
323 const Int_t idw = sign * particle.pdg;
324 const Int_t idhnu = HistogramId(idw);
325 TString name = PDG->GetParticle(idw)->GetName();
326 TString title = name;
327 title += " momentum (GeV)";
328 TString key = fVetoName;
329 key += idhnu;
330 [[maybe_unused]] TH1D* Hidhnu = new TH1D(key, title, 400, 0., 400.);
331 title = name;
332 title += " log10-p vs log10-pt";
333 key = fVetoName;
334 key += idhnu + 1000;
335 [[maybe_unused]] TH2D* Hidhnu100 =
336 new TH2D(key, title, 100, -0.3, 1.7, 100, -2., 0.5);
337 title = name;
338 title += " log10-p vs log10-pt";
339 key = fVetoName;
340 key += idhnu + 2000;
341 [[maybe_unused]] TH2D* Hidhnu200 =
342 new TH2D(key, title, 25, -0.3, 1.7, 100, -2., 0.5);
343 }
344 }
345 if (withNtuple) {
346 fNtuple = new TNtuple("4DP", "4DP", "id:px:py:pz:x:y:z");
347 }

357void exitHadronAbsorber::FinishEvent() {
358 // Checked here rather than at the start of the next event so that the last
359 // event of a run is covered too.
360 if (!fSecondaryBuffer.empty()) {
361 // No track that went on to step followed the last splitting decay of the
362 // event, so its clones could not be handed to the stack popper and their
363 // weight is lost.
364 Double_t lostWeight = 0;
365 for (const auto& trk : fSecondaryBuffer) {
366 lostWeight += trk.weight;
367 }
368 LOG(warning) << "exitHadronAbsorber: discarding " << fSecondaryBuffer.size()
369 << " buffered split clones (summed weight " << lostWeight
370 << ") left over at the end of the event";
371 }
372 fSecondaryBuffer.clear();
373}

375void exitHadronAbsorber::PostTrack() {
376 Int_t currentTrackId = gMC->GetStack()->GetCurrentTrackNumber();
377
378 if (fCloneTracks.count(currentTrackId) > 0) {
379 return;
380 }
381
382 Int_t track_pid = gMC->TrackPid();
383 bool kaon_or_pion =
384 (TMath::Abs(track_pid) == 211 || TMath::Abs(track_pid) == 321);
385
386 if (fNsplits > 0 && kaon_or_pion) {
387 bool isNaturalDecay = false;
388 TArrayI processes;
389 gMC->StepProcesses(processes);
390 for (int i = 0; i < processes.GetSize(); i++) {
391 if (processes[i] == kPDecay) {
392 isNaturalDecay = true;
393 break;
394 }
395 }
396
397 TParticle* part = gMC->GetStack()->GetCurrentTrack();
398 if (!part) return;
399
400 TLorentzVector finalPos, finalMom;
401 gMC->TrackPosition(finalPos);
402 gMC->TrackMomentum(finalMom);
403
404 double polX = 0, polY = 0, polZ = 0;
405 TVector3 polVector;
406 part->GetPolarisation(polVector);
407 polX = polVector.X();
408 polY = polVector.Y();
409 polZ = polVector.Z();
410 Int_t trueParentId = part->GetFirstMother();
411
412 // A track that decays hands all of its remaining weight to the endpoint
413 // clones. With per-step splitting its weight has already been lowered by
414 // every split along the way.
415 if (isNaturalDecay) {
416 Double_t finalEndpointWeight =
417 part->GetWeight() / fNsplits / PushedWeightScale(trueParentId);
418 for (int i = 0; i < fNsplits; ++i) {
419 TrackBuffer clone;
420 clone.pdg = track_pid;
421 clone.px = finalMom.Px();
422 clone.py = finalMom.Py();
423 clone.pz = finalMom.Pz();
424 clone.e = finalMom.E();
425 clone.x = finalPos.X();
426 clone.y = finalPos.Y();
427 clone.z = finalPos.Z();
428 clone.t = finalPos.T();
429 clone.polx = polX;
430 clone.poly = polY;
431 clone.polz = polZ;
432 clone.weight = finalEndpointWeight;
433 clone.parentID = trueParentId;
434 fSecondaryBuffer.push_back(clone);
435 }
436 fDecayedParentIDs.insert(currentTrackId);
437 }
438
439 // A track that ends any other way keeps the remaining weight itself, so
440 // there is nothing to hand on. Geant4 never gives us a track that is
441 // still going: G4TrackingManager steps while the status is fAlive or
442 // fStopButAlive, TG4TrackingAction::PostUserTrackingAction skips
443 // PostTrack for fSuspend, and IsTrackStop() covers every status that is
444 // left. StopTrack() stays because it is the one call here that can still
445 // matter for a status other than fStopAndKill.
446 gMC->StopTrack();
447 }

450void exitHadronAbsorber::PreTrack() {
451 // Invariant for this whole method: the clone buffer may only be handed to a
452 // carrier that is guaranteed to step. TG4StackPopper converts pushed tracks
453 // into Geant4 secondaries from PostStepDoIt only, and its Reset() at the
454 // start of the next track writes off everything still pending, so a carrier
455 // stopped before its first step swallows the entire set. Every reason to stop
456 // this track is therefore settled before the flush at the end.
457 gMC->TrackMomentum(fMom);
458 if ((fMom.E() - fMom.M()) < EMax) {
459 gMC->StopTrack();
460 return;
461 }
462
463 TParticle* p = gMC->GetStack()->GetCurrentTrack();
464 Int_t currentID = gMC->GetStack()->GetCurrentTrackNumber();
465
466 Int_t pdgCode = p->GetPdgCode();
467
468 // record statistics for neutrinos, electrons and photons
469 // add pi0 111 eta 221 eta' 331 omega 223
470 Int_t idabs = TMath::Abs(pdgCode);
471 if (idabs < 18 || idabs == 22 || idabs == 111 || idabs == 221 ||
472 idabs == 223 || idabs == 331 || idabs == 211 || idabs == 321 ||
473 idabs == 2212) {
474 Double_t wspill = p->GetWeight();
475 Int_t idhnu = idabs + 10000;
476 if (pdgCode < 0) {
477 idhnu += 10000;
478 }
479 Double_t l10ptot =
480 TMath::Min(TMath::Max(TMath::Log10(fMom.P()), -0.3), 1.69999);
481 Double_t l10pt =
482 TMath::Min(TMath::Max(TMath::Log10(fMom.Pt()), -2.), 0.4999);
483 TString key = fVetoName;
484 key += idhnu;
485 TH1D* h1 = dynamic_cast<TH1D*>(fout->Get(key));
486 if (h1) {
487 h1->Fill(fMom.P(), wspill);
488 }
489 key = fVetoName;
490 key += idhnu + 1000;
491 TH2D* h2 = dynamic_cast<TH2D*>(fout->Get(key));
492 if (h2) {
493 h2->Fill(l10ptot, l10pt, wspill);
494 }
495 key = fVetoName;
496 key += idhnu + 2000;
497 h2 = dynamic_cast<TH2D*>(fout->Get(key));
498 if (h2) {
499 h2->Fill(l10ptot, l10pt, wspill);
500 }
501 if (withNtuple) {
502 fNtuple->Fill(pdgCode, fMom.Px(), fMom.Py(), fMom.Pz(), fPos.X(),
503 fPos.Y(), fPos.Z());
504 }
505 if (fSkipNeutrinos && (idabs == 12 || idabs == 14 || idabs == 16)) {
506 // The statistics above are still recorded, but the track is stopped
507 // before its first step, so it must not receive the clone buffer.
508 gMC->StopTrack();
509 return;
510 }
511 }
512
513 // A module whose PreTrack() ran before ours may already have stopped the
514 // track: FairMCApplication calls the detectors in registration order, and
515 // run_fixedTarget.py registers the other exitHadronAbsorber planes ahead of
516 // the one that owns the split buffer. A module registered after this one is
517 // not covered: if it stops the track, the clones flushed below are lost.
518 if (!gMC->IsTrackAlive()) {
519 return;
520 }
521
522 if (!fSecondaryBuffer.empty()) {
523 auto* stack = dynamic_cast<ShipStack*>(gMC->GetStack());
524 Int_t ntr;
525 for (const auto& trk : fSecondaryBuffer) {
526 stack->PushTrack(1, trk.parentID, trk.pdg, trk.px, trk.py, trk.pz, trk.e,
527 trk.x, trk.y, trk.z, trk.t, trk.polx, trk.poly, trk.polz,
528 kPNoProcess, ntr, trk.weight, 999);
529 fCloneTracks.insert(ntr);
530 }
531 // Clear the buffer so we don't duplicate them for the next track
532 fSecondaryBuffer.clear();
533 }
534
535 if (fCloneTracks.find(currentID) != fCloneTracks.end()) {
536 // Force the decay time to 0
537 gMC->ForceDecayTime(0);
538 }
539}

569void RegisterDaughtersRecursively(TGeoVolume* volume,
570 exitHadronAbsorber* detector) {
571 if (!volume) return;
572 Int_t nDaughters = volume->GetNdaughters();
573 for (Int_t i = 0; i < nDaughters; ++i) {
574 TGeoNode* daughterNode = volume->GetNode(i);
575 if (daughterNode) {
576 TGeoVolume* daughterVol = daughterNode->GetVolume();
577 if (daughterVol) {
578 // register daughter volume
579 detector->AddSensitiveVolume(daughterVol);
580 LOG(info) << "[exitHadronAbsorber] Registered subvolume: "
581 << daughterVol->GetName();
582 // call function recursively
583 RegisterDaughtersRecursively(daughterVol, detector);
584 }
585 }
586 }

589void exitHadronAbsorber::ConstructGeometry() {
590 static FairGeoLoader* geoLoad = FairGeoLoader::Instance();
591 static FairGeoInterface* geoFace = geoLoad->getGeoInterface();
592 static FairGeoMedia* media = geoFace->getMedia();
593 static FairGeoBuilder* geoBuild = geoLoad->getGeoBuilder();
594
595 FairGeoMedium* ShipMedium = media->getMedium("vacuums");
596 TGeoMedium* vac = gGeoManager->GetMedium("vacuums");
597 if (vac == nullptr) geoBuild->createMedium(ShipMedium);
598 vac = gGeoManager->GetMedium("vacuums");
599 gGeoManager->GetTopVolume();
600 TGeoNavigator* nav = gGeoManager->GetCurrentNavigator();
601 Double_t zLoc;
602 if (fzPos > 1E8) {
603 // Add thin sensitive plane after hadron absorber
604 Float_t distance = 1.;
605 Double_t local[3] = {0, 0, 0};
606 if (!nav->cd("/MuonShieldArea_1/AbsorberVol_1")) {
607 nav->cd("/MuonShieldArea_1/MagnAbsorb_MagRetL_1");
608 distance = -1.;
609 }
610 TGeoBBox* tmp =
611 dynamic_cast<TGeoBBox*>(nav->GetCurrentNode()->GetVolume()->GetShape());
612 local[2] = distance * tmp->GetDZ();
613 Double_t global[3] = {0, 0, 0};
614 nav->LocalToMaster(local, global);
615 zLoc = global[2] + distance * 1. * cm;
616 } else {
617 zLoc = fzPos;
618 } // use external input
619
620 TString myname(this->GetName());
621 TString shapename_prefix(myname); // Use a prefix for shapenames
622
623 Double_t xLocPlane = 0.0;
624 Double_t yLocPlane = 0.0;
625 Double_t zLocPlane = zLoc;
626
627 if (!fCylindricalPlane) {
