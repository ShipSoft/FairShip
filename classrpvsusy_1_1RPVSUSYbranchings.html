267 def Width_H_L(self, H, L):
268 """
269 Returns the RPV neutralino decay width into neutral meson and lepton
270
271 Inputs:
272 - H is a string (name of the meson)
273 - L is a string (name of the lepton)
274 """
275 if (mass(H) + mass(L)) > self.MN:
276 return 0.0
277
278 phsp = math.sqrt(
279 self.MN**4
280 + mass(H) ** 4
281 + mass(L) ** 4
282 - 2 * self.MN**2 * mass(H) ** 2
283 - 2 * self.MN**2 * mass(L) ** 2
284 - 2 * mass(H) ** 2 * mass(L) ** 2
285 )
286 tmp_width = 0
287 if "nu_mu" in L or "nu_e" in L or "nu_tau" in L:
288 tmp_width = (
289 phsp
290 / (self.sfmass**4 * 128 * u.pi * self.MN**3)
291 * c.GST2["sneutrino"]
292 * c.decayConstant[H] ** 2
293 * (self.MN**2 + mass(L) ** 2 - mass(H) ** 2)
294 )
295 else:
296 tmp_width = (
297 phsp
298 / (self.sfmass**4 * 128 * u.pi * self.MN**3)
299 * c.GST2["slepton"]
300 * c.decayConstant[H] ** 2
301 * (self.MN**2 + mass(L) ** 2 - mass(H) ** 2)
302 )
303
304 if "K*" in H or "D*" in H or "phi" in H:
305 if "nu_mu" in L or "nu_e" in L or "nu_tau" in L:
306 tmp_width = (
307 phsp
308 / (self.sfmass**4 * 2 * u.pi * self.MN**3)
309 * c.GST2["tneutrino"]
310 * c.decayConstant[H] ** 2
311 * (2 * (self.MN**2 - mass(L) ** 2) ** 2 - mass(H) ** 2 * (mass(H) ** 2 + self.MN**2 + mass(L) ** 2))
312 )
313 else:
314 tmp_width = (
315 phsp
316 / (self.sfmass**4 * 2 * u.pi * self.MN**3)
317 * c.GST2["tlepton"]
318 * c.decayConstant[H] ** 2
319 * (2 * (self.MN**2 - mass(L) ** 2) ** 2 - mass(H) ** 2 * (mass(H) ** 2 + self.MN**2 + mass(L) ** 2))
320 )
321 width = self.U2[1] * tmp_width
322 # contributions both from decay and production couplings
323 if self.bench == 1 and ("K_" in H or "K*" in H):
324 width = (self.U2[0] + self.U2[1]) * tmp_width
325 # contributions only from production coupling
326 if self.bench == 2 and ("eta" in H or "phi" in H):
327 width = self.U2[0] * tmp_width
328
329 # Majorana case (charge conjugate channels)
330 width = 2.0 * width
331 return width
