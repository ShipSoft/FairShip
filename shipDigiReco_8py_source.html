196 def reconstructActs(self) -> None:
197 """Fit tracks and vertices with ACTS and persist them.
198
199 Fills the RecoTracks/RecoVertices/Particles branches, the MC link
200 and Tracklets, and the SBT veto links.
201 """
202 self.fACTSArray.clear()
203 self.fACTSVertexArray.clear()
204 self.fitTrack2MC.clear()
205 self.goodTracksVect.clear()
206 self.fPartArray.clear()
207 self.fTrackletsArray.clear()
208
209 tracks, vertices, track_hit_indices = self.actsTracks()
210
211 for track, hit_indices in zip(tracks, track_hit_indices, strict=True):
212 self.fACTSArray.push_back(track)
213
214 nmeas = track.nDoF()
215 if nmeas > 0:
216 global_variables.h["nmeas"].Fill(nmeas)
217 global_variables.h["chi2"].Fill(track.chi2() / nmeas)
218
219 # Save MC link: majority MC track over the hits used in the fit
220 _frac, tmax = self.fracMCsame([self.strawHits[i].track_id for i in hit_indices])
221 self.fitTrack2MC.push_back(tmax)
222
223 # Save digi-hit indices of the fitted track
224 indices_vector = ROOT.std.vector("unsigned int")()
225 for index in hit_indices:
226 indices_vector.push_back(self.strawHitToDigi[index])
227 self.fTrackletsArray.push_back(ROOT.Tracklet(1, indices_vector))
228
229 n_good_tracks = self.findGoodActsTracks()
230
231 for vtx in vertices:
232 self.fACTSVertexArray.push_back(vtx)
233
234 # Create a ShipParticle from the vertex fit from 2 track vertices
235 if len(vtx.trackIds()) != 2:
236 logger.debug("Skipping vertex with %d tracks (expected 2)", len(vtx.trackIds()))
237 continue
238 t1, t2 = vtx.trackIds()
239 # The daughters are taken to be muons, as in the GenFit vertexing
240 P = ROOT.TLorentzVector()
241 covP = np.zeros((4, 4))
242 for i in range(2):
243 p = np.array([vtx.trackPx()[i], vtx.trackPy()[i], vtx.trackPz()[i]])
244 energy = math.hypot(np.linalg.norm(p), MUON_MASS)
245 P += ROOT.TLorentzVector(float(p[0]), float(p[1]), float(p[2]), energy)
246 cov = np.zeros((3, 3))
247 cov[np.triu_indices(3)] = list(vtx.GetTrackMomentumCovariance(i))
248 cov = cov + np.triu(cov, 1).T
249 # (px, py, pz, E) from (px, py, pz)
250 jacobian = np.vstack([np.eye(3), p / energy])
251 covP += jacobian @ cov @ jacobian.T
252 vertex_position_4d = ROOT.TLorentzVector(vtx.x(), vtx.y(), vtx.z(), 0.0)
253 particle = ROOT.ShipParticle(9900015, 0, -1, -1, t1, t2, P, vertex_position_4d)
254
255 # ShipParticle::SetCovP takes the upper triangle of the 4x4 (px, py,
256 # pz, E) covariance, row by row.
257 particle.SetCovP(list(covP[np.triu_indices(4)]))
258 particle.SetCovV(list(vtx.GetCovarianceElements()))
259 particle.SetDoca(vtx.doca())
260 self.fPartArray.push_back(particle)
261
262 if hasattr(self, "digiSBT"):
263 # As in the GenFit linkVetoOnTracks, the veto links are built for the
264 # selected tracks only: entry k of the VetoHitOnTrack branch belongs
265 # to track goodTracks[k], and its hitID indexes the SBT digi hits.
266 veto_results = calculateSBTDOCA(self.actsTracker, self.digiSBT.det, list(self.goodTracksVect))
267 self.vetoHitOnTrackArray.clear()
268 for hitID, distMin in veto_results:
269 self.vetoHitOnTrackArray.push_back(ROOT.vetoHitOnTrack(hitID, float(distMin)))
270 if self.validation and hitID >= 0: # Only record real matches
271 self.validation_stats["veto_link_distance_sum"] += distMin
272 self.validation_stats["veto_link_distance_sum_sq"] += distMin * distMin
273 self.validation_stats["veto_link_distance_count"] += 1
274
275 n_tracks = len(self.fACTSArray)
276 if self.validation:
277 self.validation_stats["events_reconstructed"] += 1
278 self.validation_stats["fitted_tracks_total"] += n_tracks
279 self.validation_stats["good_tracks_total"] += n_good_tracks
280 if n_tracks > 0:
281 self.validation_stats["events_with_fitted_tracks"] += 1
282 if n_good_tracks > 0:
283 self.validation_stats["events_with_good_tracks"] += 1
284 validation_tools.record_event_stat(self.validation_stats, "event_fitted_tracks", n_tracks)
285 validation_tools.record_event_stat(self.validation_stats, "event_good_tracks", n_good_tracks)
286 if hasattr(self, "digiSBT"):
287 self.validation_stats["veto_links_total"] += len(self.vetoHitOnTrackArray)
288 validation_tools.record_event_stat(
289 self.validation_stats, "event_veto_links", len(self.vetoHitOnTrackArray)
290 )

292 def actsTracks(self) -> tuple:
293 """Collect this event's straw hits and fit track candidates with ACTS.
294
295 Returns (fitted tracks, vertices, per-track hit indices into
296 self.strawHits).
297 """
298 if global_variables.patRec in ("Truth", "Segments"):
299 # Truth hit assignment, or the ACTS tracker's own pattern
300 # recognition, need only the smeared hits
301 candidates = None
302 if global_variables.withT0:
303 self.SmearedHits = self.strawtubes.withT0Estimate()
304 else:
305 self.SmearedHits = self.strawtubes.smearHits(global_variables.withNoStrawSmearing)
306 else:
307 candidates = self.findTracks()
308
309 # Collect the hits from this event's smeared hits (single smearing,
310 # done above or inside findTracks).
311 self.strawHits = []
312 self.strawHitToDigi = []
313 digi_to_straw = {}
314 for k, sm in enumerate(self.SmearedHits):
315 digiHit = int(sm["digiHit"])
316 digi = self.strawtubes.det[digiHit]
317 point = self.sTree.strawtubesPoint[digiHit]
318 self.strawHits.append(
319 StrawHit(
320 detector_id=digi.GetDetectorID(),
321 station=digi.GetStationNumber(),
322 view=digi.GetViewNumber(),
323 layer=digi.GetLayerNumber(),
324 track_id=point.GetTrackID(),
325 z=sm["z"],
326 drift=sm["dist"],
327 )
328 )
329 self.strawHitToDigi.append(digiHit)
330 digi_to_straw[digiHit] = k
331
332 if global_variables.patRec == "Truth":
333 candidates = self._truthCandidates()
334 elif candidates is not None:
335 # Candidate hit indices refer to digi hits; remap them to positions
336 # in self.strawHits as expected by runTracking.
337 for cand in candidates:
338 cand["indices"] = [digi_to_straw[i] for i in cand["indices"] if i in digi_to_straw]
339
340 return runTracking(
341 self.actsTracker,
342 candidates,
343 self.strawHits,
344 self.sigma_spatial,
345 fit_vertex=global_variables.vertexing,
346 )

348 def _truthCandidates(self) -> list[dict]:
349 """Build track candidates from MC truth: the straw hits of each charged MC track.
350
351 Only the hit assignment comes from the truth; the fit starts from the
352 hits alone, as for any other pattern recognition.
353 """
354 hits_by_track = defaultdict(list)
355 for i, h in enumerate(self.strawHits):
356 hits_by_track[h.track_id].append(i)
357 candidates = []
358 for trID, indices in sorted(hits_by_track.items()):
359 particle = self.PDG.GetParticle(self.sTree.MCTrack[trID].GetPdgCode()) if trID >= 0 else None
360 # PyROOT returns a null proxy, not None, for an unknown PDG code
361 if not particle or particle.Charge() == 0:
362 continue
363 candidates.append({"indices": indices})
364 return candidates

388 def findTracks(self) -> list[dict]:
389 """Digitize, run pattern recognition and build fitter-agnostic track candidates.
390
391 Each candidate holds the seed state, hit measurements and indices;
392 fitting is done separately (e.g. by fitTracksGenfit).
393 """
394 hitPosLists = {}
395 hit_detector_ids = {}
396 stationCrossed: dict[int, dict[int, int]] = {}
397 listOfIndices: dict[int, list[int]] = {}
398 trackParams: dict[int, dict] = {}
399
400 #
401 if global_variables.withT0:
402 self.SmearedHits = self.strawtubes.withT0Estimate()
403 # old procedure, not including estimation of t0
404 else:
405 self.SmearedHits = self.strawtubes.smearHits(global_variables.withNoStrawSmearing)
406 if self.validation:
407 self.validation_stats["smeared_hits_total"] += len(self.SmearedHits)
408 if len(self.SmearedHits) > 0:
409 self.validation_stats["events_with_hits"] += 1
410
411 # Do pattern recognition
412 track_hits = shipPatRec.execute(self.SmearedHits, global_variables.ShipGeo, global_variables.patRec)
413 logger.debug("PatRec returned %d track candidates", len(track_hits))
414 if self.validation:
415 self.validation_stats["track_candidates_total"] += len(track_hits)
416 if len(track_hits) > 0:
417 self.validation_stats["events_with_candidates"] += 1
418 validation_tools.record_event_stat(self.validation_stats, "event_track_candidates", len(track_hits))
419 # Create hitPosLists for track fit
420 for i_track in track_hits:
421 atrack = track_hits[i_track]
422 atrack_y12 = atrack["y12"]
423 atrack_stereo12 = atrack["stereo12"]
424 atrack_y34 = atrack["y34"]
425 atrack_stereo34 = atrack["stereo34"]
426 atrack_smeared_hits = list(atrack_y12) + list(atrack_stereo12) + list(atrack_y34) + list(atrack_stereo34)
427 # Store PatRec track parameters for seeding the fitter
428 trackParams[i_track] = {
429 "k_y12": atrack.get("k_y12"),
430 "b_y12": atrack.get("b_y12"),
431 "k_y34": atrack.get("k_y34"),
432 "b_y34": atrack.get("b_y34"),
433 }
434 for sm in atrack_smeared_hits:
435 detID = sm["detID"]
436 station = self.strawtubes.det[sm["digiHit"]].GetStationNumber()
437 trID = i_track
438 # Collect hits for track fit
439 if trID not in hitPosLists:
440 hitPosLists[trID] = ROOT.std.vector("TVectorD")()
441 listOfIndices[trID] = []
442 stationCrossed[trID] = {}
443 hit_detector_ids[trID] = ROOT.std.vector("int")()
444 hit_detector_ids[trID].push_back(detID)
445 m = array("d", [sm["xtop"], sm["ytop"], sm["z"], sm["xbot"], sm["ybot"], sm["z"], sm["dist"]])
446 hitPosLists[trID].push_back(ROOT.TVectorD(7, m))
447 listOfIndices[trID].append(sm["digiHit"])
448 if station not in stationCrossed[trID]:
449 stationCrossed[trID][station] = 0
450 stationCrossed[trID][station] += 1
451
452 n_too_few_hits = 0
453 n_too_few_stations = 0
454 track_candidates = []
455
456 for atrack in hitPosLists:
457 if atrack < 0:
458 continue # these are hits not assigned to MC track because low E cut
459 # Determine charge sign from bending between stations 1-2 and 3-4.
460 # The slope difference dk = k_y34 - k_y12 encodes the charge:
461 # dk > 0 → negative charge (mu-, pi-), dk <= 0 → positive charge (mu+, pi+)
462 params = trackParams.get(atrack, {})
463 k_y12 = params.get("k_y12")
464 k_y34 = params.get("k_y34")
465 if k_y12 is not None and k_y34 is not None:
466 # The slope difference between the two station pairs gives the
467 # bending sense, and with it the charge.
468 pdg = 13 if k_y34 > k_y12 else -13
469 else:
470 pdg = 13
471 # GenFit is seeded with the PDG code, ACTS with the charge, so the two
472 # have to agree: mu- (pdg 13) carries charge -1, mu+ (pdg -13) carries +1.
473 charge = -1 if pdg > 0 else 1
474 meas = hitPosLists[atrack]
475 detIDs = hit_detector_ids[atrack]
476 nM = len(meas)
477 n_stations_crossed = len(stationCrossed[atrack])
478 if self.validation:
479 self.validation_stats["candidate_hits_sum"] += nM
480 self.validation_stats["candidate_hits_sum_sq"] += nM * nM
481 self.validation_stats["candidate_hits_count"] += 1
482 self.validation_stats["candidate_stations_sum"] += n_stations_crossed
483 self.validation_stats["candidate_stations_sum_sq"] += n_stations_crossed * n_stations_crossed
484 self.validation_stats["candidate_stations_count"] += 1
485 if nM < MIN_HITS_PER_TRACK:
486 n_too_few_hits += 1
487 continue # not enough hits for a good trackfit
488 if n_stations_crossed < MIN_STATIONS_CROSSED:
489 n_too_few_stations += 1
490 continue # not enough stations crossed to make a good trackfit
491 if global_variables.debug:
492 self.sTree.MCTrack[atrack]
493
494 # Seed state: use PatRec track parameters when available
495 posM, momM = self._compute_seed_state(atrack, meas, trackParams)
496
497 track_candidates.append(
498 {
499 "pos": posM,
500 "mom": momM,
501 "pdg": pdg,
502 "charge": charge,
503 "meas": meas,
504 "detIDs": detIDs,
505 "indices": listOfIndices[atrack],
506 }
507 )
508
509 logger.debug(
510 "findTracks: %d candidates, %d too few hits, %d too few stations, %d candidates kept for fitting",
511 len(hitPosLists),
512 n_too_few_hits,
513 n_too_few_stations,
514 len(track_candidates),
515 )
516 if self.validation:
517 self.validation_stats["track_candidates_rejected_hits"] += n_too_few_hits
518 self.validation_stats["track_candidates_rejected_stations"] += n_too_few_stations
519
520 return track_candidates

522 def fitTracksGenfit(self, candidates) -> int:
523 """Fit track candidates with GenFit and persist the fitted tracks."""
524 self.fGenFitArray.clear()
525 self.fTrackletsArray.clear()
526 self.fitTrack2MC.clear()
527
528 fittedTracks = []
529 for cand in candidates:
530 pdg = cand["pdg"]
531 meas = cand["meas"]
532 detIDs = cand["detIDs"]
533 nM = len(meas)
534 posM, momM = cand["pos"], cand["mom"]
535
536 # Try both charge hypotheses, keep the one with better chi2/NDF
537 best_track = None
538 best_chi2ndf = float("inf")
539 for try_pdg in [pdg, -pdg]:
540 if self.validation:
541 self.validation_stats["fit_hypotheses_tried"] += 1
542 # approximate covariance
543 covM = ROOT.TMatrixDSym(6)
544 resolution = self.sigma_spatial
545 if global_variables.withT0:
546 resolution *= 1.4 # worse resolution due to t0 estimate
547 for i in range(3):
548 covM[i][i] = resolution * resolution
549 # x is only measured by the small-angle stereo views, so give
550 # the seed a ~10x larger x uncertainty than y
551 covM[0][0] = resolution * resolution * 100.0
552 for i in range(3, 6):
553 covM[i][i] = ROOT.TMath.Power(resolution / nM / ROOT.TMath.Sqrt(3), 2)
554 rep = ROOT.genfit.RKTrackRep(try_pdg)
555 stateSmeared = ROOT.genfit.MeasuredStateOnPlane(rep)
556 rep.setPosMomCov(stateSmeared, posM, momM, covM)
557 seedState = ROOT.TVectorD(6)
558 seedCov = ROOT.TMatrixDSym(6)
559 rep.get6DStateCov(stateSmeared, seedState, seedCov)
560 theTrack = ROOT.genfit.Track(rep, seedState, seedCov)
561 ROOT.SetOwnership(rep, False) # genfit::Track takes ownership
562 hitCov = ROOT.TMatrixDSym(7)
563 hitCov[6][6] = resolution * resolution
564 hitID = 0
565 for m, detID in zip(meas, detIDs, strict=True):
566 tp = ROOT.genfit.TrackPoint(theTrack)
567 measurement = ROOT.genfit.WireMeasurement(m, hitCov, detID, hitID, tp)
568 measurement.setMaxDistance(
569 global_variables.ShipGeo.strawtubes_geo.outer_straw_diameter / 2.0
570 - global_variables.ShipGeo.strawtubes_geo.wall_thickness
571 )
572 tp.addRawMeasurement(measurement)
573 ROOT.SetOwnership(measurement, False) # TrackPoint takes ownership
574 theTrack.insertPoint(tp)
575 ROOT.SetOwnership(tp, False) # genfit::Track takes ownership
576 hitID += 1
577 # Fit this hypothesis
578 try:
579 theTrack.checkConsistency()
580 except ROOT.genfit.Exception:
581 if self.validation:
582 self.validation_stats["tracks_failed_consistency"] += 1
583 continue
584 try:
585 self.fitter.processTrack(theTrack)
586 except Exception as e:
587 logger.debug("Failed to processTrack for hypothesis %d: %s", try_pdg, e)
588 if self.validation:
589 self.validation_stats["tracks_failed_fit"] += 1
590 continue
591 try:
592 theTrack.checkConsistency()
593 except ROOT.genfit.Exception:
594 logger.debug("Track inconsistent after fit for hypothesis %d", try_pdg)
595 if self.validation:
596 self.validation_stats["tracks_failed_consistency"] += 1
597 continue
598 try:
599 fittedState = theTrack.getFittedState()
600 fittedState.getMomMag()
601 except Exception as e:
602 logger.debug("Failed to getFittedState/getMomMag for hypothesis %d: %s", try_pdg, e)
603 if self.validation:
604 self.validation_stats["tracks_failed_state_access"] += 1
605 continue
606 fitStatus = theTrack.getFitStatus()
607 if not fitStatus.isFitConverged():
608 continue
609 if self.validation:
610 self.validation_stats["fit_hypotheses_converged"] += 1
611 nmeas = fitStatus.getNdf()
612 if nmeas <= 0:
613 continue
614 chi2ndf = fitStatus.getChi2() / nmeas
615 if chi2ndf < best_chi2ndf:
616 best_chi2ndf = chi2ndf
617 best_track = theTrack
618 if best_track is not None:
619 fittedTracks.append((best_track, cand))
620
621 for theTrack, cand in fittedTracks:
622 # Tracks are already fitted from the dual-hypothesis loop above
623 fitStatus = theTrack.getFitStatus()
624 nmeas = fitStatus.getNdf() # guaranteed > 0 by hypothesis loop filter
625 global_variables.h["nmeas"].Fill(nmeas)
626 chi2 = fitStatus.getChi2() / nmeas
627 if self.validation:
628 self.validation_stats["chi2_sum"] += chi2
629 self.validation_stats["chi2_sum_sq"] += chi2 * chi2
630 self.validation_stats["chi2_count"] += 1
631 self.validation_stats["ndf_sum"] += nmeas
632 self.validation_stats["ndf_sum_sq"] += nmeas * nmeas
633 self.validation_stats["ndf_count"] += 1
634 global_variables.h["chi2"].Fill(chi2)
635 # make track persistent
636 # Store pointer - make a copy and let ROOT manage lifetime
637 trackCopy = ROOT.genfit.Track(theTrack)
638 ROOT.SetOwnership(trackCopy, False) # ROOT TTree owns the track
639 self.fGenFitArray.push_back(trackCopy)
640 if global_variables.debug:
641 print("save track", theTrack, chi2, nmeas, fitStatus.isFitConverged())
642 # Save MC link
643 track_ids = []
644 for index in cand["indices"]:
645 ahit = self.sTree.strawtubesPoint[index]
646 track_ids += [ahit.GetTrackID()]
647 _frac, tmax = self.fracMCsame(track_ids)
648 self.fitTrack2MC.push_back(tmax)
649 # Save hits indexes of the the fitted tracks
650 indices = ROOT.std.vector("unsigned int")()
651 for index in cand["indices"]:
652 indices.push_back(index)
653 aTracklet = ROOT.Tracklet(1, indices)
654 self.fTrackletsArray.push_back(aTracklet)
655
656 logger.debug("fitTracksGenfit: %d fitted tracks saved", len(self.fGenFitArray))
657
658 # debug
659 if global_variables.debug:
660 print("save tracklets:")
661 for x in self.recoTree.Tracklets:
662 print(x.getType(), len(x.getList()))
663 return len(self.fGenFitArray)

665 def _compute_seed_state(self, atrack, meas, trackParams):
666 """Compute seed position and momentum for the track fitter.
667
668 When PatRec track parameters (k_y, b_y) are available, use them
669 to place the seed at the first hit's Z with the correct Y position
670 and momentum direction. Prefers station 1-2 parameters, falls back
671 to station 3-4, then to a geometry-free default at the first hit.
672 """
673 params = trackParams.get(atrack)
674 k_y = None
675 b_y = None
676 if params:
677 if params.get("k_y12") is not None and params.get("b_y12") is not None:
678 k_y = params["k_y12"]
679 b_y = params["b_y12"]
680 elif params.get("k_y34") is not None and params.get("b_y34") is not None:
681 k_y = params["k_y34"]
682 b_y = params["b_y34"]
683
684 z_seed = meas[0][2] # z is the 3rd element of the TVectorD
685 if k_y is not None:
686 y_seed = k_y * z_seed + b_y
687 posM = ROOT.TVector3(0, y_seed, z_seed)
688 p_total = 3.0 * u.GeV
689 pz = p_total / ROOT.TMath.Sqrt(1.0 + k_y * k_y)
690 py = k_y * pz
691 momM = ROOT.TVector3(0, py, pz)
692 logger.debug(
693 "seed from PatRec: z=%.1f y=%.1f k_y=%.4f p=(0, %.2f, %.2f)",
694 z_seed,
695 y_seed,
696 k_y,
697 py,
698 pz,
699 )
700 else:
701 posM = ROOT.TVector3(0, 0, z_seed)
702 momM = ROOT.TVector3(0, 0, 3.0 * u.GeV)
703 return posM, momM

719 def findGoodActsTracks(self) -> int:
720 """ACTS counterpart of findGoodTracks.
721
722 Applies the same chi2/ndf < 50 criterion as the GenFit selection, so the
723 goodTracks branch means the same thing whichever fitter produced the

739 def findVetoHitOnTrack(self, track):
740 distMin = 99999.0
741 hitID = -1
742 xx = track.getFittedState()
743 rep = ROOT.genfit.RKTrackRep(xx.getPDG())
744 state = ROOT.genfit.StateOnPlane(rep)
745 rep.setPosMom(state, xx.getPos(), xx.getMom())
746 for i, vetoHit in enumerate(self.digiSBT.det):
747 vetoHitPos = vetoHit.GetXYZ()
748 try:
749 rep.extrapolateToPoint(state, vetoHitPos, False)
750 except Exception:
751 error = "shipDigiReco::findVetoHitOnTrack extrapolation did not worked"
752 ut.reportError(error)
753 if global_variables.debug:
754 print(error)
755 continue
756 dist = (rep.getPos(state) - vetoHitPos).Mag()
757 if dist < distMin:
758 distMin = dist
759 hitID = i
760 return ROOT.vetoHitOnTrack(hitID, distMin)
