32def _cov_element(cov, row, col):
33 """Read one element of an ACTS vertex-track covariance.
34
35 The binding hands back a numpy array, but an unfitted track has no
36 covariance at all. Missing or unindexable data reads as zero.
37 """
38 if cov is None:
39 return 0.0
40 try:
41 return cov[row, col]
42 except (TypeError, IndexError, KeyError):
43 pass
44 try:
45 return cov[row][col]
46 except (TypeError, IndexError, KeyError):
47 return 0.0

50def _track_vector(track, name, fallback):
51 """Return an ACTS vertex-track 3-vector as a numpy array.
52
53 Falls back to ``fallback`` when the preferred attribute is unset, and to
54 the origin when neither is available.
55 """
56 value = getattr(track, name, None)
57 if value is None:
58 value = getattr(track, fallback, None)
59 if value is None:
60 value = (0.0, 0.0, 0.0)
61 return np.array(value, dtype=float)

64def _extrapolate_to_x(point, direction, x_target):
65 """Propagate a straight line to a plane of constant x, or None if parallel."""
66 dx = direction[0]
67 if abs(dx) < 1e-12:
68 return None
69 return point + direction * ((x_target - point[0]) / dx)

237 def reconstructActs(self) -> None:
238 """Fit tracks and vertices with ACTS and persist them.
239
240 Fills the RecoTracks/RecoVertices/Particles branches, the MC link
241 and Tracklets, and the SBT veto links.
242 """
243 import acts
244 import acts.examples
245 from strawReco import calculateSBTDOCA
246
247 geo_ctx = acts.GeometryContext()
248 self.fACTSArray.clear()
249 self.fACTSVertexArray.clear()
250 self.fitTrack2MC.clear()
251 self.goodTracksVect.clear()
252 self.fPartArray.clear()
253 self.fTrackletsArray.clear()
254
255 output_tracks, vertices, track_hit_indices = self.actsTracks()
256
257 vector_ptr = ROOT.addressof(self.fACTSArray)
258 vertex_vector_ptr = ROOT.addressof(self.fACTSVertexArray)
259
260 for i, (_, hit_indices) in enumerate(zip(output_tracks, track_hit_indices, strict=True)):
261 acts.examples.pushRecoTrack(vector_ptr, geo_ctx, i, output_tracks)
262
263 nmeas = output_tracks.ndf[i]
264 if nmeas > 0:
265 global_variables.h["nmeas"].Fill(nmeas)
266 global_variables.h["chi2"].Fill(output_tracks.chi2[i] / nmeas)
267
268 # Save MC link: majority MC track over the hits used in the fit
269 track_ids = []
270 for index in hit_indices:
271 if 0 <= index < len(self.strawHits):
272 ahit = self.strawHits[index]
273 track_ids.append(int(ahit[5]))
274
275 _frac, tmax = self.fracMCsame(track_ids)
276 self.fitTrack2MC.push_back(tmax)
277
278 # Save digi-hit indices of the fitted track
279 indices_vector = ROOT.std.vector("unsigned int")()
280 for index in hit_indices:
281 if 0 <= index < len(self.strawHitToDigi):
282 indices_vector.push_back(self.strawHitToDigi[index])
283
284 aTracklet = ROOT.Tracklet(1, indices_vector)
285 self.fTrackletsArray.push_back(aTracklet)
286
287 n_good_tracks = self.findGoodActsTracks(output_tracks)
288
289 # acts-ship only exposes a process-global "last extracted params"
290 # pointer, so it can be attributed to a vertex unambiguously only when
291 # the event holds a single one.
292 input_addr = 0
293 if len(vertices) == 1 and hasattr(acts, "get_last_extracted_params_addr"):
294 try:
295 input_addr = int(acts.get_last_extracted_params_addr())
296 except (TypeError, ValueError):
297 logger.warning("Could not read the ACTS extracted-parameters address", exc_info=True)
298
299 for vtx in vertices:
300 acts.pushRecoVertex(vertex_vector_ptr, vtx, output_tracks, input_addr)
301
302 vtx_tracks = vtx.tracks()
303
304 # Create a ShipParticle from the vertex fit from 2 track vertices
305 if len(vtx_tracks) != 2:
306 logger.debug("Skipping vertex with %d tracks (expected 2)", len(vtx_tracks))
307 continue
308
309 vertex_pos = vtx.position()
310
311 # ACTS reco frame is in mm, the SHiP frame in cm
312 length_scale = 0.1 # mm -> cm
313 area_scale = 0.01 # mm^2 -> cm^2 (covariances)
314
315 # Scale units to cm and rotate
316 vx = ROOT.TVector3(
317 -vertex_pos[2] * length_scale, vertex_pos[1] * length_scale, vertex_pos[0] * length_scale
318 )
319
320 t1 = int(vtx_tracks[0].trackIndex)
321 t2 = int(vtx_tracks[1].trackIndex)
322
323 mom_daughter1 = tuple(vtx_tracks[0].momentum) if vtx_tracks[0].momentum is not None else (0.0, 0.0, 0.0)
324 mom_daughter2 = tuple(vtx_tracks[1].momentum) if vtx_tracks[1].momentum is not None else (0.0, 0.0, 0.0)
325 px_mother = mom_daughter1[0] + mom_daughter2[0]
326 py_mother = mom_daughter1[1] + mom_daughter2[1]
327 pz_mother = mom_daughter1[2] + mom_daughter2[2]
328 # Rotate into SHiP frame
329 P = ROOT.TLorentzVector(-pz_mother, py_mother, px_mother, 0)
330
331 acts_covV = vtx.covariance() # 3x3 Eigen matrix in mm^2 (ACTS reco frame)
332 covV = ROOT.TMatrixDSym(3)
333
334 covV[0][0] = acts_covV[2, 2] * area_scale
335 covV[1][1] = acts_covV[1, 1] * area_scale
336 covV[2][2] = acts_covV[0, 0] * area_scale
337
338 covV[0][1] = -acts_covV[1, 2] * area_scale
339 covV[1][0] = covV[0][1]
340
341 covV[0][2] = -acts_covV[0, 2] * area_scale
342 covV[2][0] = covV[0][2]
343
344 covV[1][2] = acts_covV[0, 1] * area_scale
345 covV[2][1] = covV[1][2]
346
347 cov_daughter1 = getattr(vtx_tracks[0], "covariance", None)
348 cov_daughter2 = getattr(vtx_tracks[1], "covariance", None)
349
350 raw_covP = ROOT.TMatrixDSym(3)
351 for ii in range(3):
352 for jj in range(3):
353 raw_covP[ii][jj] = _cov_element(cov_daughter1, ii + 3, jj + 3) + _cov_element(
354 cov_daughter2, ii + 3, jj + 3
355 )
356
357 covP = ROOT.TMatrixDSym(3)
358 covP[0][0] = raw_covP[2][2]
359 covP[1][1] = raw_covP[1][1]
360 covP[2][2] = raw_covP[0][0]
361
362 covP[0][1] = -raw_covP[1][2]
363 covP[1][0] = covP[0][1]
364
365 covP[0][2] = -raw_covP[0][2]
366 covP[2][0] = covP[0][2]
367
368 covP[1][2] = raw_covP[0][1]
369 covP[2][1] = covP[1][2]
370
371 vertex_position_4d = ROOT.TLorentzVector(vx, 0.0)
372
373 particle = ROOT.ShipParticle(9900015, 0, -1, -1, t1, t2, P, vertex_position_4d)
374 covV_list = [covV[0][0], covV[0][1], covV[0][2], covV[1][1], covV[1][2], covV[2][2]]
375
376 # ShipParticle::SetCovP takes the upper triangle of the 4x4 (px, py,
377 # pz, E) covariance, in the order (0,0), (0,1), (0,2), (0,3), (1,1),
378 # (1,2), (1,3), (2,2), (2,3), (3,3). The ACTS vertex fit gives only
379 # the 3x3 momentum block, so the four energy entries stay zero.
380 covP_list = [covP[0][0], covP[0][1], covP[0][2], 0.0, covP[1][1], covP[1][2], 0.0, covP[2][2], 0.0, 0.0]
381
382 # Track extrapolation and DOCA calculation, preferring the
383 # pre-fit track state where the vertex fit kept one
384 p1 = _track_vector(vtx_tracks[0], "originalPosition", "position")
385 p2 = _track_vector(vtx_tracks[1], "originalPosition", "position")
386 d1 = _track_vector(vtx_tracks[0], "originalMomentum", "momentum")
387 d2 = _track_vector(vtx_tracks[1], "originalMomentum", "momentum")
388
389 # normalize direction
390 n1 = np.linalg.norm(d1)
391 n2 = np.linalg.norm(d2)
392 if not (n1 > 0 and n2 > 0):
393 # Without a direction for both daughters the DOCA below collapses
394 # to zero, i.e. a perfect vertex, which would sail through every
395 # signal selection. Drop the candidate instead, as the GenFit
396 # path does when the extrapolation to the fitted vertex fails.
397 logger.warning("Skipping ACTS vertex candidate: daughter track without momentum")
398 continue
399 d1 = d1 / n1
400 d2 = d2 / n2
401
402 # target plane x in ACTS frame (mm)
403 target_x = float(vtx.position()[0])
404
405 p1_x_tmp = _extrapolate_to_x(p1, d1, target_x)
406 p1_x = p1_x_tmp if p1_x_tmp is not None else p1
407 p2_x_tmp = _extrapolate_to_x(p2, d2, target_x)
408 p2_x = p2_x_tmp if p2_x_tmp is not None else p2
409 delta = p1_x - p2_x
410 n = np.cross(d1, d2)
411 n_norm = np.linalg.norm(n)
412 if n_norm > 1e-12:
413 doca_mm = abs(np.dot(delta, n)) / n_norm
414 else:
415 # Nearly parallel tracks: perpendicular component of delta
416 # (d1 is a unit vector, so no further normalisation is needed).
417 doca_mm = np.linalg.norm(np.cross(delta, d1))
418 doca = float(doca_mm) / 10.0 # mm -> cm
419
420 particle.SetCovP(covP_list)
421 particle.SetCovV(covV_list)
422 particle.SetDoca(doca)
423 self.fPartArray.push_back(particle)
424
425 if hasattr(self, "digiSBT"):
426 # As in the GenFit linkVetoOnTracks, the veto links are built for the
427 # selected tracks only: entry k of the VetoHitOnTrack branch belongs
428 # to track goodTracks[k], and its hitID indexes the SBT digi hits.
429 veto_results = calculateSBTDOCA(
430 output_tracks,
431 self.digiSBT.det,
432 self.trackingGeometry,
433 self.actsFieldMap,
434 selected_indices=list(self.goodTracksVect),
435 )
436 self.vetoHitOnTrackArray.clear()
437 for hitID, distMin in veto_results:
438 self.vetoHitOnTrackArray.push_back(ROOT.vetoHitOnTrack(hitID, float(distMin)))
439 if self.validation and hitID >= 0: # Only record real matches
440 self.validation_stats["veto_link_distance_sum"] += distMin
441 self.validation_stats["veto_link_distance_sum_sq"] += distMin * distMin
442 self.validation_stats["veto_link_distance_count"] += 1
443
444 n_tracks = len(self.fACTSArray)
445 if self.validation:
446 self.validation_stats["events_reconstructed"] += 1
447 self.validation_stats["fitted_tracks_total"] += n_tracks
448 self.validation_stats["good_tracks_total"] += n_good_tracks
449 if n_tracks > 0:
450 self.validation_stats["events_with_fitted_tracks"] += 1
451 if n_good_tracks > 0:
452 self.validation_stats["events_with_good_tracks"] += 1
453 validation_tools.record_event_stat(self.validation_stats, "event_fitted_tracks", n_tracks)
454 validation_tools.record_event_stat(self.validation_stats, "event_good_tracks", n_good_tracks)
455 if hasattr(self, "digiSBT"):
456 self.validation_stats["veto_links_total"] += len(self.vetoHitOnTrackArray)
457 validation_tools.record_event_stat(
458 self.validation_stats, "event_veto_links", len(self.vetoHitOnTrackArray)
459 )

461 def actsTracks(self) -> tuple:
462 """Build per-hit ACTS event data and fit track candidates with ACTS.
463
464 Returns (fitted track container, vertices, per-track hit indices into
465 self.strawHits).
466 """
467 from strawReco import runTracking
468
469 if global_variables.patRec == "Truth":
470 # MC truth seeding needs no pattern recognition, only smeared hits
471 candidates = None
472 if global_variables.withT0:
473 self.SmearedHits = self.strawtubes.withT0Estimate()
474 else:
475 self.SmearedHits = self.strawtubes.smearHits(global_variables.withNoStrawSmearing)
476 else:
477 candidates = self.findTracks()
478
479 # Build the flat per-hit vectors handed to ACTS from this event's
480 # smeared hits (single smearing, done above or inside findTracks).
481 # Structure of hit vector (detector [straw=0], station, layer, view,
482 # straw, track_id, x, y, z, t, E, drift, wire-xtop, ytop, xbot, ybot)
483 self.strawHits.clear()
484 self.strawHitToDigi = []
485 digi_to_straw = {}
486 for k, sm in enumerate(self.SmearedHits):
487 digiHit = int(sm["digiHit"])
488 station = self.strawtubes.det[digiHit].GetStationNumber()
489 view = self.strawtubes.det[digiHit].GetViewNumber()
490 layer = self.strawtubes.det[digiHit].GetLayerNumber()
491 straw = self.strawtubes.det[digiHit].GetStrawNumber()
492 time = self.strawtubes.det[digiHit].GetTDC()
493 drift = sm["dist"]
494 xtop = sm["xtop"]
495 ytop = sm["ytop"]
496 xbot = sm["xbot"]
497 ybot = sm["ybot"]
498 point = self.sTree.strawtubesPoint[digiHit]
499 trID = point.GetTrackID()
500 x = point.GetX()
501 y = point.GetY()
502 z = point.GetZ()
503 deltaE = point.GetEnergyLoss()
504
505 iHit = ROOT.std.vector("float")()
506 for value in (0, station, layer, view, straw, trID, x, y, z, time, deltaE, drift, xtop, ytop, xbot, ybot):
507 iHit.push_back(value)
508 self.strawHits.push_back(iHit)
509 self.strawHitToDigi.append(digiHit)
510 digi_to_straw[digiHit] = k
511
512 if candidates is None:
513 candidates = self._truthCandidates()
514 else:
515 # Candidate hit indices refer to digi hits; remap them to positions
516 # in self.strawHits as expected by runTracking.
517 for cand in candidates:
518 cand["indices"] = [digi_to_straw[i] for i in cand["indices"] if i in digi_to_straw]
519
520 return runTracking(
521 candidates,
522 self.trackingGeometry,
523 self.actsFieldMap,
524 self.strawHits,
525 fit_vertex=global_variables.vertexing,
526 )

528 def _truthCandidates(self) -> list[dict]:
529 """Build track candidates from MC truth: one per MC track with straw hits."""
530 candidates = []
531 # Bucket the straw hits by MC track in one pass; iHit[5] is the track ID
532 hits_by_track = defaultdict(list)
533 for i, h in enumerate(self.strawHits):
534 hits_by_track[int(h[5])].append(i)
535 for trID, tr in enumerate(self.sTree.MCTrack):
536 indices = hits_by_track.get(trID)
537 if not indices:
538 continue
539 unique_indices = []
540 seen_layers = set()
541 # Sort indices by Z-position to find the first straw hit; iHit[8] is Z
542 indices.sort(key=lambda idx: self.strawHits[idx][8])
543 for idx in indices:
544 h = self.strawHits[idx]
545 # Keep only one straw per unique station/layer/view combination:
546 # the Kalman filter can't handle more than 1 measurement per acts layer
547 layer_key = (int(h[1]), int(h[2]), int(h[3]))
548 if layer_key not in seen_layers:
549 unique_indices.append(idx)
550 seen_layers.add(layer_key)
551 first_hit = self.strawHits[unique_indices[0]]
552
553 # Seed position from the first straw hit truth (indices 6, 7, 8)
554 pos = ROOT.TVector3(first_hit[6], first_hit[7], first_hit[8])
555
556 # Momentum from MCTrack
557 mom = ROOT.TVector3(tr.GetPx(), tr.GetPy(), tr.GetPz())
558
559 particle = self.PDG.GetParticle(tr.GetPdgCode())
560 if particle is None:
561 logger.warning("Skipping MCTrack %d with unknown PDG code %d", trID, tr.GetPdgCode())
562 continue
563 charge = particle.Charge() / 3.0
564 # A neutral seed has q/p = 0, and strawReco.make_seed divides by the
565 # momentum magnitude to get its q/p uncertainty. Neither case can be
566 # seeded; neutrals only reach here via the odd stray straw hit.
567 if charge == 0.0 or mom.Mag() <= 0:
568 logger.debug("Skipping unseedable MCTrack %d (pdg %d)", trID, tr.GetPdgCode())
569 continue
570
571 candidates.append({"pos": pos, "mom": mom, "indices": unique_indices, "charge": charge})
572 return candidates

594 def findTracks(self) -> list[dict]:
595 """Digitize, run pattern recognition and build fitter-agnostic track candidates.
596
597 Each candidate holds the seed state, hit measurements and indices;
598 fitting is done separately (e.g. by fitTracksGenfit).
599 """
600 hitPosLists = {}
601 hit_detector_ids = {}
602 stationCrossed: dict[int, dict[int, int]] = {}
603 listOfIndices: dict[int, list[int]] = {}
604 trackParams: dict[int, dict] = {}
605
606 #
607 if global_variables.withT0:
608 self.SmearedHits = self.strawtubes.withT0Estimate()
609 # old procedure, not including estimation of t0
610 else:
611 self.SmearedHits = self.strawtubes.smearHits(global_variables.withNoStrawSmearing)
612 if self.validation:
613 self.validation_stats["smeared_hits_total"] += len(self.SmearedHits)
614 if len(self.SmearedHits) > 0:
615 self.validation_stats["events_with_hits"] += 1
616
617 # Do pattern recognition
618 track_hits = shipPatRec.execute(self.SmearedHits, global_variables.ShipGeo, global_variables.patRec)
619 logger.debug("PatRec returned %d track candidates", len(track_hits))
620 if self.validation:
621 self.validation_stats["track_candidates_total"] += len(track_hits)
622 if len(track_hits) > 0:
623 self.validation_stats["events_with_candidates"] += 1
624 validation_tools.record_event_stat(self.validation_stats, "event_track_candidates", len(track_hits))
625 # Create hitPosLists for track fit
626 for i_track in track_hits:
627 atrack = track_hits[i_track]
628 atrack_y12 = atrack["y12"]
629 atrack_stereo12 = atrack["stereo12"]
630 atrack_y34 = atrack["y34"]
631 atrack_stereo34 = atrack["stereo34"]
632 atrack_smeared_hits = list(atrack_y12) + list(atrack_stereo12) + list(atrack_y34) + list(atrack_stereo34)
633 # Store PatRec track parameters for seeding the fitter
634 trackParams[i_track] = {
635 "k_y12": atrack.get("k_y12"),
636 "b_y12": atrack.get("b_y12"),
637 "k_y34": atrack.get("k_y34"),
638 "b_y34": atrack.get("b_y34"),
639 }
640 for sm in atrack_smeared_hits:
641 detID = sm["detID"]
642 station = self.strawtubes.det[sm["digiHit"]].GetStationNumber()
643 trID = i_track
644 # Collect hits for track fit
645 if trID not in hitPosLists:
646 hitPosLists[trID] = ROOT.std.vector("TVectorD")()
647 listOfIndices[trID] = []
648 stationCrossed[trID] = {}
649 hit_detector_ids[trID] = ROOT.std.vector("int")()
650 hit_detector_ids[trID].push_back(detID)
651 m = array("d", [sm["xtop"], sm["ytop"], sm["z"], sm["xbot"], sm["ybot"], sm["z"], sm["dist"]])
652 hitPosLists[trID].push_back(ROOT.TVectorD(7, m))
653 listOfIndices[trID].append(sm["digiHit"])
654 if station not in stationCrossed[trID]:
655 stationCrossed[trID][station] = 0
656 stationCrossed[trID][station] += 1
657
658 n_too_few_hits = 0
659 n_too_few_stations = 0
660 track_candidates = []
661
662 for atrack in hitPosLists:
663 if atrack < 0:
664 continue # these are hits not assigned to MC track because low E cut
665 # Determine charge sign from bending between stations 1-2 and 3-4.
666 # The slope difference dk = k_y34 - k_y12 encodes the charge:
667 # dk > 0 → negative charge (mu-, pi-), dk <= 0 → positive charge (mu+, pi+)
668 params = trackParams.get(atrack, {})
669 k_y12 = params.get("k_y12")
670 k_y34 = params.get("k_y34")
671 if k_y12 is not None and k_y34 is not None:
672 # The slope difference between the two station pairs gives the
673 # bending sense, and with it the charge.
674 pdg = 13 if k_y34 > k_y12 else -13
675 else:
676 pdg = 13
677 # GenFit is seeded with the PDG code, ACTS with the charge, so the two
678 # have to agree: mu- (pdg 13) carries charge -1, mu+ (pdg -13) carries +1.
679 charge = -1 if pdg > 0 else 1
680 meas = hitPosLists[atrack]
681 detIDs = hit_detector_ids[atrack]
682 nM = len(meas)
683 n_stations_crossed = len(stationCrossed[atrack])
684 if self.validation:
685 self.validation_stats["candidate_hits_sum"] += nM
686 self.validation_stats["candidate_hits_sum_sq"] += nM * nM
687 self.validation_stats["candidate_hits_count"] += 1
688 self.validation_stats["candidate_stations_sum"] += n_stations_crossed
689 self.validation_stats["candidate_stations_sum_sq"] += n_stations_crossed * n_stations_crossed
690 self.validation_stats["candidate_stations_count"] += 1
691 if nM < MIN_HITS_PER_TRACK:
692 n_too_few_hits += 1
693 continue # not enough hits for a good trackfit
694 if n_stations_crossed < MIN_STATIONS_CROSSED:
695 n_too_few_stations += 1
696 continue # not enough stations crossed to make a good trackfit
697 if global_variables.debug:
698 self.sTree.MCTrack[atrack]
699
700 # Seed state: use PatRec track parameters when available
701 posM, momM = self._compute_seed_state(atrack, meas, trackParams)
702
703 track_candidates.append(
704 {
705 "pos": posM,
706 "mom": momM,
707 "pdg": pdg,
708 "charge": charge,
709 "meas": meas,
710 "detIDs": detIDs,
711 "indices": listOfIndices[atrack],
712 }
713 )
714
715 logger.debug(
716 "findTracks: %d candidates, %d too few hits, %d too few stations, %d candidates kept for fitting",
717 len(hitPosLists),
718 n_too_few_hits,
719 n_too_few_stations,
720 len(track_candidates),
721 )
722 if self.validation:
723 self.validation_stats["track_candidates_rejected_hits"] += n_too_few_hits
724 self.validation_stats["track_candidates_rejected_stations"] += n_too_few_stations
725
726 return track_candidates

728 def fitTracksGenfit(self, candidates) -> int:
729 """Fit track candidates with GenFit and persist the fitted tracks."""
730 self.fGenFitArray.clear()
731 self.fTrackletsArray.clear()
732 self.fitTrack2MC.clear()
733
734 fittedTracks = []
735 for cand in candidates:
736 pdg = cand["pdg"]
737 meas = cand["meas"]
738 detIDs = cand["detIDs"]
739 nM = len(meas)
740 posM, momM = cand["pos"], cand["mom"]
741
742 # Try both charge hypotheses, keep the one with better chi2/NDF
743 best_track = None
744 best_chi2ndf = float("inf")
745 for try_pdg in [pdg, -pdg]:
746 if self.validation:
747 self.validation_stats["fit_hypotheses_tried"] += 1
748 # approximate covariance
749 covM = ROOT.TMatrixDSym(6)
750 resolution = self.sigma_spatial
751 if global_variables.withT0:
752 resolution *= 1.4 # worse resolution due to t0 estimate
753 for i in range(3):
754 covM[i][i] = resolution * resolution
755 # x is only measured by the small-angle stereo views, so give
756 # the seed a ~10x larger x uncertainty than y
757 covM[0][0] = resolution * resolution * 100.0
758 for i in range(3, 6):
759 covM[i][i] = ROOT.TMath.Power(resolution / nM / ROOT.TMath.Sqrt(3), 2)
760 rep = ROOT.genfit.RKTrackRep(try_pdg)
761 stateSmeared = ROOT.genfit.MeasuredStateOnPlane(rep)
762 rep.setPosMomCov(stateSmeared, posM, momM, covM)
763 seedState = ROOT.TVectorD(6)
764 seedCov = ROOT.TMatrixDSym(6)
765 rep.get6DStateCov(stateSmeared, seedState, seedCov)
766 theTrack = ROOT.genfit.Track(rep, seedState, seedCov)
767 ROOT.SetOwnership(rep, False) # genfit::Track takes ownership
768 hitCov = ROOT.TMatrixDSym(7)
769 hitCov[6][6] = resolution * resolution
770 hitID = 0
771 for m, detID in zip(meas, detIDs, strict=True):
772 tp = ROOT.genfit.TrackPoint(theTrack)
773 measurement = ROOT.genfit.WireMeasurement(m, hitCov, detID, hitID, tp)
774 measurement.setMaxDistance(
775 global_variables.ShipGeo.strawtubes_geo.outer_straw_diameter / 2.0
776 - global_variables.ShipGeo.strawtubes_geo.wall_thickness
777 )
778 tp.addRawMeasurement(measurement)
779 ROOT.SetOwnership(measurement, False) # TrackPoint takes ownership
780 theTrack.insertPoint(tp)
781 ROOT.SetOwnership(tp, False) # genfit::Track takes ownership
782 hitID += 1
783 # Fit this hypothesis
784 try:
785 theTrack.checkConsistency()
786 except ROOT.genfit.Exception:
787 if self.validation:
788 self.validation_stats["tracks_failed_consistency"] += 1
789 continue
790 try:
791 self.fitter.processTrack(theTrack)
792 except Exception as e:
793 logger.debug("Failed to processTrack for hypothesis %d: %s", try_pdg, e)
794 if self.validation:
795 self.validation_stats["tracks_failed_fit"] += 1
796 continue
797 try:
798 theTrack.checkConsistency()
799 except ROOT.genfit.Exception:
800 logger.debug("Track inconsistent after fit for hypothesis %d", try_pdg)
801 if self.validation:
802 self.validation_stats["tracks_failed_consistency"] += 1
803 continue
804 try:
805 fittedState = theTrack.getFittedState()
806 fittedState.getMomMag()
807 except Exception as e:
808 logger.debug("Failed to getFittedState/getMomMag for hypothesis %d: %s", try_pdg, e)
809 if self.validation:
810 self.validation_stats["tracks_failed_state_access"] += 1
811 continue
812 fitStatus = theTrack.getFitStatus()
813 if not fitStatus.isFitConverged():
814 continue
815 if self.validation:
816 self.validation_stats["fit_hypotheses_converged"] += 1
817 nmeas = fitStatus.getNdf()
818 if nmeas <= 0:
819 continue
820 chi2ndf = fitStatus.getChi2() / nmeas
821 if chi2ndf < best_chi2ndf:
822 best_chi2ndf = chi2ndf
823 best_track = theTrack
824 if best_track is not None:
825 fittedTracks.append((best_track, cand))
826
827 for theTrack, cand in fittedTracks:
828 # Tracks are already fitted from the dual-hypothesis loop above
829 fitStatus = theTrack.getFitStatus()
830 nmeas = fitStatus.getNdf() # guaranteed > 0 by hypothesis loop filter
831 global_variables.h["nmeas"].Fill(nmeas)
832 chi2 = fitStatus.getChi2() / nmeas
833 if self.validation:
834 self.validation_stats["chi2_sum"] += chi2
835 self.validation_stats["chi2_sum_sq"] += chi2 * chi2
836 self.validation_stats["chi2_count"] += 1
837 self.validation_stats["ndf_sum"] += nmeas
838 self.validation_stats["ndf_sum_sq"] += nmeas * nmeas
839 self.validation_stats["ndf_count"] += 1
840 global_variables.h["chi2"].Fill(chi2)
841 # make track persistent
842 # Store pointer - make a copy and let ROOT manage lifetime
843 trackCopy = ROOT.genfit.Track(theTrack)
844 ROOT.SetOwnership(trackCopy, False) # ROOT TTree owns the track
845 self.fGenFitArray.push_back(trackCopy)
846 if global_variables.debug:
847 print("save track", theTrack, chi2, nmeas, fitStatus.isFitConverged())
848 # Save MC link
849 track_ids = []
850 for index in cand["indices"]:
851 ahit = self.sTree.strawtubesPoint[index]
852 track_ids += [ahit.GetTrackID()]
853 _frac, tmax = self.fracMCsame(track_ids)
854 self.fitTrack2MC.push_back(tmax)
855 # Save hits indexes of the the fitted tracks
856 indices = ROOT.std.vector("unsigned int")()
857 for index in cand["indices"]:
858 indices.push_back(index)
859 aTracklet = ROOT.Tracklet(1, indices)
860 self.fTrackletsArray.push_back(aTracklet)
861
862 logger.debug("fitTracksGenfit: %d fitted tracks saved", len(self.fGenFitArray))
863
864 # debug
865 if global_variables.debug:
866 print("save tracklets:")
867 for x in self.recoTree.Tracklets:
868 print(x.getType(), len(x.getList()))
869 return len(self.fGenFitArray)

871 def _compute_seed_state(self, atrack, meas, trackParams):
872 """Compute seed position and momentum for the track fitter.
873
874 When PatRec track parameters (k_y, b_y) are available, use them
875 to place the seed at the first hit's Z with the correct Y position
876 and momentum direction. Prefers station 1-2 parameters, falls back
877 to station 3-4, then to a geometry-free default at the first hit.
878 """
879 params = trackParams.get(atrack)
880 k_y = None
881 b_y = None
882 if params:
883 if params.get("k_y12") is not None and params.get("b_y12") is not None:
884 k_y = params["k_y12"]
885 b_y = params["b_y12"]
886 elif params.get("k_y34") is not None and params.get("b_y34") is not None:
887 k_y = params["k_y34"]
888 b_y = params["b_y34"]
889
890 z_seed = meas[0][2] # z is the 3rd element of the TVectorD
891 if k_y is not None:
892 y_seed = k_y * z_seed + b_y
893 posM = ROOT.TVector3(0, y_seed, z_seed)
894 p_total = 3.0 * u.GeV
895 pz = p_total / ROOT.TMath.Sqrt(1.0 + k_y * k_y)
896 py = k_y * pz
897 momM = ROOT.TVector3(0, py, pz)
898 logger.debug(
899 "seed from PatRec: z=%.1f y=%.1f k_y=%.4f p=(0, %.2f, %.2f)",
900 z_seed,
901 y_seed,
902 k_y,
903 py,
904 pz,
905 )
906 else:
907 posM = ROOT.TVector3(0, 0, z_seed)
908 momM = ROOT.TVector3(0, 0, 3.0 * u.GeV)
909 return posM, momM
