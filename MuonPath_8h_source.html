14/*
15This class allows to consider a track with changes in directions between the
16different points of measurements (start, veto detectors, SST, etc...).
17
18A MuonPath can be made of several slices. For each slice, there is a start and
19an end position.
20
21When getting a random z position along the path, the extrapolation is made
22linearly using a vertex position which corresponds to the relevant point of
23measurement along the trajectory.
24*/
25
26const Double_t c_light = 29.9792458; // speed of light in cm/ns
27const Double_t muon_mass = 0.10565999895334244; // muon mass in GeV

136 inline double GetX(const double& aZ, const unsigned& idx) const {
137 if (idx >= GetNSlices() || fpvec[idx].Z() == 0) return 0;
138 return fvtx[idx].X() +
139 (aZ - fvtx[idx].Z()) * fpvec[idx].X() / fpvec[idx].Z();
140 };
141
142 inline double GetY(const double& aZ, const unsigned& idx) const {
143 if (idx >= GetNSlices() || fpvec[idx].Z() == 0) return 0;
144 return fvtx[idx].Y() +
145 (aZ - fvtx[idx].Z()) * fpvec[idx].Y() / fpvec[idx].Z();
146 };
147
148 inline double GetLength(const double& aZ, const unsigned& idx) const {
149 if (idx >= GetNSlices()) return 0;
150 return TMath::Sqrt(TMath::Power(GetX(aZ, idx) - fvtx[idx].X(), 2) +
151 TMath::Power(GetY(aZ, idx) - fvtx[idx].Y(), 2) +
152 TMath::Power(aZ - fvtx[idx].Z(), 2)); // in cm
153 };
154
155 inline double GetTimeNs(const double& aZ, const unsigned& idx) const {
156 if (idx >= GetNSlices() || fpvec[idx].Z() == 0.) return 0;
157 double P = fpvec[idx].Mag();
158 if (P == 0.) return 0;
159 double v = c_light * P /
160 TMath::Sqrt(TMath::Power(P, 2) + TMath::Power(muon_mass, 2));
161 if (v == 0.) return 0;
162 // A slice may precede its reference measurement in z.
163 return fvtxT[idx] + (aZ - fvtx[idx].Z()) * P / fpvec[idx].Z() / v;
164 };

166 private:

181 std::vector<double>
182 fendLength; // cumulative segment lengths, excluding jumps
183 std::vector<TVector3> fpvec;
184};
