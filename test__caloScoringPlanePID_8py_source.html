4"""Tests for reading the CaloScoringPlane PID file (python/caloScoringPlanePID.py).
5
6The numerical checks (rows summing to 1, energies increasing, ...) are done in
7C++ by CaloScoringPlanePID::Set() and tested in tests/test_calo_scoring_plane_pid.cxx;
8the tests at the end check that their errors reach Python with the file name.

37def test_rows_and_columns_are_reordered():
38 # file order muon, electron, hadron; P(mu -> mu) = 0.7, P(mu -> hadron) = 0.3
39 model = parse(
40 {
41 "species": ["muon", "electron", "hadron"],
42 "points": [{"energy": 1, "matrix": [[0.7, 0.0, 0.3], [0, 1, 0], [0, 0, 1]]}],
43 }
44 )
45 e, h, mu = 0, 1, 2
46 m = model.matrices
47 assert m[3 * mu + mu] == 0.7
48 assert m[3 * mu + h] == 0.3
49 assert m[3 * e + e] == 1.0
50 assert m[3 * h + h] == 1.0

76@pytest.mark.parametrize(
77 ("cfg", "message"),
78 [
79 ([1, 2], "top level must be a mapping"),

84 ({"points": []}, "'points' must be a non-empty list"),

87 ({"points": [{"energy": 1, "matrix": IDENTITY, "x": 2}]}, "points[0] must have exactly the keys"),

89 ({"points": [{"energy": 1, "matrix": [[1, 0, 0], [0, 1, 0]]}]}, "points[0].matrix must be 3 rows"),

91 ({"points": [{"energy": 1, "matrix": [[1, 0, 0], [0, "a", 0], [0, 0, 1]]}]}, "points[0].matrix[1][1]"),
92 ({"points": [{"energy": 1, "matrix": [[True, 0, 0], [0, 1, 0], [0, 0, 1]]}]}, "must be a number"),
93 (config(pdg_overrides=[22]), "'pdg_overrides' must map PDG codes"),
