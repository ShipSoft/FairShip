4"""ACTS-based straw tracker reconstruction utilities.
5
6Fit straw-tube track candidates and vertices with ACTS, and extrapolate
7fitted tracks to SBT hits for veto matching. Positions are converted from
8the SHiP frame (cm, z along the beam) to the ACTS reco frame (mm, rotated
9x -> -z) at the boundaries of this module.

30def make_seed(pos, mom, charge, surface, geo_ctx):
31 sigma_drift = 5.0 # 5mm uncertainty
32 sigma_long = 50.0 # 50mm uncertainty
33 sigma_phi = 0.05 # 50 mrad
34 sigma_theta = 0.05 # 50 mrad
35 sigma_qp = 0.2 / mom.Mag() # 20% relative error on q/p
36 sigma_time = 1e9 # deweight time parameter, as unused
37
38 cov = (
39 np.diag([sigma_drift**2, sigma_long**2, sigma_phi**2, sigma_theta**2, sigma_qp**2, sigma_time**2])
40 .flatten()
41 .tolist()
42 )
43
44 return acts.createTrackParameters(
45 pos.Z() * 10, pos.Y() * 10, pos.X() * -10, mom.Z(), mom.Y(), -mom.X(), charge, surface, cov, geo_ctx
46 )

49def runTracking(candidates, trackingGeometry, fieldMap, strawHits, fit_vertex=True):
50 """
51 Fit tracks in the Reco frame using truth, or patrec seeds,
52 fit vertices if tracks meet criteria
53
54 Mutates its arguments in place: each vector in ``strawHits`` gains the
55 drift-side sign as element 16 (appended only once, so repeated calls on the
56 same vectors are safe), and every ``cand["indices"]`` is sorted by hit z.
57 """
58
59 # Setup geo context
60 geo_ctx = acts.GeometryContext()
61
62 # Hacky way of solving left/right drift ambiguity issue,
63 # move to performing a refit, making use of the residuals
64 # rather than using true hit info
65 for hit_idx in range(len(strawHits)):
66 iHit = strawHits[hit_idx]
67
68 dx = iHit[14] - iHit[12] # xbot - xtop
69 dy = iHit[15] - iHit[13] # ybot - ytop
70
71 wire_x = (iHit[12] + iHit[14]) / 2.0
72 wire_y = (iHit[13] + iHit[15]) / 2.0
73 vec_to_hit_x = iHit[6] - wire_x
74 vec_to_hit_y = iHit[7] - wire_y
75
76 nx = -dy
77 ny = dx
78
79 dot_product = vec_to_hit_x * nx + vec_to_hit_y * ny
80
81 sign = 1.0 if dot_product > 0 else -1.0
82
83 # Append sign as index 16, unless a previous call already did so
84 if iHit.size() > 16:
85 iHit[16] = sign
86 else:
87 iHit.push_back(sign)
88
89 measurements = acts.processMeasurements(strawHits, trackingGeometry)
90 acts_index_map = build_acts_index_map(measurements, strawHits)
91 # Setup output containers
92 output_tracks = acts.examples.TrackContainer()
93
94 track_hit_indices_list = []
95
96 # Loop over seeds (e.g., from PatRec or Truth)
97 for cand in candidates:
98 cand["indices"].sort(key=lambda i: strawHits.at(i).at(8))
99 # (ACTS measurement index, straw-hit index) pairs: the first is what
100 # the fitter reads, the second indexes strawHits for the caller.
101 remapped_indices = align_candidate_indices(cand, acts_index_map)
102
103 filtered_indices = []
104 filtered_hit_indices = []
105 seen_layers = set()
106
107 for idx, hit_idx in remapped_indices:
108 gid = acts.getMeasurementGeoId(measurements, idx)
109
110 if gid.layer not in seen_layers:
111 filtered_indices.append(idx)
112 filtered_hit_indices.append(hit_idx)
113 seen_layers.add(gid.layer)
114
115 if len(filtered_indices) < MIN_HITS_PER_TRACK:
116 logger.debug("Skipping track with too few hits: %d", len(filtered_indices))
117 continue
118
119 first_idx = filtered_indices[0]
120
121 geo_id = acts.getMeasurementGeoId(measurements, first_idx)
122
123 target_surface = acts.getSurface(trackingGeometry, geo_id)
124 if not target_surface:
125 logger.warning("Could not find surface for GeoID: %s", geo_id)
126 continue
127
128 initial_params = make_seed(cand["pos"], cand["mom"], cand["charge"], target_surface, geo_ctx)
129
130 tracks_before_fit = len(output_tracks)
131
132 acts.fitTrack(measurements, filtered_indices, initial_params, output_tracks, trackingGeometry, fieldMap)
133
134 tracks_after_fit = len(output_tracks)
135
136 if tracks_after_fit > tracks_before_fit:
137 track_hit_indices_list.append(filtered_hit_indices)
138
139 const_tracks = output_tracks.makeConst()
140
141 vertices = []
142
143 good_proxies = []
144 for i, track in enumerate(const_tracks):
145 if not track.referenceSurface:
146 continue
147
148 track_chi2 = const_tracks.chi2[i]
149 track_ndf = const_tracks.ndf[i]
150
151 chi2_ndf = track_chi2 / track_ndf if track_ndf > 0 else float("inf")
152
153 if track.nMeasurements >= MIN_HITS_PER_TRACK and chi2_ndf <= MAX_CHI2_NDF:
154 good_proxies.append(track)
155
156 # Check for at least one positive and one negative track
157
158 has_pos = any(t.parameters[4] > 0 for t in good_proxies)
159 has_neg = any(t.parameters[4] < 0 for t in good_proxies)
160
161 if fit_vertex and MIN_VERTEX_TRACKS <= len(good_proxies) <= MAX_VERTEX_TRACKS and has_pos and has_neg:
162 vertices = acts.fitVertex(good_proxies, fieldMap, geo_ctx, trackingGeometry)
163
164 return const_tracks, vertices, track_hit_indices_list

167def calculateSBTDOCA(output_tracks, sbt_digis, trackingGeometry, fieldMap, selected_indices=None):
168 """
169 Extrapolates fitted tracks to the X-plane (Reco frame) of SBT hits.
170
171 Returns one ``(hitID, distMin)`` per selected track, in ascending
172 container order, so element ``k`` of the result corresponds to
173 ``selected_indices[k]``. ``selected_indices`` defaults to every track.
174 ``distMin`` is in centimetres (ROOT geometry units) to match the GenFit
175 ShipDigiReco.findVetoHitOnTrack; tracks with no match get
176 ``(-1, 99999.0)``, the same sentinel that path uses.
177 """
178
179 nav_cfg = acts.Navigator.Config()
180
181 nav_cfg.trackingGeometry = trackingGeometry
182
183 navigator = acts.Navigator(nav_cfg)
184
185 stepper = acts.EigenStepper(fieldMap)
186 propagator = acts.Propagator(stepper, navigator)
187
188 geo_ctx = acts.GeometryContext()
189 mag_ctx = acts.MagneticFieldContext()
190
191 event_parameters_array = output_tracks.parameters # shape (N, 6)
192 event_covariance_array = output_tracks.covariance # shape (N, 6, 6)
193
194 selected = None if selected_indices is None else {int(i) for i in selected_indices}
195
196 # The target planes depend only on the SBT hits, so build them once and
197 # reuse them for every track. s_idx keeps indexing sbt_digis.
198 normal_arr = np.array([1.0, 0.0, 0.0])
199 target_surfaces = [
200 (
201 s_idx,
202 acts.createPlaneSurface(
203 np.array([sbt_hit.GetZ() * 10.0, sbt_hit.GetY() * 10.0, -sbt_hit.GetX() * 10.0]), normal_arr
204 ),
205 )
206 for s_idx, sbt_hit in enumerate(sbt_digis)
207 ]
208
209 veto_results = []
210 for _t_idx, track in enumerate(output_tracks):
211 if selected is not None and _t_idx not in selected:
212 continue
213
214 distMin = 99999.0
215 hitID = -1
216
217 # A track without a reference surface cannot be extrapolated, but it
218 # still gets a placeholder so the results stay aligned with the
219 # selected tracks.
220 if track.referenceSurface:
221 track_params_numpy = event_parameters_array[_t_idx]
222 track_covariance_numpy = event_covariance_array[_t_idx]
223
224 start_params = acts.examples.makeBoundTrackParameters(
225 track.referenceSurface, track_params_numpy, track_covariance_numpy
226 )
227
228 for s_idx, target_surface in target_surfaces:
229 res_params = acts.extrapolateTrack(propagator, start_params, target_surface, geo_ctx, mag_ctx)
230
231 if res_params is not None:
232 # The target plane is centred on the SBT hit, so the local
233 # bound parameters give the in-plane distance to the hit
234 # directly (equivalent to the Y-Z distance on the X plane).
235 local_params = np.asarray(res_params.parameters)
236 doca = float(np.hypot(local_params[0], local_params[1])) / 10.0 # mm -> cm
237
238 if doca < distMin:
239 distMin = doca
240 hitID = s_idx
241
242 veto_results.append((hitID, distMin))
243
244 return veto_results

247def straw_geometry_id(hit):
248 """Return the ACTS GeometryIdentifier value of a straw hit.
249
250 This is the numbering of the acts-ship StrawtubeDetector geometry, as
251 SHiPMeasurementProvider assigns it: the whole spectrometer is volume 1,
252 and its layers are numbered consecutively with 16 per station, 4 per view
253 and 2 per layer, offset so that station 1 / view 0 / layer 0 is 2.
254 """
255 station, layer, view, straw = (int(hit[i]) for i in range(1, 5))
256 gid = acts.GeometryIdentifier()
257 gid.volume = 1
258 gid.layer = 16 * station + 4 * view + 2 * layer - 14
259 gid.sensitive = straw
260 return gid.value

263def match_hits_to_measurements(hit_gids, measurement_gids):
264 """Map straw-hit indices to ACTS measurement indices by geometry ID.
265
266 processMeasurements makes one measurement per straw hit, ordered by
267 surface and, within a surface, in hit order; it drops all hits of a
268 surface it cannot find. So the k-th hit on a straw belongs to the k-th
269 measurement on it. Hits without a measurement are left out.
270 """
271 measurements_by_gid = defaultdict(list)
272 for index, gid in enumerate(measurement_gids):
273 measurements_by_gid[gid].append(index)
274 hits_by_gid = defaultdict(list)
275 for index, gid in enumerate(hit_gids):
276 hits_by_gid[gid].append(index)
277 hit_to_measurement = {}
278 for gid, hits in hits_by_gid.items():
279 matched = measurements_by_gid.get(gid, [])
280 if matched and len(matched) != len(hits):
281 logger.warning(
282 "Straw with geometry ID %s has %d hits but %d ACTS measurements; skipping them",
283 gid,
284 len(hits),
285 len(matched),
286 )
287 continue
288 hit_to_measurement.update(zip(hits, matched, strict=False))
289 return hit_to_measurement

292def build_acts_index_map(measurements, strawHits):
293 """Map each straw-hit index to the index of its ACTS measurement."""
294 measurement_gids = []
295 # There is at most one measurement per hit; the container has no len().
296 for index in range(len(strawHits)):
297 try:
298 measurement_gids.append(acts.getMeasurementGeoId(measurements, index).value)
299 except IndexError:
300 break
301 return match_hits_to_measurements([straw_geometry_id(hit) for hit in strawHits], measurement_gids)

304def align_candidate_indices(cand, hit_to_measurement):
305 """Return (ACTS measurement index, straw-hit index) pairs for a candidate."""
306 return [(hit_to_measurement[index], index) for index in cand["indices"] if index in hit_to_measurement]
