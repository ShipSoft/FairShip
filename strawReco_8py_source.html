54def make_tracker(geometry, field):
55 """Create the ACTS straw tracker for the loaded geometry and field."""
56 if not hasattr(ROOT.SHiP, "ActsStrawTracker"):
57 raise RuntimeError("FairShip was built without ACTS, so the ACTS track fitter is not available")
58 tracker = ROOT.SHiP.ActsStrawTracker(geometry, field)
59 logger.info("ACTS straw tracker built with %d straws", tracker.GetNStraws())
60 return tracker

63def store_extrapolations(tracker, track_index, track, ubt_z=None, td_z=None):
64 """Store on ``track`` its path length and its extrapolations through the field.
65
66 The track is propagated (field only, no material) from its first fitted
67 measurement back to the UBT plane at ``ubt_z`` and from its last one on
68 to the timing-detector plane at ``td_z`` (cm). A plane given as None, or
69 one the propagation cannot reach, is not stored: ``RecoTrack`` then
70 reports it as missing (``HasUBTExtrapolation``/``HasTDExtrapolation``
71 false). The path length between the first and last measurement is NaN if
72 the tracker cannot give it.
73 """
74 track.SetTrackLength(tracker.TrackLength(track_index))
75 for z, from_last_hit, setter in (
76 (ubt_z, False, track.SetUBTExtrapolation),
77 (td_z, True, track.SetTDExtrapolation),
78 ):
79 if z is None:
80 continue
81 position, momentum = ROOT.TVector3(), ROOT.TVector3()
82 covariance = ROOT.std.vector("double")()
83 path_length = ctypes.c_double(float("nan"))
84 if tracker.ExtrapolateToZ(track_index, z, from_last_hit, position, momentum, covariance, path_length):
85 setter(position, momentum, covariance, path_length.value)
86 else:
87 logger.debug("Track %d does not reach z = %.1f cm", track_index, z)

90def runTracking(tracker, candidates, hits, resolution, fit_vertex=True, ubt_z=None, td_z=None):
91 """Find or fit tracks and, if the event qualifies, fit their vertex.
92
93 With ``candidates`` None, the tracker's own pattern recognition finds
94 the tracks. Otherwise ``candidates`` are dicts with ``indices`` into
95 ``hits`` (sorted by hit z in place), and each is fitted on its own.
96
97 Each track also stores its path length and its extrapolations to the UBT
98 plane at ``ubt_z`` and the timing-detector plane at ``td_z`` (cm); a
99 plane given as None, e.g. because that detector is not in the geometry,
100 is not stored (see ``store_extrapolations``).
101
102 Returns the fitted tracks (``SHiP::RecoTrack``, in the order of the
103 tracker's track indices), the fitted vertices (``SHiP::RecoVertex``) and,
104 per fitted track, the indices into ``hits`` that were used in the fit.
105 """
106 tracker.ClearEvent()
107 measurements = [tracker.AddMeasurement(h.detector_id, h.drift, resolution) for h in hits]
108 hit_of_measurement = {m: i for i, m in enumerate(measurements) if m >= 0}
109
110 if candidates is None:
111 tracker.FindTracks()
112 candidates = []
113 for cand in candidates:
114 cand["indices"].sort(key=lambda i: hits[i].z)
115
116 # The Kalman filter takes at most one measurement per straw layer
117 used = []
118 seen_layers = set()
119 for idx in cand["indices"]:
120 if measurements[idx] < 0:
121 logger.warning("No ACTS surface for straw %d", hits[idx].detector_id)
122 continue
123 if hits[idx].layer_key not in seen_layers:
124 used.append(idx)
125 seen_layers.add(hits[idx].layer_key)
126
127 if len(used) < MIN_HITS_PER_TRACK:
128 logger.debug("Skipping track with too few hits: %d", len(used))
129 continue
130
131 tracker.FitCandidate(ROOT.std.vector("int")([measurements[i] for i in used]))
132
133 tracks = [tracker.GetTrack(i) for i in range(tracker.GetNTracks())]
134 for i, track in enumerate(tracks):
135 store_extrapolations(tracker, i, track, ubt_z, td_z)
136 track_hit_indices = [
137 [hit_of_measurement[m] for m in tracker.GetTrackMeasurements(i)] for i in range(tracker.GetNTracks())
138 ]
139
140 good = [
141 i
142 for i, track in enumerate(tracks)
143 if track.nMeasurements() >= MIN_HITS_PER_TRACK
144 and track.nDoF() > 0
145 and track.chi2() / track.nDoF() <= MAX_CHI2_NDF
146 ]
147 # A two-track vertex for every opposite-charge pair of good tracks, as in
148 # the GenFit vertexing
149 vertices = []
150 if fit_vertex:
151 for i, j in combinations(good, 2):
152 if tracks[i].charge() * tracks[j].charge() >= 0:
153 continue
154 vertex = ROOT.SHiP.RecoVertex()
155 if tracker.FitVertex(ROOT.std.vector("int")([i, j]), vertex):
156 vertices.append(vertex)
157 else:
158 logger.debug("Vertex fit failed for tracks %d and %d", i, j)
159
160 return tracks, vertices, track_hit_indices

163def calculateSBTDOCA(tracker, sbt_digis, selected_indices):
164 """Match fitted tracks to SBT hits.
165
166 Each track is extrapolated to the plane of constant z through every SBT
167 hit. Returns one ``(hitID, distMin)`` per entry of ``selected_indices``,
168 with the distance in cm, to match the GenFit
169 ShipDigiReco.findVetoHitOnTrack; tracks with no match get
170 ``NO_VETO_MATCH``.
171 """
172 hits = [ROOT.TVector3(hit.GetX(), hit.GetY(), hit.GetZ()) for hit in sbt_digis]
173 veto_results = []
174 for track in selected_indices:
175 hitID, distMin = NO_VETO_MATCH
176 for s_idx, hit in enumerate(hits):
177 doca = tracker.DistanceToHit(int(track), hit)
178 if 0 <= doca < distMin:
179 hitID, distMin = s_idx, doca
180 veto_results.append((hitID, distMin))
181 return veto_results
