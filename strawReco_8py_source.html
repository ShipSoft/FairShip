57 def drift_sign(self) -> float:
58 """Side of the wire the track passed on, taken from the MC truth.
59
60 This resolves the left/right ambiguity with truth information. It
61 should be replaced by a refit that uses the residuals.
62 """
63 wire_x = (self.xtop + self.xbot) / 2.0
64 wire_y = (self.ytop + self.ybot) / 2.0
65 # Normal to the wire in the x-y plane
66 nx = -(self.ybot - self.ytop)
67 ny = self.xbot - self.xtop
68 return 1.0 if (self.x - wire_x) * nx + (self.y - wire_y) * ny > 0 else -1.0

71def make_tracker(geometry, field):
72 """Create the ACTS straw tracker for the loaded geometry and field."""
73 if not hasattr(ROOT.SHiP, "ActsStrawTracker"):
74 raise RuntimeError("FairShip was built without ACTS, so the ACTS track fitter is not available")
75 tracker = ROOT.SHiP.ActsStrawTracker(geometry, field)
76 logger.info("ACTS straw tracker built with %d straws", tracker.GetNStraws())
77 return tracker

80def runTracking(tracker, candidates, hits, resolution, fit_vertex=True):
81 """Fit track candidates and, if the event qualifies, their vertex.
82
83 ``candidates`` are dicts with ``pos``, ``mom`` (TVector3, SHiP frame),
84 ``charge`` and ``indices`` into ``hits``. Each candidate's indices are
85 sorted by hit z in place.
86
87 Returns the fitted tracks (``SHiP::RecoTrack``, in the order of the
88 tracker's track indices), the fitted vertices (``SHiP::RecoVertex``) and,
89 per fitted track, the indices into ``hits`` that were used in the fit.
90 """
91 tracker.ClearEvent()
92 measurements = [tracker.AddMeasurement(h.detector_id, h.drift_sign() * h.drift, resolution) for h in hits]
93
94 track_hit_indices = []
95 for cand in candidates:
96 cand["indices"].sort(key=lambda i: hits[i].z)
97
98 # The Kalman filter takes at most one measurement per straw layer
99 used = []
100 seen_layers = set()
101 for idx in cand["indices"]:
102 if measurements[idx] < 0:
103 logger.warning("No ACTS surface for straw %d", hits[idx].detector_id)
104 continue
105 if hits[idx].layer_key not in seen_layers:
106 used.append(idx)
107 seen_layers.add(hits[idx].layer_key)
108
109 if len(used) < MIN_HITS_PER_TRACK:
110 logger.debug("Skipping track with too few hits: %d", len(used))
111 continue
112
113 track_measurements = ROOT.std.vector("int")([measurements[i] for i in used])
114 track = tracker.FitTrack(track_measurements, cand["pos"], cand["mom"], round(cand["charge"]))
115 if track >= 0:
116 track_hit_indices.append(used)
117
118 tracks = [tracker.GetTrack(i) for i in range(tracker.GetNTracks())]
119
120 good = [
121 i
122 for i, track in enumerate(tracks)
123 if track.nMeasurements() >= MIN_HITS_PER_TRACK
124 and track.nDoF() > 0
125 and track.chi2() / track.nDoF() <= MAX_CHI2_NDF
126 ]
127 has_pos = any(tracks[i].charge() > 0 for i in good)
128 has_neg = any(tracks[i].charge() < 0 for i in good)
129
130 vertices = []
131 if fit_vertex and MIN_VERTEX_TRACKS <= len(good) <= MAX_VERTEX_TRACKS and has_pos and has_neg:
132 vertex = ROOT.SHiP.RecoVertex()
133 if tracker.FitVertex(ROOT.std.vector("int")(good), vertex):
134 vertices.append(vertex)
135 else:
136 logger.debug("Vertex fit failed for tracks %s", good)
137
138 return tracks, vertices, track_hit_indices

141def calculateSBTDOCA(tracker, sbt_digis, selected_indices):
142 """Match fitted tracks to SBT hits.
143
144 Each track is extrapolated to the plane of constant z through every SBT
145 hit. Returns one ``(hitID, distMin)`` per entry of ``selected_indices``,
146 with the distance in cm, to match the GenFit
147 ShipDigiReco.findVetoHitOnTrack; tracks with no match get
148 ``NO_VETO_MATCH``.
149 """
150 hits = [ROOT.TVector3(hit.GetX(), hit.GetY(), hit.GetZ()) for hit in sbt_digis]
151 veto_results = []
152 for track in selected_indices:
153 hitID, distMin = NO_VETO_MATCH
154 for s_idx, hit in enumerate(hits):
155 doca = tracker.DistanceToHit(int(track), hit)
156 if 0 <= doca < distMin:
157 hitID, distMin = s_idx, doca
158 veto_results.append((hitID, distMin))
159 return veto_results
