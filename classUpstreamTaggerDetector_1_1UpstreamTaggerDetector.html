141 def _tile_center(coordinate: float, grid_origin: float, grid_end: float, tile_size: float) -> float:
142 """Return the globally aligned tile center containing coordinate."""
143 coordinate = min(max(coordinate, grid_origin), math.nextafter(grid_end, grid_origin))
144 tile_count = round((grid_end - grid_origin) / tile_size)
145 if tile_count <= 0:
146 raise ValueError(f"Invalid UBT tile grid [{grid_origin}, {grid_end}] with tile size {tile_size}")
147 # At the positive boundary, floating-point subtraction and division
148 # can round nextafter(grid_end, grid_origin) back to an exact quotient
149 # of tile_count. Cap the result so the last valid tile is selected.
150 tile_index = min(math.floor((coordinate - grid_origin) / tile_size), tile_count - 1)
151 return grid_origin + (tile_index + 0.5) * tile_size

188 def adc_signal(self, point, tile_center: ROOT.TVector3, tile_size: float) -> float:
189 """Return the (unrounded) ADC signal of a point from its energy and position in the tile."""
190 # FairShip positions are in cm and the response-map coordinates are mm.
191 local_x_mm = 10.0 * (point.GetX() - tile_center.X())
192 local_y_mm = 10.0 * (point.GetY() - tile_center.Y())
193 response_map = self._adc_response_maps[tile_size]
194 # Clamp to the outermost bins so hits on a tile edge do not land in
195 # the under/overflow bins.
196 x_axis = response_map.GetXaxis()
197 y_axis = response_map.GetYaxis()
198 x_bin = min(max(x_axis.FindFixBin(local_x_mm), 1), x_axis.GetNbins())
199 y_bin = min(max(y_axis.FindFixBin(local_y_mm), 1), y_axis.GetNbins())
200 adc_per_mev = response_map.GetBinContent(x_bin, y_bin)
201 deposited_energy_mev = 1000.0 * point.GetEnergyLoss()

None UpstreamTaggerDetector.UpstreamTaggerDetector.digitize (   self) 
Digitize UBT MC points into one hit per fired constituent tile.

An MC point covers a (track, region) pair and a region holds 25 or 100
tiles, so a tile can collect points of several tracks, e.g. a muon and
a delta ray. Their ADC signals, each taken at its own position in the
tile, are summed before rounding and before the trigger threshold is
applied. The hit time is that of the first point in the tile.

Return a stable unique ID for a constituent tile.

The tile is identified by its lower-left cell on the tile-ID grid, the
grid of the smallest tile side (2 cm) over the tile grid bounds, with
``columns`` x ``rows`` cells:

    tile ID = (tile side / grid size - 1) * columns * rows + row * columns + column

So each tile side has its own range of ``columns * rows`` IDs, and a
2 cm and a 4 cm tile with the same lower-left cell get different IDs.
For the default geometry (340 x 504 cm2, 170 x 252 cells), the range
of 2 cm tiles is [0, 42840) and that of 4 cm tiles [42840, 85680). The
stride follows the UBT box, so decoding an ID needs the grid of its
geometry:

    side_index, cell = divmod(tile_id, columns * rows)
    row, column = divmod(cell, columns)

159 def tile_id(self, position: ROOT.TVector3, tile_size: float) -> int:
160 """Return a stable unique ID for a constituent tile.
161
162 The tile is identified by its lower-left cell on the tile-ID grid, the
163 grid of the smallest tile side (2 cm) over the tile grid bounds, with
164 ``columns`` x ``rows`` cells:
165
166 tile ID = (tile side / grid size - 1) * columns * rows + row * columns + column
167
168 So each tile side has its own range of ``columns * rows`` IDs, and a
169 2 cm and a 4 cm tile with the same lower-left cell get different IDs.
170 For the default geometry (340 x 504 cm2, 170 x 252 cells), the range
171 of 2 cm tiles is [0, 42840) and that of 4 cm tiles [42840, 85680). The
172 stride follows the UBT box, so decoding an ID needs the grid of its
173 geometry:
174
175 side_index, cell = divmod(tile_id, columns * rows)
176 row, column = divmod(cell, columns)
177 """
178 column = round((position.X() - tile_size / 2.0 - self._grid_x[0]) / self._grid_size)

183 f"maps to cell ({column}, {row}), outside the {self._columns} x {self._rows} tile-ID grid; "
184 f"x bounds={self._grid_x} cm, y bounds={self._grid_y} cm, base cell size={self._grid_size} cm"

Raise UBTGeometryMismatch if the UBT points of this geometry cannot be digitized.

The decision is taken from the simulated geometry, not from the
configuration, which configure replaces for old geometry files. A
geometry without any UBT region volume predates the scintillator UBT:
its UBT was a vacuum box, whose detector IDs are not region IDs.
Otherwise every region of the UBT map must have its volume, with the
same tile size.

78 def validate_stored_geometry(cls, geo_manager, ubt_geo) -> None:
79 """Raise UBTGeometryMismatch if the UBT points of this geometry cannot be digitized.
80
81 The decision is taken from the simulated geometry, not from the
82 configuration, which configure replaces for old geometry files. A
83 geometry without any UBT region volume predates the scintillator UBT:
84 its UBT was a vacuum box, whose detector IDs are not region IDs.
85 Otherwise every region of the UBT map must have its volume, with the
86 same tile size.
87 """

89 raise RuntimeError("Load the geometry of the simulation (FAIRGeom) before digitizing the UBT")

95 volumes[int(region)] = size
