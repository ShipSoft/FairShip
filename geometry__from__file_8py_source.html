4"""Simulate on the geometry of a geometry file (run_simScript.py -g).
5
6The modules build their geometry as usual, from the configuration (ShipGeo)
7stored in the geometry file. Building does more than create volumes: the
8detectors register which volumes are sensitive, and magnets attach their
9fields to volumes, and a geometry file keeps neither. So the run keeps all of
10that and only exchanges the volumes: a GeometrySwap module (make_swap), added last,
11replaces the geometry that was just built by the one stored in the file, before
12Geant4 converts it. Particles are then transported through the stored geometry,
13whatever the current code would build.
14
15For this to be safe, every volume and placement the current code builds must
16exist in the stored geometry (sensitive volumes are found by name, detector
17IDs come from node names and copy numbers), and the fields attached to volumes
18are copied onto the stored volumes. Fields from field maps
19are added after the swap, as usual.
20"""
21

37namespace ship_geofile {
38// Read the stored geometry without making it the current one. Reading a
39// TGeoManager deletes gGeoManager, but the geometry that was built must stay
40// alive: FairMCApplication still uses its list of volumes, and its fields are
41// copied. FairModule::ConstructRootGeometry() does the same.
42inline TGeoManager* Import(const char* path) {
43 TGeoManager* built = gGeoManager;
44 gGeoManager = nullptr;
45 TGeoManager* loaded = TGeoManager::Import(path, "FAIRGeom");
46 gGeoManager = built;
47 return loaded;
48}
49
50inline void Use(TGeoManager* geometry) { gGeoManager = geometry; }
51
52// Names of the volumes and of their placements ("volume/node").
53inline std::unordered_set<std::string> Placements(TGeoManager* geometry) {
54 std::unordered_set<std::string> names;
55 TIter next(geometry->GetListOfVolumes());
56 while (auto* volume = static_cast<TGeoVolume*>(next())) {
57 names.insert(volume->GetName());
58 for (Int_t i = 0; i < volume->GetNdaughters(); ++i) {
59 names.insert(std::string(volume->GetName()) + "/" + volume->GetNode(i)->GetName());
60 }
61 }
62 return names;
63}
64
65// Volumes and placements of `built` that `loaded` lacks.
66inline std::vector<std::string> Missing(TGeoManager* built, TGeoManager* loaded) {
67 const auto have = Placements(loaded);
68 std::vector<std::string> missing;
69 for (const auto& name : Placements(built)) {
70 if (!have.count(name)) missing.push_back(name);
71 }
72 return missing;
73}
74} // namespace ship_geofile
75"""

78def check_file(geofile: str) -> None:
79 """Raise ValueError unless geofile holds a geometry (FAIRGeom) and its configuration (ShipGeo)."""
80 try:
81 f = ROOT.TFile.Open(geofile, "READ")
82 except OSError: # recent PyROOT raises instead of returning a null file
83 f = None
84 if not f or f.IsZombie():
85 raise ValueError(f"cannot open geometry file {geofile}")
86 with f:
87 key = f.GetKey("FAIRGeom")
88 has_geometry = bool(key) and key.GetClassName() == "TGeoManager"
89 has_config = bool(f.GetKey("ShipGeo"))
90 if not has_geometry:
91 raise ValueError(f"{geofile} has no geometry FAIRGeom: give a geometry file written by run_simScript.py")
92 if not has_config:
93 raise ValueError(
94 f"{geofile} has no geometry configuration ShipGeo: give a geometry file written by run_simScript.py"
95 )

102def make_swap(geofile: str) -> Any:
103 """The module that replaces the built geometry by the one in geofile; add it last.
104
105 The class derives from pyFairModule, so it is defined on first use only:
106 this keeps the library of pyFairModule from being loaded whenever this
107 module is imported, as it is by shipDet_conf.
108 """
109 global _swap_class
110 if _swap_class is None:
111 _swap_class = _define_swap_class()
112 return _swap_class(geofile)
113
114
115def _define_swap_class() -> Any:
116 class GeometrySwap(ROOT.pyFairModule):
117 """Module that replaces the built geometry by the one in a geometry file; add it last."""
118
119 def __init__(self, geofile: str) -> None:
120 ROOT.pyFairModule.__init__(self, self)
121 self.SetName(SWAP_NAME)
122 self.geofile = geofile
123 self.loaded: Any = None
124 self.fields: list[str] = []
125 self.error: str | None = None
126
127 def ConstructGeometry(self) -> None:
128 # An exception raised here would pass through the C++ of Geant4 VMC,
129 # which calls this method. Record it instead: the built geometry then
130 # stays in use, and check_swapped() stops the run after run.Init().
131 try:
132 self._swap()
133 except Exception as e:
134 self.error = str(e)
135 print(f"ERROR in GeometrySwap: {e}", flush=True)
136
137 def _swap(self) -> None:
138 if not hasattr(ROOT, "ship_geofile"):
139 ROOT.gInterpreter.Declare(_HELPER)
140 helper = ROOT.ship_geofile # declared above
141 built = ROOT.gGeoManager
142 loaded = helper.Import(self.geofile)
143 if not loaded:
144 raise RuntimeError(f"cannot read the geometry FAIRGeom from {self.geofile}")
145
146 # Detectors find their sensitive volumes by name, and compute detector
147 # IDs from node names and copy numbers.
148 missing = sorted(str(name) for name in helper.Missing(built, loaded))
149 if missing:
150 raise RuntimeError(
151 f"the geometry in {self.geofile} lacks {len(missing)} volume(s) or placement(s) that the current"
152 f" code builds ({', '.join(missing[:10])}{', ...' if len(missing) > 10 else ''}); detectors"
153 " could record no hits or wrong detector IDs"
154 )
155
156 # Fields attached to volumes are not stored in geometry files.
157 built_volumes = list(built.GetListOfVolumes())
158 loaded_volumes = list(loaded.GetListOfVolumes())
159 same_order = [v.GetName() for v in built_volumes] == [v.GetName() for v in loaded_volumes]
160 by_name: dict[str, list[Any]] = {}
161 for volume in loaded_volumes:
162 by_name.setdefault(volume.GetName(), []).append(volume)
163 fields = []
164 for i, volume in enumerate(built_volumes):
165 field = volume.GetField()
166 if not field:
167 continue
168 if same_order:
169 target = loaded_volumes[i]
170 else:
171 candidates = by_name[volume.GetName()]
172 if len(candidates) != 1:
173 raise RuntimeError(
174 f"cannot tell which of the {len(candidates)} volumes {volume.GetName()} in {self.geofile}"
175 " gets the magnetic field of the volume of that name"
176 )
177 target = candidates[0]
178 fields.append((target, field, volume.GetName()))
179 for target, field, name in fields:
180 target.SetField(field)
181 self.fields.append(name)
182
183 helper.Use(loaded)
184 self.loaded = loaded
185 # The parameter file stores the geometry given to FairGeoParSet by FairRunSim::Init().
186 geo_par = ROOT.FairRuntimeDb.instance().getContainer("FairGeoParSet")
187 if geo_par:
188 geo_par.SetGeometry(loaded)
189 print(
190 f"GeometrySwap: transporting through the geometry of {self.geofile} ({len(loaded_volumes)} volumes);"
191 f" magnetic fields of volumes copied for: {', '.join(self.fields) or 'none'}",
192 flush=True,
193 )
194
195 return GeometrySwap

198def check_swapped(swap: Any) -> None:
199 """After run.Init(): stop unless the geometry from the file is the one in use."""
200 in_use = ROOT.gGeoManager
201 if swap.error or not swap.loaded or not in_use or ROOT.addressof(in_use) != ROOT.addressof(swap.loaded):
202 reason = swap.error or "GeometrySwap.ConstructGeometry() did not run or did not complete"
203 raise RuntimeError(f"the geometry from {swap.geofile} is not in use: {reason}")

218def stored_yaml_configs(ship_geo: Any) -> dict[str, Any]:
219 """The parts of the stored configuration that the YAML files will overwrite."""
220 return {key: ship_geo[key] for key in YAML_CONFIGS if key in ship_geo}

223def check_yaml_configs(stored: dict[str, Any], ship_geo: Any) -> None:
224 """Check that the YAML files still give the configuration stored with the geometry.
225
226 The detectors take their parameters from the YAML files, so if these changed
227 since the geometry file was written, the parameters would not match the
228 stored geometry.
229 """
230 changed = [YAML_CONFIGS[key] for key, value in stored.items() if dict(ship_geo.get(key, {})) != dict(value)]
231 if changed:
232 raise ValueError(
233 f"geometry/{', geometry/'.join(changed)} changed since the geometry file was written, so the detector"
234 " parameters would not match the stored geometry: write a new geometry file"
235 )
