338 // step drawn from the decay distribution restricted to the step,
339 // s = -lambda ln(1 - u P_decay), u uniform in [0, 1),
340 // interpolated between the pre- and post-step points. Putting every clone at
341 // the post-step point instead piles the decays of a long step (e.g. through
342 // the helium between target slabs) up at the next volume boundary.
343 //
344 // The momentum is only interpolated if the track survives the step. A step
345 // that ends the track (an inelastic interaction, its own decay) leaves the
346 // post-step point with the momentum after that process, zero for a killed
347 // track, so interpolating towards it would give the clones of every track's
348 // last step too little momentum. Up to the interaction the track only loses
349 // energy continuously, which is negligible over one step, so those clones get
350 // the pre-step momentum.

352 const G4ThreeVector xPre = pre->GetPosition() / CLHEP::cm;
353 const G4ThreeVector xPost = post->GetPosition() / CLHEP::cm;
354 const Double_t tPre = pre->GetGlobalTime() / CLHEP::s;
355 const Double_t tPost = post->GetGlobalTime() / CLHEP::s;

357 const Double_t s =

359 const Double_t f = std::clamp(s / delta_s, 0.0, 1.0);
360 const G4ThreeVector x = xPre + f * (xPost - xPre);
361 const G4ThreeVector p = pPre + f * (pEnd - pPre);
362 TrackBuffer clone;

368 clone.x = x.x();
369 clone.y = x.y();

372 clone.polx = polX;
373 clone.poly = polY;
374 clone.polz = polZ;
375 clone.weight = cloneWeight;
376 clone.parentID = trueParentId;
377 clone.splitSet = splitSet;
378 clone.splitRole = ShipMCTrack::kSplitDecay;
379 fSecondaryBuffer.push_back(clone);
380 }
381 part->SetWeight(trackWeight * (1.0 - P_decay));
