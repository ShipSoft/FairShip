31Acts::Result<Acts::Vector3> ActsShipField::getField(
32 const Acts::Vector3& position, Cache& /*cache*/) const {
33 const Acts::Vector3 shipPosition = ActsFrame::positionToShip(position);
34 const double x[3] = {shipPosition.x(), shipPosition.y(), shipPosition.z()};
35 double b[3] = {0., 0., 0.};
36 m_field->Field(x, b);
37 const Acts::Vector3 shipField(b[0], b[1], b[2]);
38 return Acts::Result<Acts::Vector3>::success(ActsFrame::toActsRotation() *
39 shipField * kKiloGauss);
40}
