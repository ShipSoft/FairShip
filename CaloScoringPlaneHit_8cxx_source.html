15CaloScoringPlaneHit::CaloScoringPlaneHit(const CaloScoringPlanePoint* p,
16 Double_t t0, Double_t pos_res,
17 Double_t time_res, Int_t pid,
18 TRandom* rng)
19 : SHiP::DetectorHit() {
20 if (!p) {
21 LOG(error) << "CaloScoringPlaneHit: null CaloScoringPlanePoint pointer";
22 return;
23 }
24 TRandom& random = rng ? *rng : *gRandom;
25 fDetectorID = p->GetDetectorID();
26 fX = pos_res > 0. ? random.Gaus(p->GetX(), pos_res) : p->GetX();
27 fY = pos_res > 0. ? random.Gaus(p->GetY(), pos_res) : p->GetY();
28 fZ = p->GetZ();
29 const Double_t t = p->GetTime() + t0;
30 fTime = time_res > 0. ? random.Gaus(t, time_res) : t;
31 fPx = p->GetPx();
32 fPy = p->GetPy();
33 fPz = p->GetPz();
34 fPid = pid;
35}
