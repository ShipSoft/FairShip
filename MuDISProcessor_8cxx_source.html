25double DISMinLength(const std::string& label, const MuonPath& path) {
26 // Restrict MS to the material tail in accumulated z, with gaps removed.
27 return label.find("MS") != label.npos
28 ? path.GetLengthAtZ(
29 path.GetstartZ() +
30 std::max(0., path.GetZLength() - kMuonShieldTailZ))
31 : 0.;
32}

70void MuDISProcessor::initPythia6() {
71 // set process 1=QCD, 2=DY/others
72 fPythia->SetMSEL(2);
73 // set min hard scale: 2 GeV --->try 1.5 for soft muons ?
74 fPythia->SetPARP(2, fMinPythiaP);
75 // disable decay for those PDGID
76 unsigned hadrons[10] = {211, 321, 130, 310, 3112,
77 3122, 3222, 3312, 3322, 3334};
78 for (unsigned ikf(0); ikf < 10; ++ikf) {
79 int kc = fPythia->Pycomp(hadrons[ikf]);
80 fPythia->SetMDCY(kc, 1, 0);
81 }
82
83 int seed = static_cast<int>(fP6seed % 900000000);
84 LOG(info) << " * Pythia6 seed set to " << fP6seed << " wrapped to " << seed
85 << std::endl;
86 fPythia->SetMRPY(1, seed);
87 // To direct specific verbose Pythia6 output to a special file.
88 fPythia->SetMSTU(11, 11);
89 // For standard output stream it would be:
90 // fPythia->SetMSTU(11, 6);
91}

93void MuDISProcessor::rotate(const TVector3& pvec, const double& theta,
94 const double& phi, TVector3& newp) {
95 // Rotate the daughter particle momentum to align with respect to the muon's
96 // momentum."""
97
98 TRotation rotation;
99 rotation.RotateY(theta); // Rotate around the Y-axis
100 rotation.RotateZ(phi); // Rotate around the Z-axis
101
102 // Apply the rotation to the momentum vector
103 newp = rotation * pvec;
104}

120Bool_t MuDISProcessor::InitFiles(const std::vector<std::string>& fileNames,
121 const int startEvent) {
122 if (fileNames.empty()) {
123 LOG(error) << "MuDISProcessor: no input files provided. "
124 << "Check the -f/--inputFile argument or input file glob.";
125 return kFALSE;
126 }
127 for (const auto& fileName : fileNames) {
128 if (fileName.empty()) {
129 LOG(error) << "MuDISProcessor: received an empty input file name. "
130 << "Check the -f/--inputFile argument.";
131 return kFALSE;
132 }
133 }
134
135 LOG(info) << "Opening input file to find keys " << fileNames.at(0);
136 TFile* testFile = TFile::Open(fileNames.at(0).c_str(), "READ");
137 auto testKeys = testFile ? testFile->GetListOfKeys() : nullptr;
138 if (testKeys == nullptr) {
139 delete testFile;
140 LOG(error) << "MuDISProcessor: Error opening input file " << fileNames.at(0)
141 << ". Check that the path is correct and the file is a readable "
142 "ROOT file.";
143 return kFALSE;
144 }
145 const bool hastree = testKeys->FindObject("cbmsim") != nullptr;
146 testFile->Close();
147 delete testFile;
148
149 if (hastree) {
150 delete ftree;
151 ftree = new TChain("cbmsim");
152 for (auto& f : fileNames) {
153 LOG(info) << "Opening input file " << f;
154 ftree->Add(f.c_str());
155 }
156 int treeEvts = ftree->GetEntries();
157 LOG(info) << "Reading " << treeEvts << " entries.";
158
159 bool ok = finEv.Setup(ftree);
160
161 if (!ok) {
162 LOG(error)
163 << "MuDISProcessor: failed to bind one or more required branches";
164 return kFALSE;
165 }
166 LOG(info) << "MuDISProcessor: Initialization successful.";
167 return kTRUE;
168 }
169 return kFALSE;
170}

178void MuDISProcessor::process_file(const std::vector<std::string>& input,
179 const std::string& output) {
180 Bool_t treeOK = InitFiles(input);
181
182 if (!treeOK) {
183 LOG(error) << " -- Error reading input files: n=" << input.size();
184 if (input.size() > 0) LOG(error) << " first file: " << input[0];
185 return;
186 }
187
188 TFile* outfile = TFile::Open(output.c_str(), "RECREATE");
189 if (!outfile) {
190 LOG(error) << " -- Error creating outputfile: " << output;
191 return;
192 }
193 outfile->cd();
194 fouttree = new TTree(
195 "MuonDIS", "Muon information, DIS products and soft interaction tracks");
196 foutEv.InitTree(fouttree);
197
198 initPythia6();
199
200 Long64_t n = ftree->GetEntries();
201 LOG(info) << " * input tree with " << n << " entries" << std::endl;
202
203 ProcessMuons();
204
205 outfile->cd();
206 fouttree->Write();
207 outfile->Close();
208
209 fGeoProcessor.PrintVolumes();
210}

265void MuDISProcessor::generateDISevents(const std::string& tType,
266 const double& amuonW,
267 const std::string& aLabel,
268 const MuonPath& aPath,
269 MuonDISBranches& aDISBr) {
270 // for Pythia beam, just use the initial muon momentum...
271 //@FIXME AMM - is this OK ? Reinitialising each time will be too heavy...
272 if (aPath.GetNSlices() < 1) {
273 LOG(error)
274 << " --- calling generateDISevents on an empty path. Doing nothing.";
275 return;
276 }
277 aDISBr.nDISevtsGenerated = 0;
278 double P = aPath.GetMomentum(0);
279 LOG(debug) << " ** initial mu momentum: " << (*finEv.MCTrack).front().GetP()
280 << " Pythia init P: " << P;
281
282 if (P < fMinPythiaP) {
283 LOG(info) << " --- calling Pythia initialise with momentum " << P
284 << " in material " << aLabel
285 << " , min value for Pythia is: " << fMinPythiaP
286 << ". Doing nothing.";
287 return;
288 }
289 fPythia->Initialize("FIXT", tType.c_str(), "p+", P); // target = "p+"
290 aDISBr.pPythia = P;
291 bool isProton = true;
292 // print summary of initialisation params
293 // fPythia->Pylist(1);
294
295 // Preserve the shield's material-tail restriction in compact z, then sample
296 // actual segment length so changes of direction do not bias the vertices.
297 const double minLength = DISMinLength(aLabel, aPath);
298 double lastxs = 0;
299 for (int ia(0); ia < fnDIS; ++ia) {
300 LOG(debug) << " ---- Processing DIS event " << ia;
301 //@FIXME AMM - half-way through, we change to neutron target with 50-50 :
302 //---> update to real material ??
303 if (ia == static_cast<int>(fnDIS / 2)) {
304 fPythia->Initialize("FIXT", tType.c_str(), "n0", P); // target = "n0"
305 isProton = false;
306 }
307
308 // Choose a point on the measured segments, excluding spatial jumps.
309 unsigned slice = 0;
310 double realz = aPath.GetZAtLength(
311 gRandom->Uniform(minLength, aPath.GetLength()), slice);
312 double sliceP = aPath.GetMomentum(slice);
313 if (sliceP == 0) {
314 LOG(error) << " --- Slice in z has momentum " << sliceP
315 << ". Skipping DIS event " << ia;
316 continue;
317 }
318
319 fPythia->GenerateEvent();
320 aDISBr.nDISevts++;
321 // clean all but final stable particles
322 fPythia->Pyedit(1);
323
324 lastxs = fPythia->GetPARI(1); // in mb
325 aDISBr.DISxsec.push_back(lastxs); // in mb
326 aDISBr.DIStarget.push_back(isProton);
327
328 aDISBr.DISvz.push_back(realz);
329 aDISBr.DISvx.push_back(aPath.GetX(realz, slice));
330 aDISBr.DISvy.push_back(aPath.GetY(realz, slice));
331 aDISBr.DISvt.push_back(aPath.GetTimeNs(realz, slice));
332
333 double theta = TMath::ACos(aPath.Getpz(slice) / sliceP);
334 // returns phi between -pi and pi
335 double phi = TMath::ATan2(aPath.Getpy(slice), aPath.Getpx(slice));
336
337 unsigned ndaugh = fPythia->GetN();
338 aDISBr.nDISdau.push_back(ndaugh);
339
340 // loop over daughters and rotate in muon input direction
341 for (unsigned itrk(1); itrk < ndaugh + 1; ++itrk) {
342 DISparticle adau;
343 adau.pid = fPythia->GetK(itrk, 2);
344 TVector3 dauP(0, 0, 0);
345 TVector3 indauP(fPythia->GetP(itrk, 1), fPythia->GetP(itrk, 2),
346 fPythia->GetP(itrk, 3));
347 rotate(indauP, theta, phi, dauP);
348 adau.px = dauP.X();
349 adau.py = dauP.Y();
350 adau.pz = dauP.Z();
351 // A spatial rotation leaves the Pythia event-record energy unchanged.
352 adau.E = fPythia->GetP(itrk, 4);
353 aDISBr.DISparticles.push_back(adau);
354 } // loop on daughters
355
356 //
357 } // loop on DIS events
358
359 aDISBr.nDISevtsGenerated = aDISBr.nDISevts;
360
361 // Normalize to the density integrated over the sampled path interval.
362 if (aDISBr.nDISevts > 0)
363 aDISBr.wDIS =
364 amuonW / aDISBr.nDISevts * aPath.GetWeightedDensity(minLength);
365
366 LOG(debug) << " -- path " << aLabel << " muonW " << amuonW
367 << " -- size of DISparticles collections: " << std::endl
368 << " ---- particles: " << aDISBr.DISparticles.size() << std::endl
369 << " ---- nDIS events: " << aDISBr.nDISevts << std::endl
370 << " ---- weightDIS: " << aDISBr.wDIS << std::endl
371 << " ---- xsec: ";
372 for (int i(0); i < aDISBr.nDISevts; ++i) {
373 LOG(debug) << aDISBr.DISxsec[i] << " ";
374 }
375 LOG(debug) << " ---- vertex: ";
376 for (int i(0); i < aDISBr.nDISevts; ++i) {
377 LOG(debug) << "(" << aDISBr.DISvx[i] << "," << aDISBr.DISvy[i] << ","
378 << aDISBr.DISvz[i] << "," << aDISBr.DISvt[i] << ") ";
379 }
380}

382void MuDISProcessor::ProcessMuons() {
383 LOG(info) << " * Start of event loop" << std::endl;
384 fGeoProcessor.ResetDiagnostics();
385
386 if (static_cast<Long64_t>(fstartEvt) >= ftree->GetEntries()) {
387 LOG(error) << " ** Trying to start from event " << fstartEvt
388 << " which is greater than the tree entries: "
389 << ftree->GetEntries() << ". Doing nothing...";
390 return;
391 }
392
393 const Long64_t nEntries =
394 fnEvts >= 0 ? std::min(static_cast<Long64_t>(fstartEvt + fnEvts),
395 ftree->GetEntries())
396 : ftree->GetEntries();
397 LOG(info) << " - Processing event " << fstartEvt << " to event "
398 << nEntries - 1 << std::endl;
399
400 unsigned nplus = 0;
401 unsigned nminus = 0;
402 unsigned skipMu_pmin = 0;
403 unsigned skipMu_acc = 0;
404 unsigned skipEvt = 0;
405 double wplus = 0.;
406 double wminus = 0.;
407 double skipMu_pminW = 0.;
408 double skipMu_accW = 0.;
409 double skipMu_pathW = 0.;
410
411 struct PathTotals {
412 Long64_t muons = 0;
413 double weight = 0.;
414 double length = 0.;
415 double weightedLength = 0.;
416 void Add(double w, double l) {
417 ++muons;
418 weight += w;
419 length += l;
420 weightedLength += w * l;
421 }
422 };
423 PathTotals pathTotals;
424 std::array<PathTotals, nMats> pathTotalsByMat = {};
425
426 for (Long64_t iEvent = static_cast<Long64_t>(fstartEvt); iEvent < nEntries;
427 ++iEvent) {
428 LOG(debug) << " --- Processing event " << iEvent << std::endl;
429 if (iEvent % 100 == 0)
430 LOG(info) << " --- Processing event " << iEvent << std::endl;
431 if (!finEv.PrepareEntry(ftree, iEvent) || ftree->GetEntry(iEvent) <= 0) {
432 LOG(error) << " --- Error reading tree entry: " << iEvent;
433 skipEvt++;
434 continue;
435 }
436
437 if (finEv.MCTrack == nullptr) {
438 skipEvt++;
439 continue;
440 }
441
442 unsigned nTr = (*finEv.MCTrack).size();
443 if (nTr == 0) {
444 skipEvt++;
445 continue;
446 }
447 // interested in the first muon track - for PG setting
448 //@FIXME AMM adapt also to MuonBack input, take all muons?
449 // for now take MCTrack[0]
450 Int_t muIdx = 0;
451
452 ShipMCTrack& track = (*finEv.MCTrack)[static_cast<unsigned>(muIdx)];
453 int pid = track.GetPdgCode();
454 double muW = track.GetWeight();
455
456 double mup = track.GetP();
457
458 if (mup < fMinPythiaP) {
459 LOG(debug) << iEvent << " skipped: muon momentum " << mup
460 << " below min value for Pythia:" << fMinPythiaP;
461 skipMu_pmin++;
462 skipMu_pminW += muW;
463 continue;
464 }
465
466 std::string targetType;
467 if (pid == -13) {
468 nplus++;
469 wplus += muW;
470 targetType = "gamma/mu+";
471 } else if (pid == 13) {
472 nminus++;
473 wminus += muW;
474 targetType = "gamma/mu-";
475 } else {
476 LOG(warning) << iEvent << " skipped: "
477 << " nTracks= " << nTr
478 << " -- 1st track pid not a muon: " << pid << std::endl;
479 skipEvt++;
480 continue;
481 }
482
483 initEvent();
484 fillMCTracks(muIdx);
485 fillUBTHits(muIdx);
486 fillSBTHits(muIdx);
487 fillSSTHits(muIdx);
488 fillTDHits(muIdx);
489
490 // Count events which have no hit in UBT, SBT, SST or TD, muon just
491 // flying out of vessel acceptance never bouncing back... For those, very
492 // numerous in the cudaMu files, skip to optimise processing.
493 //@FIXME AM to do: a study of whether the events created in the last lambda
494 // of MS could still contribute or would hit UBT...
495 if (foutEv.ubtPt.size() == 0 && foutEv.sbtPt.size() == 0 &&
496 foutEv.sstPt.size() == 0 && foutEv.tdPt.size() == 0) {
497 LOG(debug) << " Counting muon event with no hit " << iEvent
498 << " UBT Hits: " << foutEv.ubtPt.size()
499 << ", SBT Hits: " << foutEv.sbtPt.size()
500 << ", SST Hits: " << foutEv.sstPt.size()
501 << ", TD Hits: " << foutEv.tdPt.size() << std::endl;
502 skipMu_acc++;
503 skipMu_accW += muW;
504 // counting, but want to still fill DIS in MS (and UBT detector)...
505 // continue;
506 }
507
508 LOG(debug) << " -- size of hits collections: " << std::endl
509 << " ---- mcTracks: " << foutEv.mcTrks.size() << std::endl
510 << " ---- UBT Hits: " << foutEv.ubtPt.size() << std::endl
511 << " ---- SBT Hits: " << foutEv.sbtPt.size() << std::endl
512 << " ---- SST Hits: " << foutEv.sstPt.size() << std::endl;
513
514 // retrieve a map of material label, with same density, and lengths, and
515 // [zin,zout] ranges
516 if (!fGeoProcessor.initialise(foutEv)) {
517 skipEvt++;
518 skipMu_pathW += muW;
519 continue;
520 }
521 std::map<std::string, MuonPath>& lPathMap = fGeoProcessor.FillMuonPath();
522
523 if (lPathMap.size() == 0) {
524 LOG(error) << " -- No elements in path... Not doing anything...";
525 skipEvt++;
526 skipMu_pathW += muW;
527 continue;
528 }
529
530 // Persist the intervals used for DIS vertex sampling, including the MS
531 // restriction. Keep REST separately, but exclude it from the total.
532 for (unsigned i = 0; i < nMats; ++i) {
533 const auto path = lPathMap.find(MatTypeStr[i].Data());
534 if (path == lPathMap.end() || path->second.GetNSlices() < 1) continue;
535 // Crossed but not generated (including REST) is distinct from unvisited.
536 foutEv.br[i].nDISevtsGenerated = 0;
537 foutEv.pathLengthByMat[i] =
538 path->second.GetLength() - DISMinLength(path->first, path->second);
539 if (i != REST) foutEv.pathLength += foutEv.pathLengthByMat[i];
540 if (foutEv.pathLengthByMat[i] > 0.)
541 pathTotalsByMat[i].Add(muW, foutEv.pathLengthByMat[i]);
542 }
543
544 // loop over the map, and do nDIS event in each element, with weight
545 // length*density. That way, do only once the calculation of the path, and
546 // plenty of DIS in each material. fill a branch with weight = path
547 // length*density.
548 // Discarding REST: many muons go through CONCRETE,no need to record DIS
549 // there....
550 for (unsigned i(0); i < nMats - 1; ++i) {
551 if (lPathMap.find(MatTypeStr[i].Data()) != lPathMap.end())
552 generateDISevents(targetType, muW, MatTypeStr[i].Data(),
553 lPathMap.find(MatTypeStr[i].Data())->second,
554 foutEv.br[i]);
555 }
556
557 fouttree->Fill();
558 pathTotals.Add(muW, foutEv.pathLength);
559
560 } // loop on events
561
562 LOG(info) << "Found " << nplus << " mu+ and " << nminus << " mu-."
563 << std::endl
564 << "Skipped: " << skipEvt << " events and " << skipMu_pmin
565 << " muons with too low p" << std::endl
566 << "Counted: " << skipMu_acc
567 << " muons with no recorded hits in UBT,SBT,SST,TD.";
568 LOG(info) << "Weighted counts (sum MCTrack[0].fW): " << wplus << " mu+ and "
569 << wminus << " mu- (after momentum cut)." << std::endl
570 << "Skipped (sum MCTrack[0].fW): " << skipMu_pminW
571 << " with too low p, " << skipMu_pathW
572 << " muons with invalid or empty paths." << std::endl
573 << "Counted (sum MCTrack[0].fW): " << skipMu_accW
574 << " muons with no recorded hits in UBT,SBT,SST,TD.";
575 fGeoProcessor.PrintDiagnostics();
576 // Print additive quantities so jobs can be combined before taking means.
577 LOG(info) << std::setprecision(17)
578 << "Muon path totals (cm, excluding REST): output muons = "
579 << pathTotals.muons << ", sum(fW) = " << pathTotals.weight
580 << ", sum(L) = " << pathTotals.length
581 << ", sum(fW*L) = " << pathTotals.weightedLength;
582 for (unsigned i = 0; i < nMats; ++i) {
583 const auto& totals = pathTotalsByMat[i];
584 LOG(info) << std::setprecision(17) << "Muon path totals " << MatTypeStr[i]
585 << " (cm): crossing muons = " << totals.muons
586 << ", sum(fW) = " << totals.weight
587 << ", sum(L) = " << totals.length
588 << ", sum(fW*L) = " << totals.weightedLength;
589 }
