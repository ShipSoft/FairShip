41void ShipTargetStation::ConstructGeometry() {
42 TGeoVolume* top = gGeoManager->GetTopVolume();
43
44 ShipGeo::InitMedium("tungsten");
45 TGeoMedium* tungsten = gGeoManager->GetMedium("tungsten");
46 ShipGeo::InitMedium("tantalum");
47 TGeoMedium* tantalum = gGeoManager->GetMedium("tantalum");
48 ShipGeo::InitMedium("molybdenum");
49 TGeoMedium* mo = gGeoManager->GetMedium("molybdenum");
50 ShipGeo::InitMedium("iron");
51 TGeoMedium* iron = gGeoManager->GetMedium("iron");
52 ShipGeo::InitMedium("steel316L");
53 TGeoMedium* steel316L = gGeoManager->GetMedium("steel316L");
54 ShipGeo::InitMedium("copper");
55 TGeoMedium* copper = gGeoManager->GetMedium("copper");
56
57 ShipGeo::InitMedium("Inconel718");
58 TGeoMedium* inc718 = gGeoManager->GetMedium("Inconel718");
59
60 ShipGeo::InitMedium("vacuums");
61 TGeoMedium* vacuum = gGeoManager->GetMedium("vacuums");
62
63 double He_T = 273.15 + fHeT; // in K, use average 90 degrees C.
64 double He_P = 1.6e6 * 6.241509e3; // 16 bar in MeV/mm3
65
66 std::ostringstream lHename;
67 lHename << "PressurisedHe" << fHeT;
68 ShipGeo::InitMedium(lHename.str().c_str());
69 TGeoMedium* pressurised_He = gGeoManager->GetMedium(lHename.str().c_str());
70
71 // CAMM- dirty fix to have pressure and temperature correct for Geant4.
72 // Should fix this properly in future...
73 TGeoMaterial* fixedCooler = pressurised_He->GetMaterial();
74 fixedCooler->SetTemperature(He_T);
75 fixedCooler->SetPressure(He_P);
76 TGeoMedium* cooler = pressurised_He;
77
78 LOG(info) << "-- Target cooler: " << cooler->GetName()
79 << " T=" << cooler->GetMaterial()->GetTemperature()
80 << " K, P=" << cooler->GetMaterial()->GetPressure()
81 << " MeV/mm3, Density=" << cooler->GetMaterial()->GetDensity();
82
83 TGeoVolume* tTarget = new TGeoVolumeAssembly("TargetArea");
84
85 if (fDesign == 1) {
86 // Target vessel - inner dimensions for diameter, shifts and length
87 double vessel_thickness = 8 * mm;
88 double vessel_diameter = fDiameter + 150 * mm; // Inner diameter of vessel
89 double vessel_shift =
90 62 *
91 mm; // Shift in z at y=0 (round shape not implemented for vessel lids)
92 double vessel_length = fTargetLength + 2 * vessel_shift;
93
94 TGeoVolume* vessel;
95 vessel = gGeoManager->MakeTube("TargetVessel", inc718, vessel_diameter / 2.,
96 vessel_diameter / 2. + vessel_thickness,
97 vessel_length / 2.);
98 vessel->SetLineColor(28);
99 tTarget->AddNode(
100 vessel, 1,
101 new TGeoTranslation(0, 0, -1. * vessel_shift + vessel_length / 2.));
102 // Front face
103 vessel = gGeoManager->MakeTube("TargetVesselFront", inc718, 0,
104 vessel_diameter / 2. + vessel_thickness,
105 vessel_thickness / 2.);
106 vessel->SetLineColor(28);
107 tTarget->AddNode(
108 vessel, 1,
109 new TGeoTranslation(0, 0, -1. * vessel_shift - vessel_thickness / 2.));
110 // Back face
111 vessel = gGeoManager->MakeTube("TargetVesselBack", inc718, 0,
112 vessel_diameter / 2. + vessel_thickness,
113 vessel_thickness / 2.);
114 vessel->SetLineColor(28);
115 tTarget->AddNode(
116 vessel, 1,
117 new TGeoTranslation(
118 0, 0, -1. * vessel_shift + vessel_length + vessel_thickness / 2.));
119 // He inside
120 vessel = gGeoManager->MakeTube("HeVolume", cooler, 0, vessel_diameter / 2.,
121 vessel_length / 2.);
122 vessel->SetLineColor(7);
123
124 // Steel enclosure around target inside He volume
125 // Inner radius must be larger than target radius (fDiameter/2) to avoid
126 // overlaps
127 double enclosure_clearance =
128 0.1 * mm; // Clearance between target and enclosure
129 double enclosure_inner_radius = fDiameter / 2. + enclosure_clearance;
130 double enclosure_thickness = 66.9 * mm; // Updated to match BDF model
131 double enclosure_outer_radius =
132 enclosure_inner_radius + enclosure_thickness;
133 double enclosure_cutout_width_x = 160 * mm; // Width in x direction
134 double enclosure_cutout_width_y = 280 * mm; // Width in y direction
135 double enclosure_length = fTargetLength;
136
137 [[maybe_unused]] auto enclosure_outer_tube =
138 new TGeoTube("enclosure_outer_tube", enclosure_inner_radius,
139 enclosure_outer_radius, enclosure_length / 2.);
140 [[maybe_unused]] auto enclosure_cutout_box =
141 new TGeoBBox("enclosure_cutout_box", enclosure_cutout_width_x / 2.,
142 enclosure_cutout_width_y / 2., enclosure_length / 2.);
143 auto enclosure_shape = new TGeoCompositeShape(
144 "enclosure_shape", "enclosure_outer_tube - enclosure_cutout_box");
145 auto enclosure =
146 new TGeoVolume("target_enclosure", enclosure_shape, steel316L);
147 enclosure->SetLineColor(46); // Reddish color for steel
148 vessel->AddNode(
149 enclosure, 1,
150 new TGeoTranslation(
151 0, 0, -vessel_length / 2. + vessel_shift + enclosure_length / 2.));
152
153 // now place target inside He volume (and inside steel enclosure)
154 // Using nested volumes: tantalum cladding contains target core
155 double cladding_width = 1.5 * mm;
156
157 Double_t zPos = 0.;
158 unsigned slots = fnS;
159 if (slots > 0) slots = slots - 1;
160
161 TGeoVolume* claddedTarget;
162 TGeoVolume* targetCore;
163 // Double_t zPos = fTargetZ - fTargetLength/2.;
164 for (unsigned i = 0; i < fnS; i++) { // loop on layers
165 TString nmi_cladded = "CladdedTarget_";
166 nmi_cladded += i + 1;
167 TString nmi_core = "TargetCore_";
168 nmi_core += i + 1;
169
170 TGeoMedium* material = nullptr;
171 if (fM.at(i) == "molybdenum") {
172 material = mo;
173 } else if (fM.at(i) == "tungsten") {
174 material = tungsten;
175 }
176
177 // Create outer cladded volume (tantalum, full dimensions)
178 claddedTarget = gGeoManager->MakeTube(nmi_cladded, tantalum, 0.,
179 fDiameter / 2., fL.at(i) / 2.);
180 claddedTarget->SetLineColor(8); // Green for tantalum
181
182 // Create inner target core (W or Mo, reduced dimensions)
183 // Positioned at centre (z=0) of cladded volume - tantalum fills the gaps
184 // automatically
185 targetCore = gGeoManager->MakeTube(nmi_core, material, 0.,
186 fDiameter / 2. - cladding_width,
187 (fL.at(i) - 2 * cladding_width) / 2.);
188 if (fM.at(i) == "molybdenum") {
189 targetCore->SetLineColor(28);
190 } else {
191 targetCore->SetLineColor(38);
192 }; // silver/blue
193
194 // Nest core inside cladding (at centre, z=0 in cladded volume's
195 // coordinate system)
196 claddedTarget->AddNode(targetCore, 1, new TGeoTranslation(0, 0, 0));
197
198 // Place complete cladded target in He volume
199 vessel->AddNode(
200 claddedTarget, 1,
201 new TGeoTranslation(
202 0, 0, -vessel_length / 2. + vessel_shift + zPos + fL.at(i) / 2.));
203
204 if (i < slots) {
205 // slits will already be filled with He, no need to define volume
206 zPos += fL.at(i) + fG.at(i);
207 } else {
208 zPos += fL.at(i);
209 }
210 } // loop on layers
211
212 // now add the He+target to the target area.
213 tTarget->AddNode(
214 vessel, 1,
215 new TGeoTranslation(0, 0, -1. * vessel_shift + vessel_length / 2.));
216 } else if (fDesign == 2) {
217 // Design 2: 2026 BDF target, from CATIA model ST1A07710_01_AB.02.
218 // Simplified to axisymmetric shapes. All z coordinates below are measured
219 // from the front face of the first disk, which coincides with the local
220 // z = 0 of the TargetArea assembly (global z = 0).
221 if (fDiameter2 <= 0) {
222 LOG(fatal) << "Target design 2 requires the last disk diameter, "
223 "use SetLastDiskDiameter()";
224 }
225
226 // He container: encloses steel core, jacket, flanges, beam window,
227 // cover-plate bore rings and rear endcap
228 const double he_zmin = -37.8 * mm; // upstream face of the cover plate
229 const double he_zmax = 1509.7 * mm; // downstream end of rear endcap
230 const double he_rmax = 237 * mm; // rear flange outer radius
231 const double he_zc = (he_zmin + he_zmax) / 2.;
232 TGeoVolume* heVolume = gGeoManager->MakeTube("HeVolume", cooler, 0, he_rmax,
233 (he_zmax - he_zmin) / 2.);
234 heVolume->SetLineColor(7);
235
236 // Place a daughter so that it spans [z0, z1] on the axis
237 auto placeAt = [he_zc](double z0, double z1) {
238 return new TGeoTranslation(0, 0, (z0 + z1) / 2. - he_zc);
239 };
240
241 // W disks, no cladding; the last disk (rear block) is larger.
242 // Added first so that the first daughter of HeVolume starts at the
243 // target front face (FixedTargetGenerator fallback z extraction).
244 Double_t zPos = 0.;
245 for (unsigned i = 0; i < fnS; i++) {
246 TString nmi_core = "TargetCore_";
247 nmi_core += i + 1;
248 TGeoMedium* material = nullptr;
249 if (fM.at(i) == "molybdenum") {
250 material = mo;
251 } else if (fM.at(i) == "tungsten") {
252 material = tungsten;
253 }
254 double radius = (i == fnS - 1) ? fDiameter2 / 2. : fDiameter / 2.;
255 TGeoVolume* targetCore =
256 gGeoManager->MakeTube(nmi_core, material, 0., radius, fL.at(i) / 2.);
257 targetCore->SetLineColor(fM.at(i) == "molybdenum" ? 28 : 38);
258 heVolume->AddNode(targetCore, 1, placeAt(zPos, zPos + fL.at(i)));
259 zPos += fL.at(i) + fG.at(i); // last gap is 0
260 }
261
262 // Steel core: the two clamp halves around the disks, modelled as one
263 // axisymmetric piece with the He cooling grooves subtracted. The
264 // composite shape is built in a frame where z = 0 is the target front
265 // face; boolean members are placed with absolute-z translations.
266 const double core_zmin = 7.2 * mm;
267 const double core_zmax = 1443.7 * mm;
268 const double core_rout = 207 * mm;
269 const double bore_r1 = fDiameter / 2.; // matches disks 1..nS-1
270 const double bore_r2 = fDiameter2 / 2.; // matches the last disk
271 const double bore_step_z = 1005 * mm; // bore radius change
272 const double core_front_rout = 195 * mm; // front, inside the flange
273 const double core_front_zmax = 55.2 * mm; // end of front step
274 const double core_rear_rout = 190 * mm; // rear, inside rear flange
275 const double core_rear_zmin = 1233.7 * mm; // start of rear step
276 const double groove_rmax = 153 * mm;
277 const double groove_phi = 61.; // arc width in degrees, centred vertically
278 const double margin = 5 * mm; // avoid coincident boolean surfaces
279
280 std::ostringstream core_expr;
281 auto addTube = [&core_expr](const TString& name, double rmin, double rmax,
282 double z0, double z1, bool subtract) {
283 new TGeoTube(name, rmin, rmax, (z1 - z0) / 2.);
284 auto trans = new TGeoTranslation(name + "_tr", 0, 0, (z0 + z1) / 2.);
285 trans->RegisterYourself();
286 if (subtract) {
287 core_expr << " - ";
288 }
289 core_expr << name << ":" << name << "_tr";
290 };
291 auto addGroove = [&core_expr](const TString& name, double rmin, double rmax,
292 double z0, double z1, double phi1,
293 double phi2) {
294 new TGeoTubeSeg(name, rmin, rmax, (z1 - z0) / 2., phi1, phi2);
295 auto trans = new TGeoTranslation(name + "_tr", 0, 0, (z0 + z1) / 2.);
296 trans->RegisterYourself();
297 core_expr << " - " << name << ":" << name << "_tr";
298 };
299
300 addTube("tgt_core_base", bore_r1, core_rout, core_zmin, core_zmax, false);
301 addTube("tgt_core_front_step", core_front_rout, core_rout + margin,
302 core_zmin - margin, core_front_zmax, true);
303 addTube("tgt_core_rear_step", core_rear_rout, core_rout + margin,
304 core_rear_zmin, core_zmax + margin, true);
305 addTube("tgt_core_rear_bore", bore_r1 - margin, bore_r2, bore_step_z,
306 core_zmax + margin, true);
307
308 // Serpentine He cooling grooves: 61 degree arcs in the bore, staggered
309 // between the upper and lower half; He flows between them through the
310 // inter-disk slits.
311 const std::vector<std::pair<double, double>> grooves_top = {{12.2, 57.0},
312 {95.0, 153.0},
313 {191.0, 257.0},
314 {300.0, 464.0},
315 {559.0, 933.2}};
316 const std::vector<std::pair<double, double>> grooves_bottom = {
317 {45.0, 104.7}, {142.7, 200.7}, {238.7, 334.7}, {385.7, 1005.0}};
318 int grooveId = 0;
319 for (const auto& [z0, z1] : grooves_top) {
320 addGroove(TString::Format("tgt_groove_%d", grooveId++), bore_r1 - margin,
321 groove_rmax, z0 * mm, z1 * mm, 90. - groove_phi / 2.,
322 90. + groove_phi / 2.);
323 }
324 for (const auto& [z0, z1] : grooves_bottom) {
325 addGroove(TString::Format("tgt_groove_%d", grooveId++), bore_r1 - margin,
326 groove_rmax, z0 * mm, z1 * mm, 270. - groove_phi / 2.,
327 270. + groove_phi / 2.);
328 }
329 // Groove around the rear block (upper half only), ends 2 mm before the
330 // core rear face
331 addGroove("tgt_groove_rear", bore_r2 - margin, 182 * mm, bore_step_z,
332 1441.7 * mm, 90. - 49. / 2., 90. + 49. / 2.);
333
334 auto core_shape =
335 new TGeoCompositeShape("target_core_shape", core_expr.str().c_str());
336 auto core = new TGeoVolume("target_core_steel", core_shape, steel316L);
337 core->SetLineColor(46); // Reddish color for steel
338 heVolume->AddNode(core, 1, new TGeoTranslation(0, 0, -he_zc));
339
340 // Outer jacket tube with front and rear flanges; the gap between the
341 // core (r 207) and the jacket (r 217) is a He annulus. The flange bore
342 // (r 195) matches the core front step, so the flange/jacket boundary is
343 // placed at the end of the step (CAD: 64.2 mm, with a stepped flange
344 // bore).
345 const double jacket_zmin = core_front_zmax;
346 const double jacket_zmax = 1198.7 * mm;
347 auto jacket =
348 gGeoManager->MakeTube("target_jacket", steel316L, 217 * mm, 225 * mm,
349 (jacket_zmax - jacket_zmin) / 2.);
350 jacket->SetLineColor(46);
351 heVolume->AddNode(jacket, 1, placeAt(jacket_zmin, jacket_zmax));
352 // The flange proper starts at the cover-plate step; upstream of it the
353 // outer radius is the r 195 nose that slides into the cover-plate bore
354 const double flange_front_zmin = -23.8 * mm;
355 const double nose_zmin = -35.8 * mm; // upstream face of the vessel
356 auto flangeFront =
357 gGeoManager->MakeTube("target_flange_front", steel316L, core_front_rout,
358 225 * mm, (jacket_zmin - flange_front_zmin) / 2.);
359 flangeFront->SetLineColor(46);
360 heVolume->AddNode(flangeFront, 1, placeAt(flange_front_zmin, jacket_zmin));
361 // Beam window: dished membrane closing the bore, 8 mm of steel on the
362 // beam axis (z -33.5..-25.5); the dishing is simplified to a flat disc,
363 // preserving the on-axis material budget. The r 141-195 nose ring
364 // connects the window rim to the flange.
365 const double window_rmax = 141 * mm;
366 auto window = gGeoManager->MakeTube("target_front_window", steel316L, 0,
367 window_rmax, (33.5 - 25.5) * mm / 2.);
368 window->SetLineColor(46);
369 heVolume->AddNode(window, 1, placeAt(-33.5 * mm, -25.5 * mm));
370 auto nose = gGeoManager->MakeTube("target_front_nose", steel316L,
371 window_rmax, core_front_rout,
372 (flange_front_zmin - nose_zmin) / 2.);
373 nose->SetLineColor(46);
374 heVolume->AddNode(nose, 1, placeAt(nose_zmin, flange_front_zmin));
375 // Cover plate (600 x 650 x 20 mm, stepped bore r 196/226): the part
376 // within the He container radius is modelled as two rings here; the
377 // rectangular remainder is added to the TargetArea below
378 auto coverRing1 =
379 gGeoManager->MakeTube("target_cover_ring1", steel316L, 196 * mm,
380 he_rmax, (flange_front_zmin - he_zmin) / 2.);
381 coverRing1->SetLineColor(46);
382 heVolume->AddNode(coverRing1, 1, placeAt(he_zmin, flange_front_zmin));
383 auto coverRing2 =
384 gGeoManager->MakeTube("target_cover_ring2", steel316L, 226 * mm,
385 he_rmax, (-17.8 * mm - flange_front_zmin) / 2.);
386 coverRing2->SetLineColor(46);
387 heVolume->AddNode(coverRing2, 1, placeAt(flange_front_zmin, -17.8 * mm));
388 const double flange_rear_zmax = 1263.7 * mm;
389 auto flangeRear =
390 gGeoManager->MakeTube("target_flange_back", steel316L, 225 * mm,
391 he_rmax, (flange_rear_zmax - jacket_zmax) / 2.);
392 flangeRear->SetLineColor(46);
393 heVolume->AddNode(flangeRear, 1, placeAt(jacket_zmax, flange_rear_zmax));
394
395 // Rear endcap: 8 mm shell, cylindrical section closed by a domed head,
396 // approximated by three cone segments and an apex disc
397 const double endcap_zmin = flange_rear_zmax;
398 auto endcapTube =
399 gGeoManager->MakeTube("target_endcap_tube", steel316L, 229 * mm,
400 237 * mm, (1413.7 * mm - endcap_zmin) / 2.);
401 heVolume->AddNode(endcapTube, 1, placeAt(endcap_zmin, 1413.7 * mm));
402 auto endcapDome1 = gGeoManager->MakeCone(
403 "target_endcap_dome1", steel316L, (1450. - 1413.7) * mm / 2., 229 * mm,
404 237 * mm, 212 * mm, 223 * mm);
405 heVolume->AddNode(endcapDome1, 1, placeAt(1413.7 * mm, 1450. * mm));
406 auto endcapDome2 = gGeoManager->MakeCone(
407 "target_endcap_dome2", steel316L, (1480. - 1450.) * mm / 2., 212 * mm,
408 223 * mm, 142 * mm, 166.5 * mm);
409 heVolume->AddNode(endcapDome2, 1, placeAt(1450. * mm, 1480. * mm));
410 auto endcapDome3 = gGeoManager->MakeCone("target_endcap_dome3", steel316L,
411 (1505. - 1480.) * mm / 2.,
412 142 * mm, 166.5 * mm, 0, 67 * mm);
413 heVolume->AddNode(endcapDome3, 1, placeAt(1480. * mm, 1505. * mm));
414 auto endcapCap =
415 gGeoManager->MakeTube("target_endcap_cap", steel316L, 0, 67 * mm,
416 (he_zmax - 1505. * mm) / 2.);
417 heVolume->AddNode(endcapCap, 1, placeAt(1505. * mm, he_zmax));
418 for (auto* v :
419 {endcapTube, endcapDome1, endcapDome2, endcapDome3, endcapCap}) {
420 v->SetLineColor(46);
421 }
422
423 // Cover plate remainder outside the He container: rectangular plate,
424 // asymmetric about the beam axis (x +-300, y -400..+250), with the
425 // central hole covering the He container; the stepped bore is modelled
426 // by the rings inside HeVolume above
427 const double cover_y_offset = -75 * mm; // plate centre below the axis
428 [[maybe_unused]] auto cover_box =
429 new TGeoBBox("target_cover_box", 300 * mm, 325 * mm, 10 * mm);
430 [[maybe_unused]] auto cover_hole =
431 new TGeoTube("target_cover_hole", 0, he_rmax, 11 * mm);
432 auto cover_hole_shift =
433 new TGeoTranslation("target_cover_hole_shift", 0, -cover_y_offset, 0);
434 cover_hole_shift->RegisterYourself();
435 auto cover_shape = new TGeoCompositeShape(
436 "target_cover_shape",
437 "target_cover_box - target_cover_hole:target_cover_hole_shift");
438 auto coverPlate =
439 new TGeoVolume("target_cover_plate", cover_shape, steel316L);
440 coverPlate->SetLineColor(46);
441 tTarget->AddNode(coverPlate, 1,
442 new TGeoTranslation(0, cover_y_offset, -27.8 * mm));
443
444 // add the He + target to the target area
445 tTarget->AddNode(heVolume, 1, new TGeoTranslation(0, 0, he_zc));
446 } else {
447 LOG(fatal) << "Unknown target design " << fDesign << ", expected 1 or 2";
448 }
449
450 // Proximity shielding
451
452 double start_of_target = fTargetZ - fTargetLength / 2.;
453 if (fShieldingReferenceLength <= 0) {
454 LOG(fatal) << "Shielding reference length not set, "
455 "use SetShieldingReferenceLength()";
456 }
457 // Make shielding independent of actual target length
458 fTargetLength = fShieldingReferenceLength;
459 double shielding_width = 1600 * mm;
460 double shielding_length = 3000 * mm;
461 double proximity_shielding_height = 1126 * mm;
462 double proximity_shielding_thickness = 250 * mm;
463 double proximity_shielding_thickness_front = 550 * mm;
464 double proximity_shielding_hole_diameter = 200 * mm;
465 double proximity_shielding_hole_height = 735 * mm;
466 double proximity_shielding_distance_after_target = 96.1 * mm;
467 double top_shielding_height = 600 * mm;
468 double bottom_shielding_height = 545 * mm;
469 double shielding_position = start_of_target + fTargetLength +
470 proximity_shielding_distance_after_target +
471 proximity_shielding_thickness -
472 shielding_length / 2;
473 double target_box_shift = 14.45 * cm;
474 [[maybe_unused]] auto proximity_shielding_envelope =
475 new TGeoBBox("proximity_shielding_envelope", shielding_width / 2,
476 proximity_shielding_height / 2, shielding_length / 2);
477 [[maybe_unused]] auto proximity_shielding_inner = new TGeoBBox(
478 "proximity_shielding_inner",
479 shielding_width / 2 - proximity_shielding_thickness,
480 proximity_shielding_height / 2,
481 (shielding_length -
482 (proximity_shielding_thickness_front + proximity_shielding_thickness)) /
483 2);
484 [[maybe_unused]] auto proximity_shielding_hole = new TGeoTube(
485 "proximity_shielding_hole", 0, proximity_shielding_hole_diameter / 2,
486 proximity_shielding_thickness_front / 2);
487 auto proximity_shielding_inner_shift = new TGeoTranslation(
488 "proximity_shielding_inner_shift", 0, 0,
489 (proximity_shielding_thickness_front - proximity_shielding_thickness) /
490 2);
491 proximity_shielding_inner_shift->RegisterYourself();
492 auto proximity_shielding_hole_shift = new TGeoTranslation(
493 "proximity_shielding_hole_shift", 0,
494 proximity_shielding_hole_height - proximity_shielding_height / 2,
495 -shielding_length / 2 + proximity_shielding_thickness_front / 2);
496 proximity_shielding_hole_shift->RegisterYourself();
497 auto proximity_shielding_shape = new TGeoCompositeShape(
498 "proximity_shielding_shape",
499 "proximity_shielding_envelope"
500 "- proximity_shielding_inner:proximity_shielding_inner_shift"
501 "- proximity_shielding_hole:proximity_shielding_hole_shift");
502 auto proximity_shielding =
503 new TGeoVolume("proximity_shielding", proximity_shielding_shape, copper);
504 auto proximity_shielding_centre = new TGeoTranslation(
