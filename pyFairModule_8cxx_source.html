11std::vector<std::string> pyFairModule::SensitiveVolumeNames(
12 const std::string& module) {
13 std::vector<std::string> names;
14 names.reserve(fAllSensitiveVolumes.size());
15 for (const FairVolume* volume : fAllSensitiveVolumes) {
16 if (!volume) {
17 continue;
18 }
19 if (!module.empty() &&
20 (!volume->GetModule() || module != volume->GetModule()->GetName())) {
21 continue;
22 }
23 std::string name = volume->GetName();
24 // Same cut as in FairDetector::Initialize()
25 const auto pos = name.find('#', 1);
26 if (pos != std::string::npos && pos > 1) {
27 name.resize(pos);
28 }
29 names.push_back(name);
30 }
31 return names;
32}
