4"""Check that the ECN3 cross-section of the geometry configuration is that of ShipCave.
5
6ShipCave builds the cavern from the constants in passive/ShipCavernLayout.h,
7while geometry_config places the UBT and its scoring planes from its own copy
8of the ECN3 floor and side walls. Nothing else ties the two together, so if one
9changed without the other, the UBT would stand off the floor or the scoring
10planes would cut into the walls without any error.
11"""

30def layout_constants() -> dict[str, float]:
31 """The constants of ShipCavernLayout.h, in FairShip length units."""
32 pattern = re.compile(r"(k\w+)\s*=\s*(-?[\d.]+)\s*\*\s*ShipUnit::m\b")
33 return {name: float(value) * u.m for name, value in pattern.findall(LAYOUT.read_text())}
34
35
36@pytest.mark.parametrize("quantity", QUANTITIES)
37def test_ecn3_cross_section_matches_ship_cave(quantity):
38 constants = layout_constants()
39 cave = geometry_config.create_config().cave
40 assert cave[quantity] == pytest.approx(constants[QUANTITIES[quantity]], abs=1.0e-6 * u.cm)
